Saatnya menggabungkan semua yang sudah dipelajari menjadi satu program utuh: Sistem Nilai dan IPK Mahasiswa. Program ini mengambil data KHS dari "server" (simulasi), memvalidasi data yang rusak, lalu menghitung IPK, predikat, dan mata kuliah yang perlu diulang.

Spesifikasi

  • Setiap mata kuliah punya kode, nama, SKS (1 sampai 6), dan nilai angka (0 sampai 100).
  • Nilai angka dikonversi ke nilai huruf dan bobot: A=4.0, AB=3.5, B=3.0, BC=2.5, C=2.0, D=1.0, E=0.0. Nilai di bawah C wajib diulang.
  • IPK = jumlah (SKS × bobot) dibagi total SKS.
  • Predikat: ≥ 3.51 Dengan Pujian, ≥ 3.01 Sangat Memuaskan, ≥ 2.76 Memuaskan, selain itu Cukup.
  • Data dari server bisa rusak. Data yang rusak dilewati dan dilaporkan sebagai peringatan, tanpa membuat program berhenti.

Kode lengkap

Simpan sebagai satu file lalu jalankan dengan dart run sistem_nilai.dart. Komentar bernomor menandai bagian-bagian program.

📄 sistem_nilai.dart

// ===== 1. ENUM: skala nilai huruf (enhanced enum) =====
enum NilaiHuruf {
  a('A', 4.0, 85),
  ab('AB', 3.5, 80),
  b('B', 3.0, 70),
  bc('BC', 2.5, 65),
  c('C', 2.0, 60),
  d('D', 1.0, 50),
  e('E', 0.0, 0);

  const NilaiHuruf(this.label, this.bobot, this.batasBawah);

  final String label;
  final double bobot;
  final int batasBawah;

  bool get lulus => bobot >= 2.0;

  static NilaiHuruf dariAngka(double nilai) =>
      values.firstWhere((h) => nilai >= h.batasBawah);
}

// ===== 2. EXCEPTION kustom =====
class DataNilaiException implements Exception {
  final String pesan;
  DataNilaiException(this.pesan);

  @override
  String toString() => 'DataNilaiException: $pesan';
}

// ===== 3. CLASS MataKuliah (constructor, validasi, factory, getter) =====
class MataKuliah {
  final String kode;
  final String nama;
  final int sks;
  final double nilaiAngka;

  MataKuliah({
    required this.kode,
    required this.nama,
    required this.sks,
    required this.nilaiAngka,
  }) {
    if (sks < 1 || sks > 6) {
      throw DataNilaiException('SKS $kode harus 1-6');
    }
    if (nilaiAngka < 0 || nilaiAngka > 100) {
      throw DataNilaiException('Nilai $kode di luar rentang 0-100');
    }
  }

  factory MataKuliah.fromJson(Map<String, dynamic> json) {
    if (json
        case {
          'kode': String kode,
          'nama': String nama,
          'sks': int sks,
          'nilai': num nilai,
        }) {
      return MataKuliah(
        kode: kode,
        nama: nama,
        sks: sks,
        nilaiAngka: nilai.toDouble(),
      );
    }
    throw DataNilaiException('Format data ${json['kode']} tidak sesuai');
  }

  NilaiHuruf get huruf => NilaiHuruf.dariAngka(nilaiAngka);
  double get mutu => sks * huruf.bobot;
}

// ===== 4. EXTENSION =====
extension FormatAngka on double {
  String get duaDesimal => toStringAsFixed(2);
}

// ===== 5. SEALED CLASS: hasil pemuatan data =====
sealed class HasilMuat {}

class MuatBerhasil extends HasilMuat {
  final List<MataKuliah> data;
  final List<String> peringatan;
  MuatBerhasil(this.data, this.peringatan);
}

class MuatGagal extends HasilMuat {
  final String pesan;
  MuatGagal(this.pesan);
}

// ===== 6. CLASS Mahasiswa (private, getter, koleksi, record) =====
class Mahasiswa {
  final String nim;
  final String nama;
  final List<MataKuliah> _khs = [];

  Mahasiswa(this.nim, this.nama);

  List<MataKuliah> get khs => List.unmodifiable(_khs);

  void tambahSemua(List<MataKuliah> daftar) => _khs.addAll(daftar);

  int get totalSks => _khs.fold(0, (total, mk) => total + mk.sks);

  double get ipk {
    if (totalSks == 0) return 0;
    final totalMutu = _khs.fold(0.0, (total, mk) => total + mk.mutu);
    return totalMutu / totalSks;
  }

  String get predikat => switch (ipk) {
        >= 3.51 => 'Dengan Pujian',
        >= 3.01 => 'Sangat Memuaskan',
        >= 2.76 => 'Memuaskan',
        _ => 'Cukup',
      };

  List<MataKuliah> get perluMengulang =>
      _khs.where((mk) => !mk.huruf.lulus).toList();

  ({MataKuliah terbaik, MataKuliah terendah}) get ringkasan {
    final urut = [..._khs]..sort((a, b) => b.nilaiAngka.compareTo(a.nilaiAngka));
    return (terbaik: urut.first, terendah: urut.last);
  }
}

// ===== 7. ASYNC: simulasi mengambil data dari server =====
Future<HasilMuat> muatKhsDariServer(String nim) async {
  await Future.delayed(const Duration(milliseconds: 500));
  if (nim.isEmpty) return MuatGagal('NIM kosong');

  final dataJson = <Map<String, dynamic>>[
    {'kode': 'IF201', 'nama': 'Pemrograman Mobile', 'sks': 3, 'nilai': 88},
    {'kode': 'IF202', 'nama': 'Basis Data', 'sks': 3, 'nilai': 76.5},
    {'kode': 'IF203', 'nama': 'Jaringan Komputer', 'sks': 2, 'nilai': 64},
    {'kode': 'IF204', 'nama': 'Kecerdasan Buatan', 'sks': 3, 'nilai': 81},
    {'kode': 'IF205', 'nama': 'Etika Profesi', 'sks': 2, 'nilai': 55},
    {'kode': 'IF206', 'nama': 'Statistika', 'sks': 3, 'nilai': 'A'}, // rusak
    {'kode': 'IF207', 'nama': 'Kerja Praktik', 'sks': 8, 'nilai': 90}, // rusak
  ];

  final data = <MataKuliah>[];
  final peringatan = <String>[];
  for (final json in dataJson) {
    try {
      data.add(MataKuliah.fromJson(json));
    } on DataNilaiException catch (e) {
      peringatan.add(e.pesan);
    }
  }
  return MuatBerhasil(data, peringatan);
}

// ===== 8. MAIN =====
Future<void> main() async {
  final mhs = Mahasiswa('H1D023001', 'Ayu Lestari');
  print('Memuat KHS ${mhs.nama}...');

  final hasil = await muatKhsDariServer(mhs.nim);
  switch (hasil) {
    case MuatGagal(:final pesan):
      print('Gagal memuat: $pesan');
      return;
    case MuatBerhasil(:final data, :final peringatan):
      mhs.tambahSemua(data);
      for (final p in peringatan) {
        print('Peringatan: $p');
      }
  }

  print('');
  print('KHS ${mhs.nim} - ${mhs.nama}');
  for (final (i, mk) in mhs.khs.indexed) {
    print('${i + 1}. ${mk.kode} ${mk.nama.padRight(20)}'
        '${mk.sks} SKS  ${mk.nilaiAngka.toString().padLeft(4)}  ${mk.huruf.label}');
  }

  final (:terbaik, :terendah) = mhs.ringkasan;
  print('');
  print('Total SKS   : ${mhs.totalSks}');
  print('IPK         : ${mhs.ipk.duaDesimal}');
  print('Predikat    : ${mhs.predikat}');
  print('Terbaik     : ${terbaik.nama} (${terbaik.huruf.label})');
  print('Terendah    : ${terendah.nama} (${terendah.huruf.label})');
  print('Mengulang   : ${mhs.perluMengulang.map((mk) => mk.nama).join(', ')}');
}

Beberapa hal yang perlu diperhatikan:

  • Data rusak tidak membuat program berhenti. IF206 ditolak oleh pattern di fromJson karena nilainya String, sedangkan IF207 lolos pattern tetapi ditolak validasi SKS di constructor. Keduanya melempar DataNilaiException yang ditangkap di dalam perulangan, sehingga data lain tetap diproses.
  • Perhitungan IPK: total mutu = 3×4.0 + 3×3.0 + 2×2.0 + 3×3.5 + 2×1.0 = 37.5. Dibagi 13 SKS menghasilkan 2.8846..., yang dibulatkan menjadi 2.88 oleh extension duaDesimal.
  • fold(0.0, ...) sengaja diberi nilai awal 0.0 (double), karena mutu bertipe double. Jika ditulis 0, Dart menganggap akumulatornya int dan terjadi error tipe.
  • Getter khs mengembalikan list yang tidak bisa diubah. Kode di luar class tidak bisa menambah mata kuliah tanpa lewat method tambahSemua.

Bonus: progres sinkronisasi dengan Stream

Fitur tambahan yang sering muncul di aplikasi: menampilkan persentase progres saat data dikirim ke server. Ini cocok dibuat dengan async*.

Stream<int> sinkronkanKrs(int jumlahMatkul) async* {
  for (var i = 1; i <= jumlahMatkul; i++) {
    await Future.delayed(const Duration(milliseconds: 200)); // kirim satu matkul
    yield i * 100 ~/ jumlahMatkul; // persentase selesai
  }
}

Future<void> main() async {
  await for (final persen in sinkronkanKrs(4)) {
    print('Menyinkronkan... $persen%');
  }
  print('Data KRS Ayu Lestari berhasil disinkronkan');
  // Output:
  // Menyinkronkan... 25%
  // Menyinkronkan... 50%
  // Menyinkronkan... 75%
  // Menyinkronkan... 100%
  // Data KRS Ayu Lestari berhasil disinkronkan
}

Di Flutter, stream seperti ini bisa dihubungkan ke StreamBuilder untuk menggerakkan LinearProgressIndicator.

💡 Latihan pengembangan (mandiri)

  1. Tambahkan field semester pada MataKuliah, lalu hitung IPS (indeks prestasi per semester) memakai where.
  2. Jika satu mata kuliah diambil dua kali, gunakan nilai terbaik saja saat menghitung IPK. Petunjuk: kelompokkan berdasarkan kode dengan Map.
  3. Buat muatKhsDariServer mengembalikan MuatGagal secara acak (gunakan Random dari dart:math), lalu pastikan pesan gagal tampil dengan benar.
  4. Tambahkan method toJson() pada MataKuliah dan cetak seluruh KHS dalam format JSON memakai jsonEncode dari dart:convert.
