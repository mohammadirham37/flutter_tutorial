Sekarang kita gabungkan semua materi modul ini menjadi satu aplikasi utuh: formulir pendaftaran mahasiswa baru yang memakai TextFormField, validasi, password, DatePicker, Radio, Dropdown, Slider, FilterChip, Switch, Checkbox, dialog konfirmasi, dan halaman hasil.

Rancangan aplikasi

  1. Model data DataPendaftar: satu objek yang menampung semua isian. Lebih rapi daripada mengirim sepuluh variabel terpisah.
  2. Validators: aturan NIM, email, nomor HP, password, dan konfirmasi.
  3. PendaftaranScreen: form tiga bagian (Data Diri, Kontak & Akun, Data Akademik).
  4. HasilPendaftaranScreen: menampilkan ringkasan data. Perpindahan halaman memakai Navigator.push, yang akan dibahas tuntas di Modul 07.

Alur saat tombol Kirim ditekan

  1. Tutup keyboard dengan unfocus().
  2. Jalankan validate() untuk semua TextFormField dan Dropdown.
  3. Periksa manual input yang bukan FormField: jenis kelamin, minat, persetujuan.
  4. Kumpulkan isian ke objek DataPendaftar.
  5. Tampilkan dialog konfirmasi. Jika disetujui, buka halaman hasil.

class DataPendaftar {
  final String nama;

  final String email;
  final String noHp;
  final String prodi;
  final JenisKelamin jenisKelamin;
  final DateTime tanggalLahir;

  final int semester;
  final bool ikutBeasiswa;

  const DataPendaftar({
    required this.nama,
    required this.nim,
    required this.email,
    required this.noHp,

    required this.jenisKelamin,

    required this.minat,
    required this.semester,
    required this.ikutBeasiswa,

// ======================= VALIDATOR =======================
class Validators {
  static final _emailRegex = RegExp(r'^[\w.+-]+@[\w-]+(\.[\w-]+)+$');

  static String? wajib(String? v, String field) =>
      (v == null || v.trim().isEmpty) ? '$field wajib diisi' : null;

  static String? nim(String? v) {
    if (v == null || v.trim().isEmpty) return 'NIM wajib diisi';
    if (!RegExp(r'^[A-Z]\d[A-Z]\d{6}$').hasMatch(v.trim().toUpperCase())) {
      return 'Format NIM contoh: H1D024001';
    }
    return null;
  }

  static String? email(String? v) {
    if (v == null || v.trim().isEmpty) return 'Email wajib diisi';
    if (!_emailRegex.hasMatch(v.trim())) return 'Format email tidak valid';
    return null;
  }

  static String? noHp(String? v) {
    if (v == null || v.trim().isEmpty) return 'Nomor HP wajib diisi';
    if (!RegExp(r'^08\d{8,11}$').hasMatch(v.trim())) {
      return 'Nomor HP diawali 08, 10-13 digit';
    }
    return null;
  }

  static String? password(String? v) {
    if (v == null || v.isEmpty) return 'Password wajib diisi';
    if (v.length < 8) return 'Password minimal 8 karakter';
    return null;
  }

  static String? konfirmasi(String? v, String asli) {
    if (v == null || v.isEmpty) return 'Ulangi password';
    if (v != asli) return 'Konfirmasi password tidak sama';
    return null;
  }
}

String formatTanggal(DateTime t) {
  const bulan = [
    'Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni',
    'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember',
  ];
  return '${t.day} ${bulan[t.month - 1]} ${t.year}';
}

// ======================= LAYAR FORM =======================
class PendaftaranScreen extends StatefulWidget {
  const PendaftaranScreen({super.key});

  @override
  State<PendaftaranScreen> createState() => _PendaftaranScreenState();
}

class _PendaftaranScreenState extends State<PendaftaranScreen> {
  final _formKey = GlobalKey<FormState>();

  final _namaC = TextEditingController();
  final _nimC = TextEditingController();
  final _emailC = TextEditingController();
  final _hpC = TextEditingController();
  final _passwordC = TextEditingController();
  final _konfirmasiC = TextEditingController();
  final _tanggalC = TextEditingController();

  static const _daftarProdi = [
    'S1 Informatika',
    'S1 Teknik Elektro',
    'S1 Teknik Sipil',
    'S1 Teknik Industri',
  ];
  static const _daftarMinat = ['Mobile', 'Web', 'AI', 'Jaringan', 'UI/UX'];

  String? _prodi;
  JenisKelamin? _jenisKelamin;
  DateTime? _tanggalLahir;
  final Set<String> _minat = {};
  double _semester = 1;
  bool _ikutBeasiswa = false;
  bool _setuju = false;
  bool _sembunyikan = true;

  @override
  void dispose() {
    for (final c in [_namaC, _nimC, _emailC, _hpC, _passwordC, _konfirmasiC, _tanggalC]) {
      c.dispose();
    }
    super.dispose();
  }

  Future<void> _pilihTanggalLahir() async {
    final hasil = await showDatePicker(
      context: context,
      initialDate: _tanggalLahir ?? DateTime(2005, 1, 1),
      firstDate: DateTime(1990),
      lastDate: DateTime.now(),
      helpText: 'Tanggal lahir',
    );
    if (hasil == null) return;
    setState(() {
      _tanggalLahir = hasil;
      _tanggalC.text = formatTanggal(hasil);
    });
  }

  void _tampilkanError(String pesan) {
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(SnackBar(content: Text(pesan), backgroundColor: Colors.red));
  }

  Future<void> _kirim() async {
    FocusScope.of(context).unfocus();

    // 1. Validasi semua TextFormField & DropdownButtonFormField
    final formValid = _formKey.currentState!.validate();
    if (!formValid) {
      _tampilkanError('Masih ada isian yang salah');
      return;
    }
    // 2. Validasi input yang tidak memakai FormField
    if (_jenisKelamin == null) {
      _tampilkanError('Pilih jenis kelamin');
      return;
    }
    if (_minat.isEmpty) {
      _tampilkanError('Pilih minimal satu bidang minat');
      return;
    }
    if (!_setuju) {
      _tampilkanError('Centang pernyataan persetujuan');
      return;
    }

    // 3. Kumpulkan data ke satu objek
    final data = DataPendaftar(
      nama: _namaC.text.trim(),
      nim: _nimC.text.trim().toUpperCase(),
      email: _emailC.text.trim(),
      noHp: _hpC.text.trim(),
      prodi: _prodi!,
      jenisKelamin: _jenisKelamin!,
      tanggalLahir: _tanggalLahir!,
      minat: _minat.toList(),
      semester: _semester.round(),
      ikutBeasiswa: _ikutBeasiswa,
    );

    // 4. Konfirmasi lewat dialog
    final yakin = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Text('Kirim pendaftaran?'),
        content: Text('Pastikan data ${data.nama} (${data.nim}) sudah benar.'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext, false),
            child: const Text('Periksa lagi'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(dialogContext, true),
            child: const Text('Ya, kirim'),
          ),
        ],
      ),
    );
    if (yakin != true || !mounted) return;

    // 5. Tampilkan hasil di layar baru (detail navigasi dibahas di Modul 07)
    Navigator.push(
      context,
      MaterialPageRoute(builder: (_) => HasilPendaftaranScreen(data: data)),
    );
  }

  Widget _judulBagian(String teks) => Padding(
        padding: const EdgeInsets.only(top: 24, bottom: 8),
        child: Text(teks, style: Theme.of(context).textTheme.titleMedium),
      );

  @override
  Widget build(BuildContext context) {
    const jarak = SizedBox(height: 12);

    return Scaffold(
      appBar: AppBar(title: const Text('Pendaftaran Mahasiswa Baru')),
      body: Form(
        key: _formKey,
        autovalidateMode: AutovalidateMode.onUserInteraction,
        child: ListView(
          padding: const EdgeInsets.fromLTRB(16, 0, 16, 32),
          children: [
            _judulBagian('A. Data Diri'),
            TextFormField(
              controller: _namaC,
              textCapitalization: TextCapitalization.words,
              textInputAction: TextInputAction.next,
              decoration: const InputDecoration(
                labelText: 'Nama Lengkap',
                prefixIcon: Icon(Icons.person_outline),
                border: OutlineInputBorder(),
              ),
              validator: (v) => Validators.wajib(v, 'Nama'),
            ),
            jarak,
            TextFormField(
              controller: _nimC,
              textCapitalization: TextCapitalization.characters,
              textInputAction: TextInputAction.next,
              decoration: const InputDecoration(
                labelText: 'NIM',
                hintText: 'H1D024001',
                prefixIcon: Icon(Icons.badge_outlined),
                border: OutlineInputBorder(),
              ),
              validator: Validators.nim,
            ),
            jarak,
            TextFormField(
              controller: _tanggalC,
              readOnly: true,
              onTap: _pilihTanggalLahir,
              decoration: const InputDecoration(
                labelText: 'Tanggal Lahir',
                prefixIcon: Icon(Icons.cake_outlined),
                border: OutlineInputBorder(),
              ),
              validator: (v) => Validators.wajib(v, 'Tanggal lahir'),
            ),
            jarak,
            const Text('Jenis Kelamin'),
            RadioGroup<JenisKelamin>(
              groupValue: _jenisKelamin,
              onChanged: (v) => setState(() => _jenisKelamin = v),
              child: const Row(
                children: [
                  Expanded(
                    child: RadioListTile(
                      value: JenisKelamin.lakiLaki,
                      title: Text('Laki-laki'),
                      contentPadding: EdgeInsets.zero,
                    ),
                  ),
                  Expanded(
                    child: RadioListTile(
                      value: JenisKelamin.perempuan,
                      title: Text('Perempuan'),
                      contentPadding: EdgeInsets.zero,
                    ),
                  ),
                ],
              ),
            ),

            _judulBagian('B. Kontak & Akun'),
            TextFormField(
              controller: _emailC,
              keyboardType: TextInputType.emailAddress,
              textInputAction: TextInputAction.next,
              decoration: const InputDecoration(
                labelText: 'Email',
                prefixIcon: Icon(Icons.email_outlined),
                border: OutlineInputBorder(),
              ),
              validator: Validators.email,
            ),
            jarak,
            TextFormField(
              controller: _hpC,
              keyboardType: TextInputType.phone,
              textInputAction: TextInputAction.next,
              decoration: const InputDecoration(
                labelText: 'Nomor HP',
                hintText: '081234567890',
                prefixIcon: Icon(Icons.phone_outlined),
                border: OutlineInputBorder(),
              ),
              validator: Validators.noHp,
            ),
            jarak,
            TextFormField(
              controller: _passwordC,
              obscureText: _sembunyikan,
              textInputAction: TextInputAction.next,
              decoration: InputDecoration(
                labelText: 'Password',
                prefixIcon: const Icon(Icons.lock_outline),
                border: const OutlineInputBorder(),
                suffixIcon: IconButton(
                  icon: Icon(_sembunyikan ? Icons.visibility : Icons.visibility_off),
                  onPressed: () => setState(() => _sembunyikan = !_sembunyikan),
                ),
              ),
              validator: Validators.password,
            ),
            jarak,
            TextFormField(
              controller: _konfirmasiC,
              obscureText: _sembunyikan,
              textInputAction: TextInputAction.done,
              decoration: const InputDecoration(
                labelText: 'Konfirmasi Password',
                prefixIcon: Icon(Icons.lock_reset),
                border: OutlineInputBorder(),
              ),
              validator: (v) => Validators.konfirmasi(v, _passwordC.text),
            ),

            _judulBagian('C. Data Akademik'),
            DropdownButtonFormField<String>(
              initialValue: _prodi,
              isExpanded: true,
              decoration: const InputDecoration(
                labelText: 'Program Studi',
                prefixIcon: Icon(Icons.school_outlined),
                border: OutlineInputBorder(),
              ),
              items: _daftarProdi
                  .map((p) => DropdownMenuItem(value: p, child: Text(p)))
                  .toList(),
              onChanged: (v) => setState(() => _prodi = v),
              validator: (v) => v == null ? 'Pilih program studi' : null,
            ),
            jarak,
            Text('Semester saat ini: ${_semester.round()}'),
            Slider(
              value: _semester,
              min: 1,
              max: 14,
              divisions: 13,
              label: 'Semester ${_semester.round()}',
              onChanged: (v) => setState(() => _semester = v),
            ),
            const Text('Bidang minat (boleh lebih dari satu)'),
            const SizedBox(height: 8),
            Wrap(
              spacing: 8,
              children: [
                for (final m in _daftarMinat)
                  // FilterChip = versi "chip" dari checkbox
                  FilterChip(
                    label: Text(m),
                    selected: _minat.contains(m),
                    onSelected: (pilih) => setState(() {
                      pilih ? _minat.add(m) : _minat.remove(m);
                    }),
                  ),
              ],
            ),
            SwitchListTile(
              contentPadding: EdgeInsets.zero,
              title: const Text('Mengajukan beasiswa'),
              value: _ikutBeasiswa,
              onChanged: (v) => setState(() => _ikutBeasiswa = v),
            ),
            CheckboxListTile(
              contentPadding: EdgeInsets.zero,
              controlAffinity: ListTileControlAffinity.leading,
              title: const Text('Saya menyatakan data di atas benar.'),
              value: _setuju,
              onChanged: (v) => setState(() => _setuju = v ?? false),
            ),
            const SizedBox(height: 16),
            FilledButton.icon(
              onPressed: _kirim,
              icon: const Icon(Icons.send),
              label: const Text('Kirim Pendaftaran'),
              style: FilledButton.styleFrom(minimumSize: const Size.fromHeight(50)),
            ),
          ],
        ),
      ),
    );
  }
}

class HasilPendaftaranScreen extends StatelessWidget {
  final DataPendaftar data;

  const HasilPendaftaranScreen({super.key, required this.data});

  @override
  Widget build(BuildContext context) {
    final baris = <(IconData, String, String)>[
      (Icons.person, 'Nama', data.nama),
      (Icons.badge, 'NIM', data.nim),
      (Icons.wc, 'Jenis Kelamin',
          data.jenisKelamin == JenisKelamin.lakiLaki ? 'Laki-laki' : 'Perempuan'),
      (Icons.cake, 'Tanggal Lahir', formatTanggal(data.tanggalLahir)),
      (Icons.email, 'Email', data.email),
      (Icons.phone, 'Nomor HP', data.noHp),
      (Icons.school, 'Program Studi', data.prodi),
      (Icons.timeline, 'Semester', '${data.semester}'),
      (Icons.interests, 'Minat', data.minat.join(', ')),
      (Icons.card_giftcard, 'Beasiswa', data.ikutBeasiswa ? 'Ya' : 'Tidak'),
    ];

    return Scaffold(
      appBar: AppBar(title: const Text('Pendaftaran Berhasil')),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          const Icon(Icons.check_circle, color: Colors.green, size: 80),
          const SizedBox(height: 8),
          Text(
            'Selamat datang, ${data.nama}!',
            textAlign: TextAlign.center,
            style: Theme.of(context).textTheme.titleLarge,
          ),
          const SizedBox(height: 16),
          Card(
            child: Column(
              children: [
                for (final (ikon, label, nilai) in baris)
                  ListTile(
                    leading: Icon(ikon),
                    title: Text(label),
                    subtitle: Text(nilai),
                  ),
              ],
            ),
          ),
          const SizedBox(height: 16),
          OutlinedButton.icon(
            onPressed: () => Navigator.pop(context),
            icon: const Icon(Icons.edit),
            label: const Text('Kembali & ubah data'),
          ),
        ],
      ),
    );
  }
}
Bagian atas form pendaftaran
Kirim ditekan saat form kosong: pesan error dan SnackBar merah
Bagian Data Akademik: Dropdown, Slider, FilterChip, Switch, Checkbox
Dialog konfirmasi setelah semua isian valid
Halaman hasil pendaftaran

Penjelasan bagian penting:

  • Validasi NIM memakai pola ^[A-Z]\d[A-Z]\d{6}$: huruf, angka, huruf, lalu enam angka (contoh H1D024001). Isian diubah ke huruf besar dulu dengan toUpperCase().
  • Banyak controller di-dispose sekaligus dengan perulangan for (final c in [...]) c.dispose();.
  • FilterChip berfungsi seperti checkbox dalam bentuk chip. Pilihan disimpan di Set<String> agar tidak ada duplikat.
  • Radio dalam Row: dua RadioListTile dibungkus Expanded agar berbagi lebar, dan RadioGroup membungkus Row tersebut.
  • Record (IconData, String, String) di halaman hasil adalah fitur Dart 3 untuk mengelompokkan beberapa nilai tanpa membuat kelas. for (final (ikon, label, nilai) in baris) langsung memecahnya menjadi tiga variabel.
  • Data dikirim ke halaman hasil lewat konstruktor: HasilPendaftaranScreen(data: data).

💡 Uji form Anda seperti penguji

  • Tekan Kirim tanpa mengisi apa pun: semua pesan error harus muncul.
  • Isi NIM dengan huruf kecil (h1d024001): harus tetap diterima.
  • Isi email budi@, nomor HP 0812, password 5 huruf: masing-masing harus ditolak dengan pesan yang jelas.
  • Isi password dan konfirmasi berbeda satu huruf: harus ditolak.
  • Isi semua dengan benar tetapi tidak memilih minat: SnackBar harus muncul.
