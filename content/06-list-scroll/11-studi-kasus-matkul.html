Studi kasus ini menggabungkan materi Modul 05 dan 06 menjadi aplikasi Daftar Mata Kuliah dengan fitur: pencarian, hapus dengan geser + urungkan, tambah lewat form di bottom sheet (dengan validasi), ringkasan total SKS, dan empty state.

Rancangan

  • Model MataKuliah (kode, nama, sks, semester). kode dipakai sebagai id unik untuk ValueKey Dismissible.
  • Data asli _semua dan data tampil _tampil (hasil filter + urut semester) dipisah.
  • Form tambah dibuat sebagai StatefulWidget terpisah (FormMatkulSheet) agar controller-nya punya dispose sendiri. Form mengembalikan objek MataKuliah lewat Navigator.pop.

// ======================= MODEL =======================
class MataKuliah {
  final String kode;
  final String nama;
  final int sks;
  final int semester;

  const MataKuliah({
    required this.kode,
    required this.nama,
    required this.sks,
    required this.semester,
  });
}

// ======================= LAYAR UTAMA =======================
class DaftarMatkulScreen extends StatefulWidget {
  const DaftarMatkulScreen({super.key});

  @override
  State<DaftarMatkulScreen> createState() => _DaftarMatkulScreenState();
}

class _DaftarMatkulScreenState extends State<DaftarMatkulScreen> {
  final List<MataKuliah> _semua = [
    const MataKuliah(kode: 'IF101', nama: 'Algoritma dan Pemrograman', sks: 3, semester: 1),
    const MataKuliah(kode: 'IF102', nama: 'Matematika Diskrit', sks: 3, semester: 1),
    const MataKuliah(kode: 'IF201', nama: 'Struktur Data', sks: 3, semester: 2),
    const MataKuliah(kode: 'IF202', nama: 'Basis Data', sks: 3, semester: 3),
    const MataKuliah(kode: 'IF305', nama: 'Pemrograman Mobile', sks: 4, semester: 5),
    const MataKuliah(kode: 'IF306', nama: 'Jaringan Komputer', sks: 3, semester: 5),
    const MataKuliah(kode: 'IF401', nama: 'Kecerdasan Buatan', sks: 3, semester: 6),
  ];

  final _cariController = TextEditingController();
  String _kataKunci = '';

  @override
  void dispose() {
    _cariController.dispose();
    super.dispose();
  }

  List<MataKuliah> get _tampil {
    final k = _kataKunci.toLowerCase().trim();
    final hasil = _semua
        .where((m) => m.nama.toLowerCase().contains(k) || m.kode.toLowerCase().contains(k))
        .toList();
    hasil.sort((a, b) => a.semester.compareTo(b.semester)); // urut per semester
    return hasil;
  }

  int get _totalSks => _semua.fold(0, (total, m) => total + m.sks);

  // ---------- HAPUS + URUNGKAN ----------
  void _hapus(MataKuliah m) {
    // Simpan posisi di list ASLI (bukan list hasil filter)
    final posisi = _semua.indexOf(m);
    setState(() => _semua.removeAt(posisi));

    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(
          content: Text('${m.nama} dihapus'),
          action: SnackBarAction(
            label: 'URUNGKAN',
            onPressed: () => setState(() => _semua.insert(posisi, m)),
          ),
        ),
      );
  }

  // ---------- TAMBAH lewat bottom sheet ----------
  Future<void> _bukaFormTambah() async {
    final baru = await showModalBottomSheet<MataKuliah>(
      context: context,
      isScrollControlled: true, // agar sheet bisa naik saat keyboard muncul
      showDragHandle: true,
      builder: (_) => FormMatkulSheet(kodeTerpakai: _semua.map((m) => m.kode).toSet()),
    );
    if (baru == null || !mounted) return;

    setState(() => _semua.add(baru));
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text('${baru.nama} ditambahkan')),
    );
  }

  @override
  Widget build(BuildContext context) {
    final tampil = _tampil;

    return Scaffold(
      appBar: AppBar(
        title: const Text('Mata Kuliah Saya'),
        bottom: PreferredSize(
          preferredSize: const Size.fromHeight(72),
          child: Padding(
            padding: const EdgeInsets.fromLTRB(16, 0, 16, 12),
            child: SearchBar(
              controller: _cariController,
              hintText: 'Cari nama atau kode...',
              leading: const Icon(Icons.search),
              trailing: [
                if (_kataKunci.isNotEmpty)
                  IconButton(
                    icon: const Icon(Icons.clear),
                    onPressed: () {
                      _cariController.clear();
                      setState(() => _kataKunci = '');
                    },
                  ),
              ],
              onChanged: (v) => setState(() => _kataKunci = v),
            ),
          ),
        ),
      ),
      body: Column(
        children: [
          // Ringkasan
          ListTile(
            tileColor: Theme.of(context).colorScheme.surfaceContainerHighest,
            leading: const Icon(Icons.summarize_outlined),
            title: Text('${_semua.length} mata kuliah - $_totalSks SKS'),
            subtitle: Text(_kataKunci.isEmpty
                ? 'Geser ke kiri untuk menghapus'
                : '${tampil.length} hasil untuk "$_kataKunci"'),
          ),
          Expanded(child: _buildDaftar(tampil)),
        ],
      ),
      floatingActionButton: FloatingActionButton.extended(
        onPressed: _bukaFormTambah,
        icon: const Icon(Icons.add),
        label: const Text('Tambah'),
      ),
    );
  }

  Widget _buildDaftar(List<MataKuliah> tampil) {
    // Empty state: dua kemungkinan berbeda
    if (_semua.isEmpty) {
      return const _Kosong(ikon: Icons.inbox_outlined, pesan: 'Belum ada mata kuliah.\nTekan Tambah.');
    }
    if (tampil.isEmpty) {
      return const _Kosong(ikon: Icons.search_off, pesan: 'Tidak ada yang cocok.');
    }

    return ListView.builder(
      padding: const EdgeInsets.only(bottom: 88), // ruang untuk FAB
      itemCount: tampil.length,
      itemBuilder: (context, i) {
        final m = tampil[i];
        return Dismissible(
          key: ValueKey(m.kode), // kode unik
          direction: DismissDirection.endToStart,
          background: Container(
            color: Colors.red,
            alignment: Alignment.centerRight,
            padding: const EdgeInsets.only(right: 24),
            child: const Icon(Icons.delete, color: Colors.white),
          ),
          onDismissed: (_) => _hapus(m),
          child: ListTile(
            leading: CircleAvatar(child: Text('${m.sks}')),
            title: Text(m.nama),
            subtitle: Text('${m.kode} - Semester ${m.semester}'),
            trailing: const Text('SKS'),
          ),
        );
      },
    );
  }
}

class _Kosong extends StatelessWidget {
  final IconData ikon;
  final String pesan;

  const _Kosong({required this.ikon, required this.pesan});

  @override
  Widget build(BuildContext context) {
    return Center(
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Icon(ikon, size: 72, color: Colors.grey),
          const SizedBox(height: 12),
          Text(pesan, textAlign: TextAlign.center),
        ],
      ),
    );
  }
}

// ======================= FORM TAMBAH (di dalam bottom sheet) =======================
class FormMatkulSheet extends StatefulWidget {
  final Set<String> kodeTerpakai;

  const FormMatkulSheet({super.key, required this.kodeTerpakai});

  @override
  State<FormMatkulSheet> createState() => _FormMatkulSheetState();
}

class _FormMatkulSheetState extends State<FormMatkulSheet> {
  final _formKey = GlobalKey<FormState>();
  final _kodeC = TextEditingController();
  final _namaC = TextEditingController();
  int? _sks;
  double _semester = 1;

  @override
  void dispose() {
    _kodeC.dispose();
    _namaC.dispose();
    super.dispose();
  }

  void _simpan() {
    if (!_formKey.currentState!.validate()) return;
    Navigator.pop(
      context,
      MataKuliah(
        kode: _kodeC.text.trim().toUpperCase(),
        nama: _namaC.text.trim(),
        sks: _sks!,
        semester: _semester.round(),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    // viewInsets.bottom = tinggi keyboard; padding ini mendorong form ke atas
    final tinggiKeyboard = MediaQuery.viewInsetsOf(context).bottom;

    return Padding(
      padding: EdgeInsets.fromLTRB(24, 0, 24, 24 + tinggiKeyboard),
      child: Form(
        key: _formKey,
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Text('Tambah Mata Kuliah', style: Theme.of(context).textTheme.titleLarge),
            const SizedBox(height: 16),
            TextFormField(
              controller: _kodeC,
              autofocus: true,
              textCapitalization: TextCapitalization.characters,
              textInputAction: TextInputAction.next,
              decoration: const InputDecoration(
                labelText: 'Kode (contoh: IF307)',
                border: OutlineInputBorder(),
              ),
              validator: (v) {
                final kode = (v ?? '').trim().toUpperCase();
                if (kode.isEmpty) return 'Kode wajib diisi';
                if (!RegExp(r'^[A-Z]{2}\d{3}$').hasMatch(kode)) return 'Format: 2 huruf + 3 angka';
                if (widget.kodeTerpakai.contains(kode)) return 'Kode sudah ada';
                return null;
              },
            ),
            const SizedBox(height: 12),
            TextFormField(
              controller: _namaC,
              textCapitalization: TextCapitalization.words,
              decoration: const InputDecoration(
                labelText: 'Nama Mata Kuliah',
                border: OutlineInputBorder(),
              ),
              validator: (v) =>
                  (v == null || v.trim().length < 3) ? 'Nama minimal 3 karakter' : null,
            ),
            const SizedBox(height: 12),
            DropdownButtonFormField<int>(
              initialValue: _sks,
              decoration: const InputDecoration(labelText: 'SKS', border: OutlineInputBorder()),
              items: [for (final s in [1, 2, 3, 4, 6]) DropdownMenuItem(value: s, child: Text('$s SKS'))],
              onChanged: (v) => setState(() => _sks = v),
              validator: (v) => v == null ? 'Pilih jumlah SKS' : null,
            ),
            const SizedBox(height: 12),
            Text('Semester: ${_semester.round()}'),
            Slider(
              value: _semester,
              min: 1,
              max: 8,
              divisions: 7,
              label: '${_semester.round()}',
              onChanged: (v) => setState(() => _semester = v),
            ),
            FilledButton(onPressed: _simpan, child: const Text('Simpan')),
          ],
        ),
      ),
    );
  }
}

Penjelasan bagian penting:

  • Hapus dari hasil filter: _hapus(m) menerima objek, lalu mencari posisinya di _semua dengan indexOf. Dengan begitu penghapusan tetap benar walaupun daftar sedang difilter atau diurutkan.
  • SearchBar adalah kolom pencarian bergaya Material 3. Diletakkan di AppBar.bottom melalui PreferredSize.
  • Dua jenis empty state: "belum ada mata kuliah" (data asli kosong) berbeda dengan "tidak ada yang cocok" (filter tanpa hasil).
  • Bottom sheet berisi form: isScrollControlled: true membolehkan sheet lebih tinggi dari setengah layar, dan padding bawah sebesar MediaQuery.viewInsetsOf(context).bottom (tinggi keyboard) mendorong form ke atas keyboard.
  • Validasi kode unik: daftar kode yang sudah dipakai dikirim ke form sebagai Set<String>, lalu diperiksa di validator.
  • Total SKS dihitung dengan fold: mulai dari 0, tambahkan sks setiap mata kuliah.
  • padding: EdgeInsets.only(bottom: 88) pada ListView memberi ruang agar item terakhir tidak tertutup FAB.
