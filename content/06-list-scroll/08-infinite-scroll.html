Aplikasi seperti forum atau media sosial tidak memuat ribuan data sekaligus. Data diambil per halaman (misalnya 15 item), dan halaman berikutnya dimuat saat pengguna mendekati akhir daftar. Teknik ini disebut infinite scroll atau load more.

Konsep: ScrollController

  • ScrollController dihubungkan ke ListView lewat controller:. Sama seperti TextEditingController: dibuat di State, wajib di-dispose.
  • controller.position.pixels = jarak yang sudah digulir; maxScrollExtent = jarak maksimum.
  • addListener dipanggil setiap kali posisi gulir berubah.
  • animateTo(0, ...) / jumpTo(0) menggulir ke posisi tertentu, misalnya kembali ke atas.

Contoh kecil

final _scrollController = ScrollController();

@override
void initState() {
  super.initState();
  _scrollController.addListener(() {
    final posisi = _scrollController.position;
    // pixels = jarak yang sudah digulir, maxScrollExtent = jarak maksimum
    if (posisi.pixels >= posisi.maxScrollExtent - 200) {
      debugPrint('Hampir sampai bawah -> muat data berikutnya');
    }
  });
}

@override
void dispose() {
  _scrollController.dispose(); // sama seperti TextEditingController
  super.dispose();
}

@override
Widget build(BuildContext context) {
  return ListView.builder(
    controller: _scrollController, // hubungkan ke list
    itemCount: 50,
    itemBuilder: (context, i) => ListTile(title: Text('Item $i')),
  );
}

Contoh lengkap: forum dengan load more

📄 lib/main.dart

// lib/main.dart — salin SELURUH kode ini untuk menggantikan isi lib/main.dart
import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const ForumScreen(),
    );
  }
}

class ForumScreen extends StatefulWidget {
  const ForumScreen({super.key});

  @override
  State<ForumScreen> createState() => _ForumScreenState();
}

class _ForumScreenState extends State<ForumScreen> {
  static const _ukuranHalaman = 15;
  static const _totalDiServer = 70; // anggap server hanya punya 70 postingan

  final _scrollController = ScrollController();
  final List<String> _postingan = [];
  int _halaman = 0;
  bool _sedangMemuat = false;
  bool _masihAda = true;

  @override
  void initState() {
    super.initState();
    _muatHalamanBerikutnya();
    _scrollController.addListener(_cekPosisiScroll);
  }

  @override
  void dispose() {
    _scrollController.removeListener(_cekPosisiScroll);
    _scrollController.dispose();
    super.dispose();
  }

  void _cekPosisiScroll() {
    final posisi = _scrollController.position;
    if (posisi.pixels >= posisi.maxScrollExtent - 300) {
      _muatHalamanBerikutnya();
    }
  }

  // Simulasi API: GET /postingan?page=N
  Future<List<String>> _ambilDariServer(int halaman) async {
    await Future.delayed(const Duration(milliseconds: 1200));
    final mulai = halaman * _ukuranHalaman;
    final akhir = (mulai + _ukuranHalaman).clamp(0, _totalDiServer);
    return [for (var i = mulai; i < akhir; i++) 'Postingan forum #${i + 1}'];
  }

  Future<void> _muatHalamanBerikutnya() async {
    // Penjaga: jangan memuat dua kali bersamaan & berhenti jika data habis
    if (_sedangMemuat || !_masihAda) return;
    setState(() => _sedangMemuat = true);

    final baru = await _ambilDariServer(_halaman);
    if (!mounted) return;

    setState(() {
      _postingan.addAll(baru);
      _halaman++;
      _sedangMemuat = false;
      _masihAda = baru.length == _ukuranHalaman;
    });
  }

  Future<void> _refresh() async {
    setState(() {
      _postingan.clear();
      _halaman = 0;
      _masihAda = true;
    });
    await _muatHalamanBerikutnya();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Forum (${_postingan.length} dimuat)')),
      body: RefreshIndicator(
        onRefresh: _refresh,
        child: ListView.builder(
          controller: _scrollController,
          physics: const AlwaysScrollableScrollPhysics(),
          // +1 untuk baris paling bawah (loading / "sudah habis")
          itemCount: _postingan.length + 1,
          itemBuilder: (context, index) {
            if (index == _postingan.length) {
              return Padding(
                padding: const EdgeInsets.all(24),
                child: Center(
                  child: _masihAda
                      ? const CircularProgressIndicator()
                      : const Text('Semua postingan sudah dimuat'),
                ),
              );
            }
            return ListTile(
              leading: CircleAvatar(child: Text('${index + 1}')),
              title: Text(_postingan[index]),
              subtitle: Text('Halaman ${index ~/ _ukuranHalaman + 1}'),
            );
          },
        ),
      ),
      floatingActionButton: FloatingActionButton.small(
        tooltip: 'Ke atas',
        // animateTo: gulir ke posisi tertentu dengan animasi
        onPressed: () => _scrollController.animateTo(
          0,
          duration: const Duration(milliseconds: 500),
          curve: Curves.easeOut,
        ),
        child: const Icon(Icons.arrow_upward),
      ),
    );
  }
}

Penjelasan kode:

  • _cekPosisiScroll memuat halaman berikutnya saat posisi tinggal 300 piksel dari bawah, sehingga data baru sudah siap sebelum pengguna benar-benar sampai di ujung.
  • Dua penjaga di awal _muatHalamanBerikutnya sangat penting: _sedangMemuat mencegah permintaan ganda (listener terpanggil puluhan kali per detik), dan _masihAda menghentikan pemuatan jika server sudah tidak punya data.
  • itemCount: _postingan.length + 1: item terakhir dipakai sebagai baris status (loading atau "sudah habis").
  • Data dianggap habis jika halaman yang diterima kurang dari _ukuranHalaman.
  • FAB memakai animateTo(0) untuk kembali ke atas, dan _refresh mereset semuanya ke halaman pertama.

⚠️ Kesalahan umum

  • Tanpa penjaga _sedangMemuat: halaman yang sama dimuat berkali-kali sehingga data dobel.
  • Lupa dispose ScrollController dan removeListener.
  • Membaca controller.position sebelum ListView tampil. Controller belum terpasang; periksa dengan controller.hasClients jika perlu.
