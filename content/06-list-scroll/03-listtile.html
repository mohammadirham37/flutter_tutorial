ListTile adalah satu baris daftar siap pakai dengan slot leading (kiri), title, subtitle, dan trailing (kanan). Hampir semua aplikasi daftar (kontak, chat, pengaturan) bisa dibuat dengannya.

Contoh kecil

ListTile(
  leading: const CircleAvatar(child: Text('IF')),   // kiri
  title: const Text('Pemrograman Mobile'),           // judul
  subtitle: const Text('IF305 - 4 SKS'),              // keterangan
  trailing: const Icon(Icons.chevron_right),          // kanan
  onTap: () => debugPrint('Buka detail'),             // seluruh baris bisa ditekan
),
ListTile dengan avatar, judul, subjudul, dan ikon panah

Contoh lengkap: sepuluh ragam ListTile

📄 lib/main.dart

// lib/main.dart — salin SELURUH kode ini untuk menggantikan isi lib/main.dart
import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.teal),
      home: const KatalogListTileScreen(),
    );
  }
}

class KatalogListTileScreen extends StatefulWidget {
  const KatalogListTileScreen({super.key});

  @override
  State<KatalogListTileScreen> createState() => _KatalogListTileScreenState();
}

class _KatalogListTileScreenState extends State<KatalogListTileScreen> {
  int _terpilih = 1;
  bool _hadir = true;
  bool _wifi = false;

  @override
  Widget build(BuildContext context) {
    final warna = Theme.of(context).colorScheme;

    return Scaffold(
      appBar: AppBar(title: const Text('Ragam ListTile')),
      body: ListView(
        children: [
          // 1. Paling dasar
          const ListTile(title: Text('1. Hanya judul')),

          // 2. Ikon + subjudul
          const ListTile(
            leading: Icon(Icons.event),
            title: Text('2. Ujian Tengah Semester'),
            subtitle: Text('Senin, 14 Oktober - Ruang C.2'),
          ),

          // 3. Avatar + trailing teks + bisa ditekan
          ListTile(
            leading: CircleAvatar(
              backgroundColor: warna.primaryContainer,
              child: const Text('NP'),
            ),
            title: const Text('3. Nadia Putri'),
            subtitle: const Text('Pesan: Besok kumpul jam 8 ya'),
            trailing: const Text('09:41'),
            onTap: () {},
          ),

          // 4. Tiga baris
          const ListTile(
            isThreeLine: true,
            leading: Icon(Icons.article_outlined),
            title: Text('4. Pengumuman Beasiswa'),
            subtitle: Text(
              'Pendaftaran beasiswa prestasi dibuka hingga akhir bulan. '
              'Syarat IPK minimal 3,25 dan aktif berorganisasi.',
            ),
          ),

          // 5. Badge di trailing
          ListTile(
            leading: const Icon(Icons.assignment),
            title: const Text('5. Tugas belum dikumpul'),
            trailing: Badge(
              label: const Text('3'),
              child: Icon(Icons.notifications, color: warna.primary),
            ),
          ),

          // 6. Pilihan aktif (selected)
          for (var i = 0; i < 3; i++)
            ListTile(
              selected: _terpilih == i,
              selectedTileColor: warna.secondaryContainer,
              leading: const Icon(Icons.class_outlined),
              title: Text('6. Kelas ${String.fromCharCode(65 + i)}'),
              trailing: _terpilih == i ? const Icon(Icons.check) : null,
              onTap: () => setState(() => _terpilih = i),
            ),

          // 7. Versi siap pakai dengan checkbox / switch
          CheckboxListTile(
            secondary: const Icon(Icons.how_to_reg),
            title: const Text('7. Tandai hadir'),
            value: _hadir,
            onChanged: (v) => setState(() => _hadir = v ?? false),
          ),
          SwitchListTile(
            secondary: const Icon(Icons.wifi),
            title: const Text('8. Wi-Fi kampus'),
            value: _wifi,
            onChanged: (v) => setState(() => _wifi = v),
          ),

          // 9. Di dalam Card agar terlihat seperti kartu
          Card(
            margin: const EdgeInsets.all(12),
            child: ListTile(
              contentPadding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
              leading: const Icon(Icons.menu_book, size: 36),
              title: const Text('9. Basis Data'),
              subtitle: const Text('Dosen: Nadia Fitri, M.T.'),
              trailing: IconButton(
                icon: const Icon(Icons.more_vert),
                onPressed: () {},
              ),
            ),
          ),

          // 10. Rapat (dense)
          const ListTile(
            dense: true,
            leading: Icon(Icons.info_outline),
            title: Text('10. Versi dense: lebih rapat untuk daftar panjang'),
          ),
        ],
      ),
    );
  }
}
Sepuluh ragam ListTile dalam satu layar

Penjelasan kode:

  • Nomor 4: isThreeLine: true memberi ruang untuk subjudul dua baris.
  • Nomor 5: Badge menampilkan angka kecil di atas ikon, cocok untuk jumlah notifikasi.
  • Nomor 6: selected mengubah warna teks/ikon, dan selectedTileColor mengubah warna latar baris terpilih.
  • Nomor 7-8: CheckboxListTile dan SwitchListTile sudah dibahas di Modul 05. Parameter secondary menaruh ikon di sisi berlawanan dari kontrol.
  • Nomor 9: ListTile di dalam Card tampil seperti kartu. trailing bisa berisi tombol.
  • Nomor 10: dense: true membuat baris lebih rapat.

⚠️ Kesalahan umum

  • Memberi trailing berupa Row tanpa mainAxisSize: MainAxisSize.min. Row mengambil seluruh lebar dan terjadi error layout.
  • Memasang ListTile di dalam Row. ListTile butuh lebar yang jelas; bungkus dengan Expanded.
  • Membungkus ListTile dengan InkWell/GestureDetector. Tidak perlu, cukup isi onTap.
