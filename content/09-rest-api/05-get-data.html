GET adalah request yang paling sering dipakai: mengambil daftar data (list) dan mengambil satu data berdasarkan id (detail). Di section ini kita membaca data post dari JSONPlaceholder lalu menampilkannya di layar.

Langkah-langkah Satu Request GET

  1. Susun Uri.
  2. Panggil await http.get(uri) untuk mendapatkan http.Response.
  3. Cek response.statusCode.
  4. Decode response.body dengan jsonDecode.
  5. Ubah Map/List menjadi model dengan fromJson.

Contoh Kecil: GET Daftar

import 'dart:convert';

import 'package:http/http.dart' as http;

import 'models/post.dart';

Future<List<Post>> fetchPosts() async {
  final uri = Uri.parse('https://jsonplaceholder.typicode.com/posts'); // 1
  final response = await http.get(uri);                                  // 2

  if (response.statusCode != 200) {                                      // 3
    throw Exception('Gagal memuat post (HTTP ${response.statusCode})');
  }

  final List<dynamic> data = jsonDecode(response.body) as List<dynamic>; // 4
  return data
      .map((e) => Post.fromJson(e as Map<String, dynamic>))                // 5
      .toList();
}

Contoh Kecil: GET Detail dan GET dengan Query

import 'dart:convert';

import 'package:http/http.dart' as http;

import 'models/post.dart';

// GET /posts/{id} -> satu object
Future<Post> fetchPostById(int id) async {
  final response = await http.get(
    Uri.parse('https://jsonplaceholder.typicode.com/posts/$id'),
  );
  if (response.statusCode == 404) {
    throw Exception('Post dengan id $id tidak ditemukan');
  }
  if (response.statusCode != 200) {
    throw Exception('HTTP ${response.statusCode}');
  }
  return Post.fromJson(jsonDecode(response.body) as Map<String, dynamic>);
}

// GET /posts?userId=1 -> filter memakai query parameter
Future<List<Post>> fetchPostsByUser(int userId) async {
  final uri = Uri.https('jsonplaceholder.typicode.com', '/posts', {
    'userId': '$userId',
  });
  final response = await http.get(uri);
  if (response.statusCode != 200) {
    throw Exception('HTTP ${response.statusCode}');
  }
  final data = jsonDecode(response.body) as List<dynamic>;
  return data.map((e) => Post.fromJson(e as Map<String, dynamic>)).toList();
}

Perbedaannya hanya pada hasil decode: endpoint daftar mengembalikan array (List), endpoint detail mengembalikan object (Map).

Contoh Lengkap: Daftar Post dan Halaman Detail

Kita kumpulkan fungsi GET dalam satu file service, lalu memakainya di UI. Model Post diambil dari section sebelumnya (lib/models/post.dart).

📄 lib/services/post_service.dart

import 'dart:convert';

import 'package:http/http.dart' as http;

import '../models/post.dart';

class PostService {
  static const String _baseUrl = 'https://jsonplaceholder.typicode.com';

  Future<List<Post>> fetchPosts({int limit = 20}) async {
    final uri = Uri.parse('$_baseUrl/posts?_limit=$limit');
    final response = await http.get(uri);
    if (response.statusCode != 200) {
      throw Exception('Gagal memuat daftar post (HTTP ${response.statusCode})');
    }
    final data = jsonDecode(response.body) as List<dynamic>;
    return data.map((e) => Post.fromJson(e as Map<String, dynamic>)).toList();
  }

  Future<Post> fetchPost(int id) async {
    final response = await http.get(Uri.parse('$_baseUrl/posts/$id'));
    if (response.statusCode != 200) {
      throw Exception('Gagal memuat post $id (HTTP ${response.statusCode})');
    }
    return Post.fromJson(jsonDecode(response.body) as Map<String, dynamic>);
  }

  // Komentar milik satu post: GET /posts/{id}/comments
  Future<List<String>> fetchCommentBodies(int postId) async {
    final response = await http.get(Uri.parse('$_baseUrl/posts/$postId/comments'));
    if (response.statusCode != 200) {
      throw Exception('Gagal memuat komentar (HTTP ${response.statusCode})');
    }
    final data = jsonDecode(response.body) as List<dynamic>;
    return data.map((e) => (e as Map<String, dynamic>)['body'] as String).toList();
  }
}

📄 lib/main.dart

// Pastikan sudah: flutter pub add http
// serta membuat lib/models/post.dart dan lib/services/post_service.dart
import 'package:flutter/material.dart';

import 'models/post.dart';
import 'services/post_service.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const PostListScreen(),
    );
  }
}

class PostListScreen extends StatefulWidget {
  const PostListScreen({super.key});

  @override
  State<PostListScreen> createState() => _PostListScreenState();
}

class _PostListScreenState extends State<PostListScreen> {
  final _service = PostService();
  List<Post> _posts = [];
  bool _loading = true;
  String? _error;

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    setState(() {
      _loading = true;
      _error = null;
    });
    try {
      final hasil = await _service.fetchPosts();
      if (!mounted) return;
      setState(() => _posts = hasil);
    } catch (e) {
      if (!mounted) return;
      setState(() => _error = e.toString());
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    Widget body;
    if (_loading) {
      body = const Center(child: CircularProgressIndicator());
    } else if (_error != null) {
      body = Center(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Text(_error!, textAlign: TextAlign.center),
            const SizedBox(height: 12),
            FilledButton(onPressed: _load, child: const Text('Coba Lagi')),
          ],
        ),
      );
    } else {
      body = ListView.separated(
        itemCount: _posts.length,
        separatorBuilder: (context, index) => const Divider(height: 1),
        itemBuilder: (context, index) {
          final post = _posts[index];
          return ListTile(
            leading: CircleAvatar(child: Text('${post.id}')),
            title: Text(post.title, maxLines: 1, overflow: TextOverflow.ellipsis),
            subtitle: Text(post.body, maxLines: 2, overflow: TextOverflow.ellipsis),
            onTap: () => Navigator.push(
              context,
              MaterialPageRoute(builder: (_) => PostDetailScreen(postId: post.id!)),
            ),
          );
        },
      );
    }

    return Scaffold(
      appBar: AppBar(title: const Text('Daftar Post (GET)')),
      body: body,
    );
  }
}

class PostDetailScreen extends StatefulWidget {
  final int postId;
  const PostDetailScreen({super.key, required this.postId});

  @override
  State<PostDetailScreen> createState() => _PostDetailScreenState();
}

class _PostDetailScreenState extends State<PostDetailScreen> {
  final _service = PostService();
  Post? _post;
  List<String> _komentar = [];
  String? _error;

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    try {
      // Dua request dijalankan bersamaan dengan Future.wait
      final hasil = await Future.wait([
        _service.fetchPost(widget.postId),
        _service.fetchCommentBodies(widget.postId),
      ]);
      if (!mounted) return;
      setState(() {
        _post = hasil[0] as Post;
        _komentar = hasil[1] as List<String>;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() => _error = e.toString());
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Post #${widget.postId}')),
      body: _error != null
          ? Center(child: Text(_error!))
          : _post == null
              ? const Center(child: CircularProgressIndicator())
              : ListView(
                  padding: const EdgeInsets.all(16),
                  children: [
                    Text(_post!.title, style: Theme.of(context).textTheme.headlineSmall),
                    const SizedBox(height: 8),
                    Text(_post!.body),
                    const Divider(height: 32),
                    Text('${_komentar.length} Komentar',
                        style: Theme.of(context).textTheme.titleMedium),
                    for (final k in _komentar)
                      Card(child: Padding(padding: const EdgeInsets.all(12), child: Text(k))),
                  ],
                ),
    );
  }
}

Penjelasan:

  • Layar daftar menyimpan tiga state: _loading, _error, dan _posts. Ketiganya menentukan widget apa yang tampil. Di section FutureBuilder nanti, pola ini dibuat lebih ringkas.
  • if (!mounted) return; setelah await mencegah error setState() called after dispose() ketika pengguna sudah keluar dari layar sebelum data datang.
  • Halaman detail hanya menerima postId, lalu mengambil data sendiri. Future.wait menjalankan dua request sekaligus sehingga lebih cepat daripada berurutan.
  • ?_limit=20 adalah fitur JSONPlaceholder untuk membatasi jumlah data. Setiap API punya nama parameter sendiri (limit, per_page, size), cek dokumentasinya.

Contoh Lain: Memetakan Nama Field yang Berbeda

Kadang nama field di server berbeda dengan model kita. Service berikut mengambil /users lalu memetakannya ke MahasiswaModel dari section 3 (field name menjadi nama, dan NIM dibuat dari id sebagai data contoh).

📄 lib/services/mahasiswa_api_service.dart

// lib/services/mahasiswa_api_service.dart
import 'dart:convert';
import 'package:http/http.dart' as http;

import '../models/mahasiswa_model.dart';

class MahasiswaApiService {
  // Ganti dengan endpoint server API kampus Anda atau mock API
  static const String baseUrl = "https://jsonplaceholder.typicode.com";

  // 1. GET: Mengambil daftar mahasiswa
  Future<List<MahasiswaModel>> fetchMahasiswa() async {
    final response = await http.get(
      Uri.parse('$baseUrl/users'),
      headers: {'Accept': 'application/json'},
    ).timeout(const Duration(seconds: 10));

    if (response.statusCode == 200) {
      final List<dynamic> rawList = json.decode(response.body);
      return rawList.map((item) => MahasiswaModel.fromJson({
        'id': item['id'],
        'nama': item['name'],
        'email': item['email'],
        'nim': '2110${item['id']}',
        'avatar': 'https://i.pravatar.cc/150?u=${item['id']}'
      })).toList();
    } else {
      throw Exception("Gagal memuat data dari server (HTTP ${response.statusCode})");
    }
  }

  // 2. POST: Mengirim data mahasiswa baru ke server
  Future<bool> createMahasiswa(MahasiswaModel mhs) async {
    final response = await http.post(
      Uri.parse('$baseUrl/users'),
      headers: {'Content-Type': 'application/json; charset=UTF-8'},
      body: json.encode(mhs.toJson()),
    );

    return response.statusCode == 201; // 201 Created
  }
}

⚠️ Kesalahan Umum

  • Lupa await sehingga variabel berisi Future<Response>, bukan Response.
  • Decode endpoint detail sebagai List (atau sebaliknya). Lihat dulu bentuk JSON-nya di browser.
  • Memanggil fungsi GET di dalam build(). build() bisa dipanggil puluhan kali, sehingga request terkirim berulang. Panggil di initState().
  • Teks dengan huruf non-ASCII tampil aneh (mis. Ã©). Jika server tidak mengirim charset, gunakan utf8.decode(response.bodyBytes) sebagai pengganti response.body.
