Persiapan: flutter pub add http provider, lalu salin empat file "dari section" di atas. Kita hanya menulis file yang BARU.

ℹ️ Menyiasati JSONPlaceholder

Karena server palsu selalu memberi id: 101 untuk post baru dan tidak menyimpan perubahan, provider kita (1) memberi id unik lokal untuk post baru, dan (2) tidak mengirim PUT untuk post buatan sendiri (id > 100) karena server akan membalas 500. Di backend sungguhan, dua siasat ini tidak diperlukan.

1. Provider CRUD

📄 lib/providers/post_crud_provider.dart

import 'package:flutter/foundation.dart';

import '../models/post.dart';
import '../repositories/post_repository.dart';

enum LoadStatus { initial, loading, success, error }

class PostCrudProvider extends ChangeNotifier {
  final PostRepository _repo;
  PostCrudProvider(this._repo);

  List<Post> _posts = [];
  LoadStatus _status = LoadStatus.initial;
  String? _error;
  bool _saving = false; // true saat POST/PUT/DELETE berjalan

  List<Post> get posts => List.unmodifiable(_posts);
  LoadStatus get status => _status;
  String? get error => _error;
  bool get saving => _saving;

  Post? findById(int id) {
    for (final p in _posts) {
      if (p.id == id) return p;
    }
    return null;
  }

  // READ
  Future<void> load() async {
    _status = LoadStatus.loading;
    _error = null;
    notifyListeners();
    try {
      _posts = await _repo.getAll();
      _status = LoadStatus.success;
    } catch (e) {
      _error = '$e';
      _status = LoadStatus.error;
    }
    notifyListeners();
  }

  // CREATE
  Future<void> add(String title, String body) async {
    await _runSaving(() async {
      final created = await _repo.create(Post(userId: 1, title: title, body: body));
      // JSONPlaceholder selalu memberi id 101, jadi buat id unik lokal
      final nextId = _posts.fold<int>(100, (max, p) => p.id! > max ? p.id! : max) + 1;
      _posts.insert(0, created.copyWith(id: nextId));
    });
  }

  // UPDATE
  Future<void> edit(Post post) async {
    await _runSaving(() async {
      // Post buatan sendiri (id > 100) tidak dikenal server palsu, cukup ubah lokal
      final saved = post.id! > 100 ? post : await _repo.update(post);
      final i = _posts.indexWhere((p) => p.id == post.id);
      if (i != -1) _posts[i] = saved;
    });
  }

  // DELETE
  Future<void> remove(int id) async {
    await _runSaving(() async {
      if (id <= 100) await _repo.delete(id);
      _posts.removeWhere((p) => p.id == id);
    });
  }

  // Pembantu: set flag saving, jalankan aksi, lempar ulang error ke UI
  Future<void> _runSaving(Future<void> Function() action) async {
    _saving = true;
    notifyListeners();
    try {
      await action();
    } finally {
      _saving = false;
      notifyListeners();
    }
  }
}

Penjelasan: method add, edit, dan remove tidak menangkap error. Error dibiarkan naik ke layar form/detail agar layar itu bisa menampilkan SnackBar dan tidak menutup form jika gagal. try/finally di _runSaving memastikan flag saving kembali false walaupun terjadi error.

2. Layar Daftar

📄 lib/screens/post_list_screen.dart

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../providers/post_crud_provider.dart';
import 'post_detail_screen.dart';
import 'post_form_screen.dart';

class PostListScreen extends StatelessWidget {
  const PostListScreen({super.key});

  @override
  Widget build(BuildContext context) {
    final prov = context.watch<PostCrudProvider>();

    Widget body;
    switch (prov.status) {
      case LoadStatus.initial:
      case LoadStatus.loading:
        body = const Center(child: CircularProgressIndicator());
      case LoadStatus.error:
        body = Center(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Icon(Icons.cloud_off, size: 56),
              const SizedBox(height: 8),
              Text(prov.error ?? 'Terjadi kesalahan'),
              TextButton(onPressed: prov.load, child: const Text('Coba Lagi')),
            ],
          ),
        );
      case LoadStatus.success:
        body = prov.posts.isEmpty
            ? const Center(child: Text('Belum ada post.'))
            : RefreshIndicator(
                onRefresh: prov.load,
                child: ListView.separated(
                  physics: const AlwaysScrollableScrollPhysics(),
                  itemCount: prov.posts.length,
                  separatorBuilder: (_, _) => const Divider(height: 1),
                  itemBuilder: (context, i) {
                    final post = prov.posts[i];
                    return ListTile(
                      leading: CircleAvatar(child: Text('${post.id}')),
                      title: Text(post.title, maxLines: 1, overflow: TextOverflow.ellipsis),
                      subtitle: Text(post.body, maxLines: 2, overflow: TextOverflow.ellipsis),
                      onTap: () => Navigator.push(
                        context,
                        MaterialPageRoute(builder: (_) => PostDetailScreen(postId: post.id!)),
                      ),
                    );
                  },
                ),
              );
    }

    return Scaffold(
      appBar: AppBar(title: const Text('Aplikasi Posts')),
      body: body,
      floatingActionButton: FloatingActionButton.extended(
        onPressed: () => Navigator.push(
          context,
          MaterialPageRoute(builder: (_) => const PostFormScreen()),
        ),
        icon: const Icon(Icons.add),
        label: const Text('Post Baru'),
      ),
    );
  }
}

3. Layar Detail (dengan Edit & Hapus)

📄 lib/screens/post_detail_screen.dart

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../providers/post_crud_provider.dart';
import 'post_form_screen.dart';

class PostDetailScreen extends StatelessWidget {
  final int postId;
  const PostDetailScreen({super.key, required this.postId});

  Future<void> _konfirmasiHapus(BuildContext context) async {
    final yakin = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('Hapus post?'),
        content: const Text('Post yang dihapus tidak bisa dikembalikan.'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('Batal')),
          FilledButton(onPressed: () => Navigator.pop(ctx, true), child: const Text('Hapus')),
        ],
      ),
    );
    if (yakin != true || !context.mounted) return;

    final messenger = ScaffoldMessenger.of(context);
    final navigator = Navigator.of(context);
    try {
      await context.read<PostCrudProvider>().remove(postId);
      navigator.pop(); // kembali ke daftar
      messenger.showSnackBar(const SnackBar(content: Text('Post dihapus')));
    } catch (e) {
      messenger.showSnackBar(SnackBar(content: Text('Gagal menghapus: $e')));
    }
  }

  @override
  Widget build(BuildContext context) {
    final prov = context.watch<PostCrudProvider>();
    final post = prov.findById(postId);

    // Post bisa saja sudah terhapus ketika layar ini masih terbuka
    if (post == null) {
      return Scaffold(appBar: AppBar(), body: const Center(child: Text('Post tidak ditemukan')));
    }

    return Scaffold(
      appBar: AppBar(
        title: Text('Post #${post.id}'),
        actions: [
          IconButton(
            tooltip: 'Edit',
            icon: const Icon(Icons.edit),
            onPressed: () => Navigator.push(
              context,
              MaterialPageRoute(builder: (_) => PostFormScreen(post: post)),
            ),
          ),
          IconButton(
            tooltip: 'Hapus',
            icon: const Icon(Icons.delete_outline),
            onPressed: prov.saving ? null : () => _konfirmasiHapus(context),
          ),
        ],
      ),
      body: ListView(
        padding: const EdgeInsets.all(20),
        children: [
          Text(post.title, style: Theme.of(context).textTheme.headlineSmall),
          const SizedBox(height: 4),
          Text('Penulis: user ${post.userId}', style: Theme.of(context).textTheme.labelMedium),
          const Divider(height: 32),
          Text(post.body, style: Theme.of(context).textTheme.bodyLarge),
        ],
      ),
    );
  }
}

4. Satu Form untuk Tambah dan Edit

Jika parameter post bernilai null, form berada dalam mode tambah; jika diisi, mode edit dan field terisi data lama.

📄 lib/screens/post_form_screen.dart

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../models/post.dart';
import '../providers/post_crud_provider.dart';

class PostFormScreen extends StatefulWidget {
  final Post? post; // null = tambah, ada isi = edit
  const PostFormScreen({super.key, this.post});

  @override
  State<PostFormScreen> createState() => _PostFormScreenState();
}

class _PostFormScreenState extends State<PostFormScreen> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _titleCtrl;
  late final TextEditingController _bodyCtrl;

  bool get _isEdit => widget.post != null;

  @override
  void initState() {
    super.initState();
    _titleCtrl = TextEditingController(text: widget.post?.title ?? '');
    _bodyCtrl = TextEditingController(text: widget.post?.body ?? '');
  }

  @override
  void dispose() {
    _titleCtrl.dispose();
    _bodyCtrl.dispose();
    super.dispose();
  }

  Future<void> _simpan() async {
    if (!_formKey.currentState!.validate()) return;
    final prov = context.read<PostCrudProvider>();
    final title = _titleCtrl.text.trim();
    final body = _bodyCtrl.text.trim();

    try {
      if (_isEdit) {
        await prov.edit(widget.post!.copyWith(title: title, body: body));
      } else {
        await prov.add(title, body);
      }
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(_isEdit ? 'Perubahan disimpan' : 'Post ditambahkan')),
      );
      Navigator.pop(context);
    } catch (e) {
      if (!mounted) return;
      // Form TIDAK ditutup agar isian pengguna tidak hilang
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Gagal: $e')));
    }
  }

  @override
  Widget build(BuildContext context) {
    final saving = context.watch<PostCrudProvider>().saving;
    return Scaffold(
      appBar: AppBar(title: Text(_isEdit ? 'Edit Post' : 'Post Baru')),
      body: Form(
        key: _formKey,
        child: ListView(
          padding: const EdgeInsets.all(16),
          children: [
            TextFormField(
              controller: _titleCtrl,
              decoration: const InputDecoration(labelText: 'Judul', border: OutlineInputBorder()),
              validator: (v) => (v == null || v.trim().length < 3) ? 'Judul minimal 3 karakter' : null,
            ),
            const SizedBox(height: 12),
            TextFormField(
              controller: _bodyCtrl,
              maxLines: 5,
              decoration: const InputDecoration(labelText: 'Isi', border: OutlineInputBorder()),
              validator: (v) => (v == null || v.trim().isEmpty) ? 'Isi tidak boleh kosong' : null,
            ),
            const SizedBox(height: 20),
            FilledButton.icon(
              onPressed: saving ? null : _simpan,
              icon: saving
                  ? const SizedBox(width: 18, height: 18, child: CircularProgressIndicator(strokeWidth: 2))
                  : const Icon(Icons.save),
              label: Text(saving ? 'Menyimpan...' : 'Simpan'),
            ),
          ],
        ),
      ),
    );
  }
}

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import 'core/api_client.dart';
import 'providers/post_crud_provider.dart';
import 'repositories/post_repository.dart';
import 'screens/post_list_screen.dart';

void main() {
  final api = ApiClient(baseUrl: 'https://jsonplaceholder.typicode.com');
  runApp(
    ChangeNotifierProvider(
      create: (_) => PostCrudProvider(RemotePostRepository(api))..load(),
      child: const PostsApp(),
    ),
  );
}

class PostsApp extends StatelessWidget {
  const PostsApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Aplikasi Posts',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.teal),
      home: const PostListScreen(),
    );
  }
}

Alur Data Saat Menambah Post

PostFormScreen._simpan()
   └─> PostCrudProvider.add()          saving = true, notifyListeners (tombol jadi "Menyimpan...")
         └─> RemotePostRepository.create()
               └─> ApiClient.post('/posts', json)   -> 201 Created
         <─ Post baru dimasukkan ke _posts, saving = false, notifyListeners
   <─ SnackBar "Post ditambahkan", Navigator.pop
PostListScreen (context.watch) otomatis rebuild dan post baru muncul paling atas

⚠️ Kesalahan Umum pada Proyek CRUD

  • Menutup form (Navigator.pop) sebelum await selesai, sehingga error tidak pernah terlihat.
  • Mengirim objek yang sama ke halaman detail lalu mengeditnya: detail tidak berubah. Kirim id lalu cari data terbaru dari provider (seperti findById).
  • Menekan tombol Simpan berkali-kali sehingga data dobel. Nonaktifkan tombol saat saving.
  • Memakai context setelah await tanpa cek mounted/context.mounted.
