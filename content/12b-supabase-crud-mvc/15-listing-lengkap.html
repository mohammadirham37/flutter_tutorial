1. SQL Supabase

📄 supabase/setup.sql

-- 1. Tabel mahasiswa: setiap baris dimiliki satu user (user_id)
create table if not exists public.mahasiswa (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null default auth.uid()
    references auth.users (id) on delete cascade,
  nim text not null,
  nama text not null,
  prodi text not null,
  semester int not null check (semester between 1 and 14),
  created_at timestamptz not null default now(),
  unique (user_id, nim)
);

-- 2. Izinkan user yang login memakai tabel ini lewat API
grant select, insert, update, delete on public.mahasiswa to authenticated;

-- 3. Kunci tabel dengan Row Level Security
alter table public.mahasiswa enable row level security;

-- 4. Policy: user hanya boleh menyentuh baris miliknya sendiri
drop policy if exists "Lihat data sendiri" on public.mahasiswa;
create policy "Lihat data sendiri"
on public.mahasiswa for select
to authenticated
using ((select auth.uid()) = user_id);

drop policy if exists "Tambah data sendiri" on public.mahasiswa;
create policy "Tambah data sendiri"
on public.mahasiswa for insert
to authenticated
with check ((select auth.uid()) = user_id);

drop policy if exists "Ubah data sendiri" on public.mahasiswa;
create policy "Ubah data sendiri"
on public.mahasiswa for update
to authenticated
using ((select auth.uid()) = user_id)
with check ((select auth.uid()) = user_id);

drop policy if exists "Hapus data sendiri" on public.mahasiswa;
create policy "Hapus data sendiri"
on public.mahasiswa for delete
to authenticated
using ((select auth.uid()) = user_id);

-- 5. Index agar filter berdasarkan user_id tetap cepat
create index if not exists mahasiswa_user_id_idx on public.mahasiswa (user_id);

class Env {

  static const String supabaseKey = String.fromEnvironment(
    'SUPABASE_PUBLISHABLE_KEY',

4. Model Mahasiswa

/// MODEL: representasi data Mahasiswa.
/// Class ini murni berisi data (tidak tahu soal UI maupun GetX).
class Mahasiswa {
  /// null  -> data baru, belum tersimpan di Supabase.
  /// berisi -> uuid yang dibuat otomatis oleh Postgres.
  final String? id;
  final String nim;
  final String nama;
  final String prodi;
  final int semester;

  Mahasiswa({
    this.id,
    required this.nim,
    required this.nama,
    required this.prodi,
    required this.semester,
  });

  /// Map dari Supabase (satu baris tabel) -> objek Mahasiswa.
  factory Mahasiswa.fromJson(Map<String, dynamic> json) {
    return Mahasiswa(
      id: json['id'] as String,
      nim: json['nim'] as String,
      nama: json['nama'] as String,
      prodi: json['prodi'] as String,
      semester: json['semester'] as int,
    );
  }

  /// Objek Mahasiswa -> Map untuk insert / update.
  /// id dan user_id TIDAK dikirim: keduanya diisi otomatis oleh database.
  Map<String, dynamic> toJson() {
    return {'nim': nim, 'nama': nama, 'prodi': prodi, 'semester': semester};
  }

  /// Membuat salinan objek dengan sebagian field yang diubah.
  Mahasiswa copyWith({
    String? nim,
    String? nama,
    String? prodi,
    int? semester,
  }) {
    return Mahasiswa(
      id: id,
      nim: nim ?? this.nim,
      nama: nama ?? this.nama,
      prodi: prodi ?? this.prodi,
      semester: semester ?? this.semester,
    );
  }
}

/// Lapisan akses Supabase Auth (bagian dari MODEL pada MVC).

    required String email,

}

6. MahasiswaService

📄 lib/services/mahasiswa_service.dart

import 'package:supabase_flutter/supabase_flutter.dart';

import '../models/mahasiswa_model.dart';

/// Lapisan akses data (bagian dari MODEL pada MVC).
/// Pengganti HiveService: semua operasi CRUD ke tabel `mahasiswa`
/// di Supabase dikumpulkan di sini.
class MahasiswaService {
  MahasiswaService(this._client);

  final SupabaseClient _client;

  static const String table = 'mahasiswa';

  // READ: RLS otomatis membatasi hanya data milik user yang login.
  Future<List<Mahasiswa>> getAll() async {
    final rows = await _client
        .from(table)
        .select()
        .order('nama', ascending: true);
    return rows.map(Mahasiswa.fromJson).toList();
  }

  // CREATE: id dan user_id diisi otomatis oleh database.
  Future<void> create(Mahasiswa mhs) async {
    await _client.from(table).insert(mhs.toJson());
  }

  // UPDATE: ubah baris dengan id tertentu.
  Future<void> update(String id, Mahasiswa mhs) async {
    await _client.from(table).update(mhs.toJson()).eq('id', id);
  }

  // DELETE
  Future<void> delete(String id) async {
    await _client.from(table).delete().eq('id', id);
  }
}

7. AuthController

📄 lib/controllers/auth_controller.dart

import 'dart:async';

import 'package:flutter/material.dart';
import 'package:get/get.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../routes/app_routes.dart';
import '../services/auth_service.dart';

/// CONTROLLER untuk login, register, dan logout.
class AuthController extends GetxController {
  AuthController(this._authService);

  final AuthService _authService;

  // ------------------ STATE REAKTIF ------------------
  final isLoading = false.obs;
  final hidePassword = true.obs;

  // ------------------ STATE FORM ------------------
  // Login dan register punya form sendiri-sendiri, karena saat halaman
  // register dibuka, halaman login masih ada di bawahnya.
  final loginFormKey = GlobalKey<FormState>();
  final registerFormKey = GlobalKey<FormState>();
  final emailC = TextEditingController();
  final passwordC = TextEditingController();
  final regEmailC = TextEditingController();
  final regPasswordC = TextEditingController();
  final regKonfirmasiC = TextEditingController();

  StreamSubscription<AuthState>? _authSub;

  bool get isLoggedIn => _authService.currentUser != null;
  String get email => _authService.currentUser?.email ?? '';

  @override
  void onInit() {
    super.onInit();
    // Jika sesi berakhir (logout, token kedaluwarsa, akun dihapus),
    // kembalikan pengguna ke halaman login dari halaman mana pun.
    _authSub = _authService.authStateChanges.listen((data) {
      if (data.event == AuthChangeEvent.signedOut) {
        Get.offAllNamed(AppRoutes.login);
      }
    });
  }

  // ------------------ LOGIN ------------------
  Future<void> login() async {
    if (!loginFormKey.currentState!.validate()) return;

    isLoading.value = true;
    try {
      await _authService.signIn(
        email: emailC.text.trim(),
        password: passwordC.text,
      );
      passwordC.clear();
      Get.offAllNamed(AppRoutes.home);
    } on AuthException catch (e) {
      _tampilkanError(_pesanAuth(e));
    } catch (e) {
      _tampilkanError('Tidak dapat terhubung ke server. Periksa internet.');
    } finally {
      isLoading.value = false;
    }
  }

  // ------------------ REGISTER ------------------
  void openRegister() {
    regEmailC.text = emailC.text.trim();
    regPasswordC.clear();
    regKonfirmasiC.clear();
    Get.toNamed(AppRoutes.register);
  }

  Future<void> register() async {
    if (!registerFormKey.currentState!.validate()) return;

    isLoading.value = true;
    try {
      final res = await _authService.signUp(
        email: regEmailC.text.trim(),
        password: regPasswordC.text,
      );

      if (res.session != null) {
        // "Confirm email" dimatikan: user langsung login.
        Get.offAllNamed(AppRoutes.home);
        Get.snackbar(
          'Berhasil',
          'Akun dibuat, selamat datang!',
          snackPosition: SnackPosition.BOTTOM,
        );
      } else {
        // "Confirm email" aktif: user harus klik link di email dulu.
        emailC.text = regEmailC.text.trim();
        tutup(); // kembali ke halaman login
        Get.snackbar(
          'Cek email Anda',
          'Klik link verifikasi di email, lalu login.',
          snackPosition: SnackPosition.BOTTOM,
          duration: const Duration(seconds: 5),
        );
      }
    } on AuthException catch (e) {
      _tampilkanError(_pesanAuth(e));
    } catch (e) {
      _tampilkanError('Tidak dapat terhubung ke server. Periksa internet.');
    } finally {
      isLoading.value = false;
    }
  }

  // ------------------ LOGOUT ------------------
  Future<void> confirmLogout() async {
    final yakin = await Get.dialog<bool>(
      AlertDialog(
        title: const Text('Keluar'),
        content: const Text('Yakin ingin keluar dari akun ini?'),
        actions: [
          TextButton(onPressed: () => tutup(false), child: const Text('Batal')),
          FilledButton(
            onPressed: () => tutup(true),
            child: const Text('Keluar'),
          ),
        ],
      ),
    );
    if (yakin == true) {
      // Tidak perlu navigasi di sini: listener di onInit menerima event
      // signedOut lalu membuka halaman login.
      await _authService.signOut();
    }
  }

  // ------------------ HELPER ------------------
  /// Menutup halaman / dialog paling atas (sama seperti di Modul 11B).
  void tutup<T>([T? result]) => Get.key.currentState?.pop<T>(result);

  void _tampilkanError(String pesan) {
    Get.snackbar('Gagal', pesan, snackPosition: SnackPosition.BOTTOM);
  }

  /// Menerjemahkan kode error Supabase Auth ke bahasa Indonesia.
  String _pesanAuth(AuthException e) {
    if (e is AuthRetryableFetchException) {
      return 'Tidak dapat terhubung ke server. Periksa internet.';
    }
    switch (e.code) {
      case 'invalid_credentials':
        return 'Email atau password salah.';
      case 'email_not_confirmed':
        return 'Email belum diverifikasi. Buka link di inbox Anda.';
      case 'user_already_exists':
      case 'email_exists':
        return 'Email sudah terdaftar. Silakan login.';
      case 'weak_password':
        return 'Password terlalu lemah (minimal 6 karakter).';
      case 'email_address_invalid':
        return 'Alamat email ditolak server.';
      case 'over_email_send_rate_limit':
      case 'over_request_rate_limit':
        return 'Terlalu banyak percobaan. Tunggu beberapa menit.';
    }
    return e.message;
  }

  // ------------------ VALIDATOR ------------------
  String? validateEmail(String? v) {
    final text = v?.trim() ?? '';
    if (text.isEmpty) return 'Email wajib diisi';
    if (!RegExp(r'^[^@\s]+@[^@\s]+\.[^@\s]+$').hasMatch(text)) {
      return 'Format email tidak valid';
    }
    return null;
  }

  String? validatePassword(String? v) {
    if (v == null || v.isEmpty) return 'Password wajib diisi';
    if (v.length < 6) return 'Password minimal 6 karakter';
    return null;
  }

  String? validateKonfirmasi(String? v) {
    if (v == null || v.isEmpty) return 'Ulangi password wajib diisi';
    if (v != regPasswordC.text) return 'Konfirmasi password tidak sama';
    return null;
  }

  @override
  void onClose() {
    _authSub?.cancel();
    emailC.dispose();
    passwordC.dispose();
    regEmailC.dispose();
    regPasswordC.dispose();
    regKonfirmasiC.dispose();
    super.onClose();
  }
}

8. MahasiswaController

📄 lib/controllers/mahasiswa_controller.dart

import 'package:flutter/material.dart';
import 'package:get/get.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../models/mahasiswa_model.dart';
import '../routes/app_routes.dart';
import '../services/mahasiswa_service.dart';

/// CONTROLLER: penghubung antara View dan Model.
/// Isinya hampir sama dengan versi Hive. Perbedaannya: semua akses data
/// sekarang async (lewat internet), jadi ada state loading dan error.
class MahasiswaController extends GetxController {
  MahasiswaController(this._service);

  final MahasiswaService _service;

  // ------------------ STATE REAKTIF ------------------
  final mahasiswaList = <Mahasiswa>[].obs;
  final keyword = ''.obs;
  final selectedProdi = 'Informatika'.obs;
  final isLoading = false.obs; // sedang memuat daftar
  final isSaving = false.obs; // sedang menyimpan form
  final errorMessage = RxnString(); // null = tidak ada error

  final List<String> daftarProdi = const [
    'Informatika',
    'Sistem Informasi',
    'Teknik Komputer',
    'Data Science',
  ];

  // ------------------ STATE FORM ------------------
  final formKey = GlobalKey<FormState>();
  final nimC = TextEditingController();
  final namaC = TextEditingController();
  final semesterC = TextEditingController();

  /// null  -> mode TAMBAH
  /// berisi id -> mode EDIT
  String? editingId;
  bool get isEdit => editingId != null;

  /// Data yang ditampilkan (sudah difilter pencarian).
  List<Mahasiswa> get filteredList {
    final q = keyword.value.toLowerCase();
    if (q.isEmpty) return mahasiswaList;
    return mahasiswaList
        .where((m) => m.nama.toLowerCase().contains(q) || m.nim.contains(q))
        .toList();
  }

  @override
  void onInit() {
    super.onInit();
    loadData();
  }

  // ------------------ READ ------------------
  Future<void> loadData() async {
    isLoading.value = true;
    errorMessage.value = null;
    try {
      final data = await _service.getAll();
      mahasiswaList.assignAll(data);
    } catch (e) {
      errorMessage.value = _pesanError(e);
    } finally {
      isLoading.value = false;
    }
  }

  // ------------------ NAVIGASI KE FORM ------------------
  void openAddForm() {
    editingId = null;
    nimC.clear();
    namaC.clear();
    semesterC.clear();
    selectedProdi.value = daftarProdi.first;
    Get.toNamed(AppRoutes.form);
  }

  void openEditForm(Mahasiswa mhs) {
    editingId = mhs.id;
    nimC.text = mhs.nim;
    namaC.text = mhs.nama;
    semesterC.text = mhs.semester.toString();
    selectedProdi.value = mhs.prodi;
    Get.toNamed(AppRoutes.form);
  }

  // ------------------ CREATE & UPDATE ------------------
  Future<void> save() async {
    if (isSaving.value) return; // cegah tombol ditekan dua kali
    if (!formKey.currentState!.validate()) return;

    final nim = nimC.text.trim();

    // Cegah NIM ganda (cek cepat di memori sebelum ke server)
    final nimSudahAda = mahasiswaList.any(
      (m) => m.nim == nim && m.id != editingId,
    );
    if (nimSudahAda) {
      Get.snackbar(
        'Gagal',
        'NIM $nim sudah terdaftar',
        snackPosition: SnackPosition.BOTTOM,
      );
      return;
    }

    final mhs = Mahasiswa(
      nim: nim,
      nama: namaC.text.trim(),
      prodi: selectedProdi.value,
      semester: int.parse(semesterC.text.trim()),
    );

    isSaving.value = true;
    try {
      if (isEdit) {
        await _service.update(editingId!, mhs);
      } else {
        await _service.create(mhs);
      }

      final pesan = isEdit
          ? 'Data berhasil diperbarui'
          : 'Data berhasil ditambahkan';
      await loadData();
      tutup(); // kembali ke halaman daftar
      Get.snackbar('Sukses', pesan, snackPosition: SnackPosition.BOTTOM);
    } catch (e) {
      Get.snackbar(
        'Gagal',
        _pesanError(e),
        snackPosition: SnackPosition.BOTTOM,
      );
    } finally {
      isSaving.value = false;
    }
  }

  // ------------------ DELETE ------------------
  Future<void> confirmDelete(Mahasiswa mhs) async {
    final yakin = await Get.dialog<bool>(
      AlertDialog(
        title: const Text('Hapus Data'),
        content: Text('Yakin ingin menghapus ${mhs.nama}?'),
        actions: [
          TextButton(onPressed: () => tutup(false), child: const Text('Batal')),
          FilledButton(
            onPressed: () => tutup(true),
            child: const Text('Hapus'),
          ),
        ],
      ),
    );

    if (yakin != true) return;

    try {
      await _service.delete(mhs.id!);
      await loadData();
      Get.snackbar(
        'Terhapus',
        '${mhs.nama} telah dihapus',
        snackPosition: SnackPosition.BOTTOM,
      );
    } catch (e) {
      Get.snackbar(
        'Gagal',
        _pesanError(e),
        snackPosition: SnackPosition.BOTTOM,
      );
    }
  }

  // ------------------ HELPER ------------------
  /// Menutup halaman / dialog paling atas (opsional membawa [result]).
  void tutup<T>([T? result]) => Get.key.currentState?.pop<T>(result);

  /// Mengubah exception menjadi pesan yang mudah dipahami pengguna.
  String _pesanError(Object e) {
    if (e is PostgrestException) {
      if (e.code == '23505') return 'NIM sudah terdaftar';
      if (e.code == '42501') return 'Akses ditolak. Silakan login ulang.';
      return 'Kesalahan database: ${e.message}';
    }
    return 'Tidak dapat terhubung ke server. Periksa internet.';
  }

  // ------------------ VALIDATOR ------------------
  String? validateNim(String? v) {
    if (v == null || v.trim().isEmpty) return 'NIM wajib diisi';
    if (int.tryParse(v.trim()) == null) return 'NIM harus berupa angka';
    if (v.trim().length < 8) return 'NIM minimal 8 digit';
    return null;
  }

  String? validateNama(String? v) {
    if (v == null || v.trim().isEmpty) return 'Nama wajib diisi';
    return null;
  }

  String? validateSemester(String? v) {
    final s = int.tryParse(v?.trim() ?? '');
    if (s == null) return 'Semester wajib berupa angka';
    if (s < 1 || s > 14) return 'Semester antara 1 - 14';
    return null;
  }

  @override
  void onClose() {
    nimC.dispose();
    namaC.dispose();
    semesterC.dispose();
    super.onClose();
  }
}

  final _events = StreamController<AuthState>.broadcast();

  User _buatUser(String email) => User(

    appMetadata: const {},
    userMetadata: const {},
    aud: 'authenticated',
    email: email,
    createdAt: DateTime.now().toIso8601String(),
  );

  @override
  Stream<AuthState> get authStateChanges => _events.stream;

        statusCode: '400',
      );
    }
    _user = _buatUser(email);
    final session = Session(
      accessToken: 'token',
      tokenType: 'bearer',
      user: _user!,
    );
    _events.add(AuthState(AuthChangeEvent.signedIn, session));
    return AuthResponse(session: session);
  }

  @override

/// MahasiswaService palsu: data disimpan di List biasa (memori).
class FakeMahasiswaService implements MahasiswaService {
  final List<Mahasiswa> _data = [];
  int _nextId = 1;

  @override
  Future<List<Mahasiswa>> getAll() async {
    final hasil = [..._data]..sort((a, b) => a.nama.compareTo(b.nama));
    return hasil;
  }

  @override
  Future<void> create(Mahasiswa mhs) async {
    if (_data.any((m) => m.nim == mhs.nim)) {
      // tiru error unique constraint dari Postgres
      throw const PostgrestException(message: 'duplicate', code: '23505');
    }
    _data.add(
      Mahasiswa(
        id: '${_nextId++}',
        nim: mhs.nim,
        nama: mhs.nama,
        prodi: mhs.prodi,
        semester: mhs.semester,
      ),
    );
  }

  @override
  Future<void> update(String id, Mahasiswa mhs) async {
    final i = _data.indexWhere((m) => m.id == id);
    _data[i] = Mahasiswa(
      id: id,
      nim: mhs.nim,
      nama: mhs.nama,
      prodi: mhs.prodi,
      semester: mhs.semester,
    );
  }

  @override
  Future<void> delete(String id) async {
    _data.removeWhere((m) => m.id == id);
  }
}
