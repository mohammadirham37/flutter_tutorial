1. MahasiswaListView

Ganti isi lib/views/mahasiswa_list_view.dart:

📄 lib/views/mahasiswa_list_view.dart

import 'package:flutter/material.dart';
import 'package:get/get.dart';

import '../controllers/auth_controller.dart';
import '../controllers/mahasiswa_controller.dart';

/// VIEW: halaman daftar mahasiswa (READ + tombol Edit/Hapus + Logout).
class MahasiswaListView extends GetView<MahasiswaController> {
  const MahasiswaListView({super.key});

  @override
  Widget build(BuildContext context) {
    final auth = Get.find<AuthController>();

    return Scaffold(
      appBar: AppBar(
        title: const Text('Data Mahasiswa'),
        centerTitle: true,
        actions: [
          IconButton(
            tooltip: 'Keluar',
            icon: const Icon(Icons.logout),
            onPressed: auth.confirmLogout,
          ),
        ],
      ),
      body: Column(
        children: [
          // ---------- Info akun ----------
          Padding(
            padding: const EdgeInsets.fromLTRB(12, 12, 12, 0),
            child: Row(
              children: [
                const Icon(Icons.account_circle, size: 18),
                const SizedBox(width: 6),
                Expanded(
                  child: Text(
                    'Masuk sebagai ${auth.email}',
                    overflow: TextOverflow.ellipsis,
                  ),
                ),
              ],
            ),
          ),

          // ---------- Kolom pencarian ----------
          Padding(
            padding: const EdgeInsets.all(12),
            child: TextField(
              onChanged: (v) => controller.keyword.value = v,
              decoration: const InputDecoration(
                hintText: 'Cari nama atau NIM...',
                prefixIcon: Icon(Icons.search),
                border: OutlineInputBorder(),
              ),
            ),
          ),

          // ---------- Daftar data (reaktif) ----------
          Expanded(
            child: Obx(() {
              final data = controller.filteredList;

              // 1. Loading pertama kali (belum ada data sama sekali)
              if (controller.isLoading.value &&
                  controller.mahasiswaList.isEmpty) {
                return const Center(child: CircularProgressIndicator());
              }

              // 2. Gagal memuat (misalnya tidak ada internet)
              final error = controller.errorMessage.value;
              if (error != null) {
                return Center(
                  child: Padding(
                    padding: const EdgeInsets.all(24),
                    child: Column(
                      mainAxisSize: MainAxisSize.min,
                      children: [
                        const Icon(Icons.cloud_off, size: 48),
                        const SizedBox(height: 8),
                        Text(error, textAlign: TextAlign.center),
                        const SizedBox(height: 12),
                        OutlinedButton.icon(
                          onPressed: controller.loadData,
                          icon: const Icon(Icons.refresh),
                          label: const Text('Coba Lagi'),
                        ),
                      ],
                    ),
                  ),
                );
              }

              // 3. Data ada / kosong. RefreshIndicator = tarik ke bawah.
              return RefreshIndicator(
                onRefresh: controller.loadData,
                child: data.isEmpty
                    ? ListView(
                        physics: const AlwaysScrollableScrollPhysics(),
                        children: const [
                          SizedBox(height: 120),
                          Center(child: Text('Belum ada data mahasiswa')),
                        ],
                      )
                    : ListView.builder(
                        physics: const AlwaysScrollableScrollPhysics(),
                        padding: const EdgeInsets.only(bottom: 80),
                        itemCount: data.length,
                        itemBuilder: (context, index) {
                          final mhs = data[index];
                          return Card(
                            margin: const EdgeInsets.symmetric(
                              horizontal: 12,
                              vertical: 6,
                            ),
                            child: ListTile(
                              leading: CircleAvatar(
                                child: Text(mhs.nama[0].toUpperCase()),
                              ),
                              title: Text(mhs.nama),
                              subtitle: Text(
                                'NIM ${mhs.nim}\n${mhs.prodi} • Smt ${mhs.semester}',
                              ),
                              isThreeLine: true,
                              trailing: Row(
                                mainAxisSize: MainAxisSize.min,
                                children: [
                                  IconButton(
                                    visualDensity: VisualDensity.compact,
                                    icon: const Icon(
                                      Icons.edit,
                                      color: Colors.orange,
                                    ),
                                    onPressed: () =>
                                        controller.openEditForm(mhs),
                                  ),
                                  IconButton(
                                    visualDensity: VisualDensity.compact,
                                    icon: const Icon(
                                      Icons.delete,
                                      color: Colors.red,
                                    ),
                                    onPressed: () =>
                                        controller.confirmDelete(mhs),
                                  ),
                                ],
                              ),
                            ),
                          );
                        },
                      ),
              );
            }),
          ),
        ],
      ),
      floatingActionButton: FloatingActionButton.extended(
        onPressed: controller.openAddForm,
        icon: const Icon(Icons.add),
        label: const Text('Tambah'),
      ),
    );
  }
}

Yang baru di halaman daftar

  1. Tombol Keluar di AppBar: memanggil auth.confirmLogout. AuthController diambil dengan Get.find<AuthController>(), karena properti controller milik GetView sudah dipakai untuk MahasiswaController.
  2. Info akun: teks "Masuk sebagai ..." menampilkan email yang sedang login. Ini membantu saat menguji dengan dua akun.
  3. Empat keadaan di dalam Obx, diperiksa berurutan:
    UrutanKondisiYang tampil
    1isLoading dan daftar masih kosongLingkaran loading di tengah
    2errorMessage tidak nullIkon awan, pesan error, tombol Coba Lagi
    3Data kosong"Belum ada data mahasiswa"
    4Data adaDaftar kartu mahasiswa
    Syarat pertama memakai mahasiswaList.isEmpty agar saat refresh, daftar lama tetap tampil (tidak berkedip menjadi loading).
  4. Tarik untuk refresh: RefreshIndicator(onRefresh: controller.loadData). Karena loadData mengembalikan Future, putaran loading di atas daftar berhenti tepat setelah data selesai dimuat. AlwaysScrollableScrollPhysics membuat daftar tetap bisa ditarik walaupun isinya sedikit atau kosong.

2. MahasiswaFormView

Ganti isi lib/views/mahasiswa_form_view.dart. Satu-satunya perubahan dari Modul 11B ada di tombol paling bawah, yang sekarang dibungkus Obx:

📄 lib/views/mahasiswa_form_view.dart

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:get/get.dart';

import '../controllers/mahasiswa_controller.dart';

/// VIEW: halaman form untuk CREATE dan UPDATE.
class MahasiswaFormView extends GetView<MahasiswaController> {
  const MahasiswaFormView({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(controller.isEdit ? 'Edit Mahasiswa' : 'Tambah Mahasiswa'),
      ),
      body: Form(
        key: controller.formKey,
        // pesan error diperbarui otomatis saat user mengetik
        autovalidateMode: AutovalidateMode.onUserInteraction,
        child: ListView(
          padding: const EdgeInsets.all(16),
          children: [
            TextFormField(
              controller: controller.nimC,
              keyboardType: TextInputType.number,
              inputFormatters: [FilteringTextInputFormatter.digitsOnly],
              validator: controller.validateNim,
              decoration: const InputDecoration(
                labelText: 'NIM',
                prefixIcon: Icon(Icons.badge),
                border: OutlineInputBorder(),
              ),
            ),
            const SizedBox(height: 16),
            TextFormField(
              controller: controller.namaC,
              textCapitalization: TextCapitalization.words,
              validator: controller.validateNama,
              decoration: const InputDecoration(
                labelText: 'Nama Lengkap',
                prefixIcon: Icon(Icons.person),
                border: OutlineInputBorder(),
              ),
            ),
            const SizedBox(height: 16),
            DropdownButtonFormField<String>(
              initialValue: controller.selectedProdi.value,
              items: controller.daftarProdi
                  .map((p) => DropdownMenuItem(value: p, child: Text(p)))
                  .toList(),
              onChanged: (v) {
                if (v != null) controller.selectedProdi.value = v;
              },
              decoration: const InputDecoration(
                labelText: 'Program Studi',
                prefixIcon: Icon(Icons.school),
                border: OutlineInputBorder(),
              ),
            ),
            const SizedBox(height: 16),
            TextFormField(
              controller: controller.semesterC,
              keyboardType: TextInputType.number,
              inputFormatters: [FilteringTextInputFormatter.digitsOnly],
              validator: controller.validateSemester,
              decoration: const InputDecoration(
                labelText: 'Semester',
                prefixIcon: Icon(Icons.calendar_month),
                border: OutlineInputBorder(),
              ),
            ),
            const SizedBox(height: 24),
            Obx(
              () => FilledButton.icon(
                // nonaktif selama proses simpan ke Supabase
                onPressed: controller.isSaving.value ? null : controller.save,
                icon: controller.isSaving.value
                    ? const SizedBox(
                        height: 20,
                        width: 20,
                        child: CircularProgressIndicator(strokeWidth: 2),
                      )
                    : const Icon(Icons.save),
                label: Text(controller.isEdit ? 'Perbarui' : 'Simpan'),
                style: FilledButton.styleFrom(
                  padding: const EdgeInsets.symmetric(vertical: 16),
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }
}
