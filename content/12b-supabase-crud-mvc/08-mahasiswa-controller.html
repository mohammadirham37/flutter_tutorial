MahasiswaController dari Modul 11B hanya perlu diubah di beberapa tempat. Logika form, validator, pencarian, dan dialog hapus tetap sama. Yang berubah: data sekarang diambil lewat internet, sehingga perlu await, indikator loading, dan penanganan error.

Ganti seluruh isi lib/controllers/mahasiswa_controller.dart:

📄 lib/controllers/mahasiswa_controller.dart

import 'package:flutter/material.dart';
import 'package:get/get.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../models/mahasiswa_model.dart';
import '../routes/app_routes.dart';
import '../services/mahasiswa_service.dart';

/// CONTROLLER: penghubung antara View dan Model.
/// Isinya hampir sama dengan versi Hive. Perbedaannya: semua akses data
/// sekarang async (lewat internet), jadi ada state loading dan error.
class MahasiswaController extends GetxController {
  MahasiswaController(this._service);

  final MahasiswaService _service;

  // ------------------ STATE REAKTIF ------------------
  final mahasiswaList = <Mahasiswa>[].obs;
  final keyword = ''.obs;
  final selectedProdi = 'Informatika'.obs;
  final isLoading = false.obs; // sedang memuat daftar
  final isSaving = false.obs; // sedang menyimpan form
  final errorMessage = RxnString(); // null = tidak ada error

  final List<String> daftarProdi = const [
    'Informatika',
    'Sistem Informasi',
    'Teknik Komputer',
    'Data Science',
  ];

  // ------------------ STATE FORM ------------------
  final formKey = GlobalKey<FormState>();
  final nimC = TextEditingController();
  final namaC = TextEditingController();
  final semesterC = TextEditingController();

  /// null  -> mode TAMBAH
  /// berisi id -> mode EDIT
  String? editingId;
  bool get isEdit => editingId != null;

  /// Data yang ditampilkan (sudah difilter pencarian).
  List<Mahasiswa> get filteredList {
    final q = keyword.value.toLowerCase();
    if (q.isEmpty) return mahasiswaList;
    return mahasiswaList
        .where((m) => m.nama.toLowerCase().contains(q) || m.nim.contains(q))
        .toList();
  }

  @override
  void onInit() {
    super.onInit();
    loadData();
  }

  // ------------------ READ ------------------
  Future<void> loadData() async {
    isLoading.value = true;
    errorMessage.value = null;
    try {
      final data = await _service.getAll();
      mahasiswaList.assignAll(data);
    } catch (e) {
      errorMessage.value = _pesanError(e);
    } finally {
      isLoading.value = false;
    }
  }

  // ------------------ NAVIGASI KE FORM ------------------
  void openAddForm() {
    editingId = null;
    nimC.clear();
    namaC.clear();
    semesterC.clear();
    selectedProdi.value = daftarProdi.first;
    Get.toNamed(AppRoutes.form);
  }

  void openEditForm(Mahasiswa mhs) {
    editingId = mhs.id;
    nimC.text = mhs.nim;
    namaC.text = mhs.nama;
    semesterC.text = mhs.semester.toString();
    selectedProdi.value = mhs.prodi;
    Get.toNamed(AppRoutes.form);
  }

  // ------------------ CREATE & UPDATE ------------------
  Future<void> save() async {
    if (isSaving.value) return; // cegah tombol ditekan dua kali
    if (!formKey.currentState!.validate()) return;

    final nim = nimC.text.trim();

    // Cegah NIM ganda (cek cepat di memori sebelum ke server)
    final nimSudahAda = mahasiswaList.any(
      (m) => m.nim == nim && m.id != editingId,
    );
    if (nimSudahAda) {
      Get.snackbar(
        'Gagal',
        'NIM $nim sudah terdaftar',
        snackPosition: SnackPosition.BOTTOM,
      );
      return;
    }

    final mhs = Mahasiswa(
      nim: nim,
      nama: namaC.text.trim(),
      prodi: selectedProdi.value,
      semester: int.parse(semesterC.text.trim()),
    );

    isSaving.value = true;
    try {
      if (isEdit) {
        await _service.update(editingId!, mhs);
      } else {
        await _service.create(mhs);
      }

      final pesan = isEdit
          ? 'Data berhasil diperbarui'
          : 'Data berhasil ditambahkan';
      await loadData();
      tutup(); // kembali ke halaman daftar
      Get.snackbar('Sukses', pesan, snackPosition: SnackPosition.BOTTOM);
    } catch (e) {
      Get.snackbar(
        'Gagal',
        _pesanError(e),
        snackPosition: SnackPosition.BOTTOM,
      );
    } finally {
      isSaving.value = false;
    }
  }

  // ------------------ DELETE ------------------
  Future<void> confirmDelete(Mahasiswa mhs) async {
    final yakin = await Get.dialog<bool>(
      AlertDialog(
        title: const Text('Hapus Data'),
        content: Text('Yakin ingin menghapus ${mhs.nama}?'),
        actions: [
          TextButton(onPressed: () => tutup(false), child: const Text('Batal')),
          FilledButton(
            onPressed: () => tutup(true),
            child: const Text('Hapus'),
          ),
        ],
      ),
    );

    if (yakin != true) return;

    try {
      await _service.delete(mhs.id!);
      await loadData();
      Get.snackbar(
        'Terhapus',
        '${mhs.nama} telah dihapus',
        snackPosition: SnackPosition.BOTTOM,
      );
    } catch (e) {
      Get.snackbar(
        'Gagal',
        _pesanError(e),
        snackPosition: SnackPosition.BOTTOM,
      );
    }
  }

  // ------------------ HELPER ------------------
  /// Menutup halaman / dialog paling atas (opsional membawa [result]).
  void tutup<T>([T? result]) => Get.key.currentState?.pop<T>(result);

  /// Mengubah exception menjadi pesan yang mudah dipahami pengguna.
  String _pesanError(Object e) {
    if (e is PostgrestException) {
      if (e.code == '23505') return 'NIM sudah terdaftar';
      if (e.code == '42501') return 'Akses ditolak. Silakan login ulang.';
      return 'Kesalahan database: ${e.message}';
    }
    return 'Tidak dapat terhubung ke server. Periksa internet.';
  }

  // ------------------ VALIDATOR ------------------
  String? validateNim(String? v) {
    if (v == null || v.trim().isEmpty) return 'NIM wajib diisi';
    if (int.tryParse(v.trim()) == null) return 'NIM harus berupa angka';
    if (v.trim().length < 8) return 'NIM minimal 8 digit';
    return null;
  }

  String? validateNama(String? v) {
    if (v == null || v.trim().isEmpty) return 'Nama wajib diisi';
    return null;
  }

  String? validateSemester(String? v) {
    final s = int.tryParse(v?.trim() ?? '');
    if (s == null) return 'Semester wajib berupa angka';
    if (s < 1 || s > 14) return 'Semester antara 1 - 14';
    return null;
  }

  @override
  void onClose() {
    nimC.dispose();
    namaC.dispose();
    semesterC.dispose();
    super.onClose();
  }
}

Tiga state baru

final isLoading = false.obs;        // sedang memuat daftar
final isSaving = false.obs;         // sedang menyimpan form
final errorMessage = RxnString();   // null = tidak ada error
  • RxnString adalah versi reaktif dari String? (boleh null). Huruf n berarti nullable. Bentuk panjangnya Rxn<String>().
  • View membaca ketiganya di dalam Obx, sehingga tampilan berganti sendiri antara loading, error, kosong, dan daftar data.

loadData() dengan pola try / catch / finally

Future<void> loadData() async {
  isLoading.value = true;
  errorMessage.value = null;          // hapus error lama
  try {
    final data = await _service.getAll();
    mahasiswaList.assignAll(data);
  } catch (e) {
    errorMessage.value = _pesanError(e);
  } finally {
    isLoading.value = false;          // selalu dimatikan
  }
}

Pola ini dipakai di semua method yang menghubungi server. Hafalkan urutannya: nyalakan loading, coba, tangkap error, matikan loading. Method ini mengembalikan Future agar bisa dipakai oleh RefreshIndicator (tarik layar ke bawah) yang menunggu sampai proses selesai.

save(): mencegah tombol ditekan dua kali

Baris pertama if (isSaving.value) return; menghentikan penyimpanan kedua jika tombol ditekan saat proses pertama belum selesai. Tanpa baris ini, koneksi yang lambat bisa membuat data tersimpan dua kali. Tombol juga dinonaktifkan dari View selama isSaving bernilai true.

Lapisan pengaman NIM ganda

  1. Di aplikasi: mahasiswaList.any(...) mengecek daftar yang sudah dimuat, sama seperti Modul 11B. Hasilnya cepat dan tidak butuh internet.
  2. Di database: constraint unique (user_id, nim). Jika lapisan pertama terlewat (misalnya data ditambah dari HP lain), Postgres menolak dengan kode error 23505, lalu _pesanError menerjemahkannya menjadi "NIM sudah terdaftar".

_pesanError: menerjemahkan error database

ErrorArtinyaPesan
PostgrestException kode 23505Melanggar constraint uniqueNIM sudah terdaftar
PostgrestException kode 42501Ditolak oleh RLS atau izinAkses ditolak. Silakan login ulang.
PostgrestException lainnyaKesalahan database lainKesalahan database: ...
Selain ituUmumnya tidak ada internet atau server tidak terjangkauTidak dapat terhubung ke server. Periksa internet.

💡 mhs.id! di confirmDelete

Tanda seru berarti "saya yakin nilai ini tidak null". Ini aman karena data yang tampil di daftar selalu berasal dari fromJson, sehingga pasti punya id.

Coba jalankan: MahasiswaController dengan halaman sederhana

Controller ini butuh pengguna yang sudah login, karena RLS hanya menampilkan data milik akun tersebut. Gunakan sesi dari percobaan section 6 atau 7: selama Anda belum menekan Keluar, sesi tetap tersimpan di HP. Ganti seluruh isi lib/main.dart dengan:

📄 lib/main.dart (sementara)

// lib/main.dart (SEMENTARA) — mencoba MahasiswaController.
// Syarat: sudah login lewat contoh section 6 atau 7 (sesi tersimpan di HP).
// Jalankan: flutter run --dart-define-from-file=env.json
import 'package:flutter/material.dart';
import 'package:get/get.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import 'config/env.dart';
import 'controllers/mahasiswa_controller.dart';
import 'routes/app_routes.dart';
import 'services/mahasiswa_service.dart';

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await Supabase.initialize(
    url: Env.supabaseUrl,
    publishableKey: Env.supabaseKey,
  );
  Get.put(MahasiswaService(Supabase.instance.client), permanent: true);
  runApp(const CobaControllerApp());
}

class CobaControllerApp extends StatelessWidget {
  const CobaControllerApp({super.key});

  @override
  Widget build(BuildContext context) {
    return GetMaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo, useMaterial3: true),
      initialRoute: AppRoutes.home,
      getPages: [
        GetPage(
          name: AppRoutes.home,
          page: () => const DemoDaftarPage(),
          binding: BindingsBuilder(() {
            Get.put(MahasiswaController(Get.find()));
          }),
        ),
        GetPage(name: AppRoutes.form, page: () => const DemoFormPage()),
      ],
    );
  }
}

/// Halaman daftar versi sederhana (versi rapi dibuat di section 11).
class DemoDaftarPage extends GetView<MahasiswaController> {
  const DemoDaftarPage({super.key});

  @override
  Widget build(BuildContext context) {
    final user = Supabase.instance.client.auth.currentUser;
    return Scaffold(
      appBar: AppBar(
        title: Text(user?.email ?? 'BELUM login'),
        actions: [
          IconButton(
            icon: const Icon(Icons.refresh),
            onPressed: controller.loadData,
          ),
        ],
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: controller.openAddForm,
        child: const Icon(Icons.add),
      ),
      body: Column(
        children: [
          Padding(
            padding: const EdgeInsets.all(8),
            child: TextField(
              onChanged: (v) => controller.keyword.value = v,
              decoration: const InputDecoration(
                hintText: 'Cari nama / NIM',
                prefixIcon: Icon(Icons.search),
              ),
            ),
          ),
          Expanded(
            child: Obx(() {
              if (controller.isLoading.value) {
                return const Center(child: CircularProgressIndicator());
              }
              final error = controller.errorMessage.value;
              if (error != null) return Center(child: Text(error));
              final data = controller.filteredList;
              if (data.isEmpty) {
                return const Center(child: Text('Belum ada data'));
              }
              return ListView.builder(
                itemCount: data.length,
                itemBuilder: (context, i) {
                  final m = data[i];
                  return ListTile(
                    title: Text(m.nama),
                    subtitle: Text('${m.nim} - ${m.prodi} - smt ${m.semester}'),
                    onTap: () => controller.openEditForm(m),
                    trailing: IconButton(
                      icon: const Icon(Icons.delete),
                      onPressed: () => controller.confirmDelete(m),
                    ),
                  );
                },
              );
            }),
          ),
        ],
      ),
    );
  }
}

class DemoFormPage extends GetView<MahasiswaController> {

Form sederhana terisi
Tersimpan: daftar dimuat ulang + snackbar

Jalankan dengan flutter run --dart-define-from-file=env.json, lalu coba:

  1. Saat dibuka, indikator loading tampil sebentar (isLoading), lalu daftar muncul. Data yang dibuat di section 6 ikut tampil.
  2. Tekan +, isi NIM, nama, dan semester, lalu Simpan. Halaman form tertutup, daftar dimuat ulang, dan snackbar "Data berhasil ditambahkan" muncul.
  3. Tambah lagi dengan NIM yang sama. Muncul "NIM ... sudah terdaftar" tanpa menghubungi server.
  4. Ketuk salah satu data untuk mode edit, ubah semesternya, lalu simpan.
  5. Ketik sebagian nama di kolom cari. Daftar tersaring (filteredList).
  6. Tekan ikon hapus, lalu pilih Hapus di dialog konfirmasi.
  7. Matikan internet emulator lalu tekan ikon refresh. Muncul "Tidak dapat terhubung ke server. Periksa internet." dari errorMessage.
  • BindingsBuilder membuat MahasiswaController saat route home dibuka. Di section 9, tugas ini dipindahkan ke MahasiswaBinding.
  • Halaman daftar membaca tiga state dalam satu Obx: isLoading, lalu errorMessage, lalu filteredList. Urutan pengecekan ini sama dengan view final di section 11.
  • Halaman form memakai formKey, nimC, namaC, dan semesterC milik controller. Prodi tidak ditampilkan di sini, sehingga nilainya memakai selectedProdi bawaan (Informatika). Dropdown prodi dibuat di section 11.

⚠️ Judul AppBar "BELUM login"?

Berarti Anda sudah logout di percobaan sebelumnya. Daftar akan kosong dan tombol Simpan menampilkan "Akses ditolak. Silakan login ulang." Jalankan lagi contoh section 7, login, lalu kembali ke contoh ini.
