MahasiswaController dari Modul 11B hanya perlu diubah di beberapa tempat. Logika form, validator, pencarian, dan dialog hapus tetap sama. Yang berubah: data sekarang diambil lewat internet, sehingga perlu await, indikator loading, dan penanganan error.

Ganti seluruh isi lib/controllers/mahasiswa_controller.dart:

📄 lib/controllers/mahasiswa_controller.dart

import 'package:flutter/material.dart';
import 'package:get/get.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../models/mahasiswa_model.dart';
import '../routes/app_routes.dart';
import '../services/mahasiswa_service.dart';

/// CONTROLLER: penghubung antara View dan Model.
/// Isinya hampir sama dengan versi Hive. Perbedaannya: semua akses data
/// sekarang async (lewat internet), jadi ada state loading dan error.
class MahasiswaController extends GetxController {
  MahasiswaController(this._service);

  final MahasiswaService _service;

  // ------------------ STATE REAKTIF ------------------
  final mahasiswaList = <Mahasiswa>[].obs;
  final keyword = ''.obs;
  final selectedProdi = 'Informatika'.obs;
  final isLoading = false.obs; // sedang memuat daftar
  final isSaving = false.obs; // sedang menyimpan form
  final errorMessage = RxnString(); // null = tidak ada error

  final List<String> daftarProdi = const [
    'Informatika',
    'Sistem Informasi',
    'Teknik Komputer',
    'Data Science',
  ];

  // ------------------ STATE FORM ------------------
  final formKey = GlobalKey<FormState>();
  final nimC = TextEditingController();
  final namaC = TextEditingController();
  final semesterC = TextEditingController();

  /// null  -> mode TAMBAH
  /// berisi id -> mode EDIT
  String? editingId;
  bool get isEdit => editingId != null;

  /// Data yang ditampilkan (sudah difilter pencarian).
  List<Mahasiswa> get filteredList {
    final q = keyword.value.toLowerCase();
    if (q.isEmpty) return mahasiswaList;
    return mahasiswaList
        .where((m) => m.nama.toLowerCase().contains(q) || m.nim.contains(q))
        .toList();
  }

  @override
  void onInit() {
    super.onInit();
    loadData();
  }

  // ------------------ READ ------------------
  Future<void> loadData() async {
    isLoading.value = true;
    errorMessage.value = null;
    try {
      final data = await _service.getAll();
      mahasiswaList.assignAll(data);
    } catch (e) {
      errorMessage.value = _pesanError(e);
    } finally {
      isLoading.value = false;
    }
  }

  // ------------------ NAVIGASI KE FORM ------------------
  void openAddForm() {
    editingId = null;
    nimC.clear();
    namaC.clear();
    semesterC.clear();
    selectedProdi.value = daftarProdi.first;
    Get.toNamed(AppRoutes.form);
  }

  void openEditForm(Mahasiswa mhs) {
    editingId = mhs.id;
    nimC.text = mhs.nim;
    namaC.text = mhs.nama;
    semesterC.text = mhs.semester.toString();
    selectedProdi.value = mhs.prodi;
    Get.toNamed(AppRoutes.form);
  }

  // ------------------ CREATE & UPDATE ------------------
  Future<void> save() async {
    if (isSaving.value) return; // cegah tombol ditekan dua kali
    if (!formKey.currentState!.validate()) return;

    final nim = nimC.text.trim();

    // Cegah NIM ganda (cek cepat di memori sebelum ke server)
    final nimSudahAda = mahasiswaList.any(
      (m) => m.nim == nim && m.id != editingId,
    );
    if (nimSudahAda) {
      Get.snackbar(
        'Gagal',
        'NIM $nim sudah terdaftar',
        snackPosition: SnackPosition.BOTTOM,
      );
      return;
    }

    final mhs = Mahasiswa(
      nim: nim,
      nama: namaC.text.trim(),
      prodi: selectedProdi.value,
      semester: int.parse(semesterC.text.trim()),
    );

    isSaving.value = true;
    try {
      if (isEdit) {
        await _service.update(editingId!, mhs);
      } else {
        await _service.create(mhs);
      }

      final pesan = isEdit
          ? 'Data berhasil diperbarui'
          : 'Data berhasil ditambahkan';
      await loadData();
      tutup(); // kembali ke halaman daftar
      Get.snackbar('Sukses', pesan, snackPosition: SnackPosition.BOTTOM);
    } catch (e) {
      Get.snackbar(
        'Gagal',
        _pesanError(e),
        snackPosition: SnackPosition.BOTTOM,
      );
    } finally {
      isSaving.value = false;
    }
  }

  // ------------------ DELETE ------------------
  Future<void> confirmDelete(Mahasiswa mhs) async {
    final yakin = await Get.dialog<bool>(
      AlertDialog(
        title: const Text('Hapus Data'),
        content: Text('Yakin ingin menghapus ${mhs.nama}?'),
        actions: [
          TextButton(onPressed: () => tutup(false), child: const Text('Batal')),
          FilledButton(
            onPressed: () => tutup(true),
            child: const Text('Hapus'),
          ),
        ],
      ),
    );

    if (yakin != true) return;

    try {
      await _service.delete(mhs.id!);
      await loadData();
      Get.snackbar(
        'Terhapus',
        '${mhs.nama} telah dihapus',
        snackPosition: SnackPosition.BOTTOM,
      );
    } catch (e) {
      Get.snackbar(
        'Gagal',
        _pesanError(e),
        snackPosition: SnackPosition.BOTTOM,
      );
    }
  }

  // ------------------ HELPER ------------------
  /// Menutup halaman / dialog paling atas (opsional membawa [result]).
  void tutup<T>([T? result]) => Get.key.currentState?.pop<T>(result);

  /// Mengubah exception menjadi pesan yang mudah dipahami pengguna.
  String _pesanError(Object e) {
    if (e is PostgrestException) {
      if (e.code == '23505') return 'NIM sudah terdaftar';
      if (e.code == '42501') return 'Akses ditolak. Silakan login ulang.';
      return 'Kesalahan database: ${e.message}';
    }
    return 'Tidak dapat terhubung ke server. Periksa internet.';
  }

  // ------------------ VALIDATOR ------------------
  String? validateNim(String? v) {
    if (v == null || v.trim().isEmpty) return 'NIM wajib diisi';
    if (int.tryParse(v.trim()) == null) return 'NIM harus berupa angka';
    if (v.trim().length < 8) return 'NIM minimal 8 digit';
    return null;
  }

  String? validateNama(String? v) {
    if (v == null || v.trim().isEmpty) return 'Nama wajib diisi';
    return null;
  }

  String? validateSemester(String? v) {
    final s = int.tryParse(v?.trim() ?? '');
    if (s == null) return 'Semester wajib berupa angka';
    if (s < 1 || s > 14) return 'Semester antara 1 - 14';
    return null;
  }

  @override
  void onClose() {
    nimC.dispose();
    namaC.dispose();
    semesterC.dispose();
    super.onClose();
  }
}

Tiga state baru

final isLoading = false.obs;        // sedang memuat daftar
final isSaving = false.obs;         // sedang menyimpan form
final errorMessage = RxnString();   // null = tidak ada error
  • RxnString adalah versi reaktif dari String? (boleh null). Huruf n berarti nullable. Bentuk panjangnya Rxn<String>().
  • View membaca ketiganya di dalam Obx, sehingga tampilan berganti sendiri antara loading, error, kosong, dan daftar data.

loadData() dengan pola try / catch / finally

Future<void> loadData() async {
  isLoading.value = true;
  errorMessage.value = null;          // hapus error lama
  try {
    final data = await _service.getAll();
    mahasiswaList.assignAll(data);
  } catch (e) {
    errorMessage.value = _pesanError(e);
  } finally {
    isLoading.value = false;          // selalu dimatikan
  }
}

Pola ini dipakai di semua method yang menghubungi server. Hafalkan urutannya: nyalakan loading, coba, tangkap error, matikan loading. Method ini mengembalikan Future agar bisa dipakai oleh RefreshIndicator (tarik layar ke bawah) yang menunggu sampai proses selesai.

save(): mencegah tombol ditekan dua kali

Baris pertama if (isSaving.value) return; menghentikan penyimpanan kedua jika tombol ditekan saat proses pertama belum selesai. Tanpa baris ini, koneksi yang lambat bisa membuat data tersimpan dua kali. Tombol juga dinonaktifkan dari View selama isSaving bernilai true.

Lapisan pengaman NIM ganda

  1. Di aplikasi: mahasiswaList.any(...) mengecek daftar yang sudah dimuat, sama seperti Modul 11B. Hasilnya cepat dan tidak butuh internet.
  2. Di database: constraint unique (user_id, nim). Jika lapisan pertama terlewat (misalnya data ditambah dari HP lain), Postgres menolak dengan kode error 23505, lalu _pesanError menerjemahkannya menjadi "NIM sudah terdaftar".

Tanda seru berarti "saya yakin nilai ini tidak null". Ini aman karena data yang tampil di daftar selalu berasal dari fromJson, sehingga pasti punya id.
