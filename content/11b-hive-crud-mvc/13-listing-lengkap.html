/// MODEL: representasi data Mahasiswa.
/// Class ini murni berisi data (tidak tahu soal UI maupun GetX).
class Mahasiswa {
  final String id;
  final String nim;
  final String nama;
  final String prodi;
  final int semester;

  Mahasiswa({
    required this.id,
    required this.nim,
    required this.nama,
    required this.prodi,
    required this.semester,
  });

  /// Membuat salinan objek dengan sebagian field yang diubah.
  Mahasiswa copyWith({
    String? nim,
    String? nama,
    String? prodi,
    int? semester,
  }) {
    return Mahasiswa(
      id: id,
      nim: nim ?? this.nim,
      nama: nama ?? this.nama,
      prodi: prodi ?? this.prodi,
      semester: semester ?? this.semester,
    );
  }
}

3. TypeAdapter

📄 lib/models/mahasiswa_adapter.dart

import 'package:hive_ce/hive_ce.dart';

import 'mahasiswa_model.dart';

/// TypeAdapter memberi tahu Hive cara menyimpan (write)
/// dan membaca (read) objek Mahasiswa dalam bentuk biner.
///
/// ATURAN PENTING:
/// 1. typeId harus unik per class (0 - 223).
/// 2. Urutan read HARUS sama persis dengan urutan write.
class MahasiswaAdapter extends TypeAdapter<Mahasiswa> {
  @override
  final int typeId = 0;

  @override
  Mahasiswa read(BinaryReader reader) {
    return Mahasiswa(
      id: reader.readString(),
      nim: reader.readString(),
      nama: reader.readString(),
      prodi: reader.readString(),
      semester: reader.readInt(),
    );
  }

  @override
  void write(BinaryWriter writer, Mahasiswa obj) {
    writer.writeString(obj.id);
    writer.writeString(obj.nim);
    writer.writeString(obj.nama);
    writer.writeString(obj.prodi);
    writer.writeInt(obj.semester);
  }
}

4. HiveService

📄 lib/services/hive_service.dart

import 'package:hive_ce/hive_ce.dart';

import '../models/mahasiswa_model.dart';

/// Lapisan akses data (bagian dari MODEL pada MVC).
/// Semua operasi CRUD ke Hive dikumpulkan di sini,
/// sehingga Controller tidak perlu tahu detail database.
class HiveService {
  static const String boxName = 'mahasiswa_box';

  Box<Mahasiswa> get _box => Hive.box<Mahasiswa>(boxName);

  // READ
  List<Mahasiswa> getAll() => _box.values.toList();

  // CREATE (key = id mahasiswa)
  Future<void> create(Mahasiswa mhs) => _box.put(mhs.id, mhs);

  // UPDATE (put dengan key yang sama akan menimpa data lama)
  Future<void> update(Mahasiswa mhs) => _box.put(mhs.id, mhs);

  // DELETE
  Future<void> delete(String id) => _box.delete(id);
}

5. MahasiswaController

📄 lib/controllers/mahasiswa_controller.dart

import 'package:flutter/material.dart';
import 'package:get/get.dart';

import '../models/mahasiswa_model.dart';
import '../routes/app_routes.dart';
import '../services/hive_service.dart';

/// CONTROLLER: penghubung antara View dan Model.
/// Menyimpan state (reaktif) dan berisi seluruh logika CRUD.
class MahasiswaController extends GetxController {
  final HiveService _service = HiveService();

  // ------------------ STATE REAKTIF ------------------
  final mahasiswaList = <Mahasiswa>[].obs;
  final keyword = ''.obs;
  final selectedProdi = 'Informatika'.obs;

  final List<String> daftarProdi = const [
    'Informatika',
    'Sistem Informasi',
    'Teknik Komputer',
    'Data Science',
  ];

  // ------------------ STATE FORM ------------------
  final formKey = GlobalKey<FormState>();
  final nimC = TextEditingController();
  final namaC = TextEditingController();
  final semesterC = TextEditingController();

  /// null  -> mode TAMBAH
  /// berisi id -> mode EDIT
  String? editingId;
  bool get isEdit => editingId != null;

  /// Data yang ditampilkan (sudah difilter pencarian).
  List<Mahasiswa> get filteredList {
    final q = keyword.value.toLowerCase();
    if (q.isEmpty) return mahasiswaList;
    return mahasiswaList
        .where((m) =>
            m.nama.toLowerCase().contains(q) || m.nim.contains(q))
        .toList();
  }

  @override
  void onInit() {
    super.onInit();
    loadData();
  }

  // ------------------ READ ------------------
  void loadData() {
    final data = _service.getAll()
      ..sort((a, b) => a.nama.toLowerCase().compareTo(b.nama.toLowerCase()));
    mahasiswaList.assignAll(data);
  }

  // ------------------ NAVIGASI KE FORM ------------------
  void openAddForm() {
    editingId = null;
    nimC.clear();
    namaC.clear();
    semesterC.clear();
    selectedProdi.value = daftarProdi.first;
    Get.toNamed(AppRoutes.form);
  }

  void openEditForm(Mahasiswa mhs) {
    editingId = mhs.id;
    nimC.text = mhs.nim;
    namaC.text = mhs.nama;
    semesterC.text = mhs.semester.toString();
    selectedProdi.value = mhs.prodi;
    Get.toNamed(AppRoutes.form);
  }

  // ------------------ CREATE & UPDATE ------------------
  Future<void> save() async {
    if (!formKey.currentState!.validate()) return;

    final nim = nimC.text.trim();

    // Cegah NIM ganda
    final nimSudahAda =
        mahasiswaList.any((m) => m.nim == nim && m.id != editingId);
    if (nimSudahAda) {
      Get.snackbar('Gagal', 'NIM $nim sudah terdaftar',
          snackPosition: SnackPosition.BOTTOM);
      return;
    }

    final mhs = Mahasiswa(
      id: editingId ?? DateTime.now().millisecondsSinceEpoch.toString(),
      nim: nim,
      nama: namaC.text.trim(),
      prodi: selectedProdi.value,
      semester: int.parse(semesterC.text.trim()),
    );

    if (isEdit) {
      await _service.update(mhs);
    } else {
      await _service.create(mhs);
    }

    final pesan = isEdit ? 'Data berhasil diperbarui' : 'Data berhasil ditambahkan';
    loadData();
    tutup(); // kembali ke halaman daftar
    Get.snackbar('Sukses', pesan, snackPosition: SnackPosition.BOTTOM);
  }

  // ------------------ DELETE ------------------
  Future<void> confirmDelete(Mahasiswa mhs) async {
    final yakin = await Get.dialog<bool>(
      AlertDialog(
        title: const Text('Hapus Data'),
        content: Text('Yakin ingin menghapus ${mhs.nama}?'),
        actions: [
          TextButton(
            onPressed: () => tutup(false),
            child: const Text('Batal'),
          ),
          FilledButton(
            onPressed: () => tutup(true),
            child: const Text('Hapus'),
          ),
        ],
      ),
    );

    if (yakin == true) {
      await _service.delete(mhs.id);
      loadData();
      Get.snackbar('Terhapus', '${mhs.nama} telah dihapus',
          snackPosition: SnackPosition.BOTTOM);
    }
  }

  // ------------------ HELPER NAVIGASI ------------------
  /// Menutup halaman / dialog paling atas (opsional membawa [result]).
  ///
  /// Kenapa tidak memakai Get.back()? Pada GetX 4.x, jika masih ada
  /// snackbar yang tampil, Get.back() HANYA menutup snackbar tersebut,
  /// sehingga halaman form / dialog tidak tertutup. Get.key adalah
  /// navigator milik GetMaterialApp, jadi pop() selalu menutup route.
  void tutup<T>([T? result]) => Get.key.currentState?.pop<T>(result);

  // ------------------ VALIDATOR ------------------
  String? validateNim(String? v) {
    if (v == null || v.trim().isEmpty) return 'NIM wajib diisi';
    if (int.tryParse(v.trim()) == null) return 'NIM harus berupa angka';
    if (v.trim().length < 8) return 'NIM minimal 8 digit';
    return null;
  }

  String? validateNama(String? v) {
    if (v == null || v.trim().isEmpty) return 'Nama wajib diisi';
    return null;
  }

  String? validateSemester(String? v) {
    final s = int.tryParse(v?.trim() ?? '');
    if (s == null) return 'Semester wajib berupa angka';
    if (s < 1 || s > 14) return 'Semester antara 1 - 14';
    return null;
  }

  @override
  void onClose() {
    nimC.dispose();
    namaC.dispose();
    semesterC.dispose();
    super.onClose();
  }
}

/// VIEW: halaman daftar mahasiswa (READ + tombol Edit/Hapus).

  const MahasiswaListView({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Data Mahasiswa'),
        centerTitle: true,
      ),
      body: Column(
        children: [
          // ---------- Kolom pencarian ----------

            padding: const EdgeInsets.all(12),

              decoration: const InputDecoration(
                hintText: 'Cari nama atau NIM...',
                prefixIcon: Icon(Icons.search),
                border: OutlineInputBorder(),
              ),
            ),
          ),

          // ---------- Daftar data (reaktif) ----------
          Expanded(
            child: Obx(() {
              final data = controller.filteredList;

              if (data.isEmpty) {
                return const Center(
                  child: Text('Belum ada data mahasiswa'),

              }

              return ListView.builder(
                padding: const EdgeInsets.only(bottom: 80),
                itemCount: data.length,
                itemBuilder: (context, index) {
                  final mhs = data[index];
                  return Card(
                    margin: const EdgeInsets.symmetric(
                        horizontal: 12, vertical: 6),
                    child: ListTile(
                      leading: CircleAvatar(
                        child: Text(mhs.nama[0].toUpperCase()),
                      ),
                      title: Text(mhs.nama),
                      subtitle: Text(
                          'NIM ${mhs.nim}\n${mhs.prodi} • Smt ${mhs.semester}'),
                      isThreeLine: true,
                      trailing: Row(
                        mainAxisSize: MainAxisSize.min,
                        children: [
                          IconButton(
                            visualDensity: VisualDensity.compact,
                            icon: const Icon(Icons.edit, color: Colors.orange),
                            onPressed: () => controller.openEditForm(mhs),
                          ),
                          IconButton(
                            visualDensity: VisualDensity.compact,

        icon: const Icon(Icons.add),
        label: const Text('Tambah'),

import 'dart:io';

import 'package:crud_mahasiswa/models/mahasiswa_adapter.dart';
import 'package:crud_mahasiswa/models/mahasiswa_model.dart';
import 'package:crud_mahasiswa/services/hive_service.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:hive_ce/hive_ce.dart';

void main() {
  late Directory dir;
  final service = HiveService();

  setUpAll(() async {
    dir = await Directory.systemTemp.createTemp('hive_test');
    Hive.init(dir.path);
    Hive.registerAdapter(MahasiswaAdapter());
    await Hive.openBox<Mahasiswa>(HiveService.boxName);
  });

  tearDownAll(() async {
    await Hive.deleteFromDisk();
    await dir.delete(recursive: true);
  });

  final budi = Mahasiswa(
      id: '1', nim: '220101001', nama: 'Budi', prodi: 'Informatika', semester: 5);

  test('CREATE & READ', () async {
    await service.create(budi);
    final all = service.getAll();
    expect(all.length, 1);
    expect(all.first.nama, 'Budi');
    expect(all.first.semester, 5);
  });

  test('UPDATE', () async {
    await service.update(budi.copyWith(nama: 'Budi Santoso', semester: 6));
    final all = service.getAll();
    expect(all.length, 1);
    expect(all.first.nama, 'Budi Santoso');
    expect(all.first.semester, 6);
  });

  test('Data tetap ada setelah box ditutup & dibuka lagi', () async {
    await Hive.box<Mahasiswa>(HiveService.boxName).close();
    await Hive.openBox<Mahasiswa>(HiveService.boxName);
    expect(service.getAll().first.nama, 'Budi Santoso');
  });

  test('DELETE', () async {
    await service.delete('1');
    expect(service.getAll(), isEmpty);
  });
}

/// Tap tombol lalu beri waktu nyata agar operasi IO Hive selesai.
Future<void> tapDanTunggu(WidgetTester tester, Finder f) async {
  await tester.tap(f);
  for (var i = 0; i < 5; i++) {
    await tester.pump(const Duration(milliseconds: 300));
    await tester.runAsync(
        () => Future.delayed(const Duration(milliseconds: 100)));
  }
}

void main() {
  testWidgets('Alur CRUD lewat UI', (tester) async {
    await tester.runAsync(() async {
      final dir = await Directory.systemTemp.createTemp('hive_ui');
      Hive.init(dir.path);
      Hive.registerAdapter(MahasiswaAdapter());
      await Hive.openBox<Mahasiswa>(HiveService.boxName);
    });

    await tester.pumpWidget(const MyApp());
    await tester.pumpAndSettle();
    expect(find.text('Belum ada data mahasiswa'), findsOneWidget);

    // CREATE
    await tester.tap(find.text('Tambah'));
    await tester.pumpAndSettle();
    await tester.enterText(find.widgetWithText(TextFormField, 'NIM'), '22010100');
    await tester.enterText(find.widgetWithText(TextFormField, 'Nama Lengkap'), 'Siti');
    await tester.enterText(find.widgetWithText(TextFormField, 'Semester'), '3');
    await tapDanTunggu(tester, find.text('Simpan'));
    await tester.pumpAndSettle(const Duration(seconds: 4));
    expect(find.text('Siti'), findsOneWidget);

    // UPDATE
    await tester.tap(find.byIcon(Icons.edit));
    await tester.pumpAndSettle();
    await tester.enterText(find.widgetWithText(TextFormField, 'Nama Lengkap'), 'Siti Aminah');
    await tapDanTunggu(tester, find.text('Perbarui'));
    await tester.pumpAndSettle(const Duration(seconds: 4));
    expect(find.text('Siti Aminah'), findsOneWidget);

    // DELETE
    await tester.tap(find.byIcon(Icons.delete));
    await tester.pumpAndSettle();
    await tapDanTunggu(tester, find.text('Hapus'));
    await tester.pumpAndSettle(const Duration(seconds: 4));
    expect(find.text('Belum ada data mahasiswa'), findsOneWidget);
    Get.reset();
  });
}
