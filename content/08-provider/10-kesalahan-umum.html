Bagian ini merangkum kesalahan Provider yang paling sering muncul di praktikum, lengkap dengan pesan error dan perbaikannya.

1. read di build, watch di callback

// ❌ SALAH
class TampilanSalah extends StatelessWidget {
  const TampilanSalah({super.key});

  @override
  Widget build(BuildContext context) {
    // read di build: angka TIDAK pernah ter-update di layar
    final nilai = context.read<CounterModel>().nilai;
    return ElevatedButton(
      // watch di callback: error "Tried to listen to a value exposed
      // with provider, from outside of the widget tree"
      onPressed: () => context.watch<CounterModel>().tambah(),
      child: Text('$nilai'),
    );
  }
}

// ✅ BENAR
class TampilanBenar extends StatelessWidget {
  const TampilanBenar({super.key});

  @override
  Widget build(BuildContext context) {
    final nilai = context.watch<CounterModel>().nilai; // watch di build
    return ElevatedButton(
      onPressed: () => context.read<CounterModel>().tambah(), // read di callback
      child: Text('$nilai'),
    );
  }
}

2. Memanggil provider di initState

initState berjalan saat widget sedang dibangun. Jika method provider langsung memanggil notifyListeners(), Flutter menolak karena ada widget lain yang diminta rebuild di tengah proses build.

@override
void initState() {
  super.initState();

  // ❌ SALAH: muatData() langsung memanggil notifyListeners() padahal
  // Flutter sedang membangun widget -> error
  // "setState() or markNeedsBuild() called during build"
  // context.read<DataProvider>().muatData();

  // ✅ BENAR: jalankan setelah frame pertama selesai digambar
  WidgetsBinding.instance.addPostFrameCallback((_) {
    if (!mounted) return;
    context.read<DataProvider>().muatData();
  });

  // ✅ Setara, gaya lama: Provider.of dengan listen: false
  // WidgetsBinding.instance.addPostFrameCallback((_) {
  //   Provider.of<DataProvider>(context, listen: false).muatData();
  // });
}

// ❌ context.watch TIDAK boleh dipanggil di initState (hanya di build)

3. notifyListeners setelah dispose

Provider yang dibuat khusus untuk satu halaman akan di-dispose saat halaman ditutup. Jika request async masih berjalan, notifyListeners() di akhir request akan error.

import 'package:flutter/foundation.dart';

class PencarianProvider extends ChangeNotifier {
  bool _sudahDispose = false;
  List<String> hasil = [];

  Future<void> cari(String kata) async {
    await Future.delayed(const Duration(seconds: 3)); // request lama
    hasil = ['$kata 1', '$kata 2'];
    // Jika halaman sudah ditutup selama 3 detik itu, provider sudah di-dispose.
    // Memanggil notifyListeners() sekarang -> error
    // "A PencarianProvider was used after being disposed".
    if (_sudahDispose) return; // ✅ penjaga
    notifyListeners();
  }

  @override
  void dispose() {
    _sudahDispose = true;
    super.dispose();
  }
}

4. Meneruskan provider ke rute baru

// ✅ Meneruskan provider yang SUDAH ADA ke rute baru: pakai .value
final keranjang = context.read<KeranjangProvider>();
Navigator.push(
  context,
  MaterialPageRoute(
    builder: (_) => ChangeNotifierProvider.value(
      value: keranjang,
      child: const DetailPage(),
    ),
  ),
);

// ❌ SALAH: create: (_) => keranjang  -> provider akan men-dispose
//    objek milik orang lain saat DetailPage ditutup.
// ❌ SALAH: ChangeNotifierProvider.value(value: KeranjangProvider())
//    -> objek baru dibuat ulang setiap build dan tidak pernah di-dispose.

Contoh lengkap: semua perbaikan dalam satu aplikasi

📄 lib/main.dart

// lib/main.dart — salin SELURUH kode ini untuk menggantikan isi lib/main.dart
// Contoh BENAR: memuat data di initState + berbagi provider ke halaman lain
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

class PengumumanProvider extends ChangeNotifier {
  bool _sudahDispose = false;
  bool _isLoading = false;
  List<String> _items = [];
  final Set<String> _dibaca = {};

  bool get isLoading => _isLoading;
  List<String> get items => List.unmodifiable(_items);
  bool sudahDibaca(String p) => _dibaca.contains(p);

  Future<void> muat() async {
    _isLoading = true;
    _aman();
    await Future.delayed(const Duration(seconds: 1));
    _items = ['Libur Dies Natalis', 'Jadwal UTS', 'Beasiswa PPA'];
    _isLoading = false;
    _aman();
  }

  void tandaiDibaca(String p) {
    _dibaca.add(p);
    _aman();
  }

  // notifyListeners yang tidak crash walau provider sudah di-dispose
  void _aman() {
    if (!_sudahDispose) notifyListeners();
  }

  @override
  void dispose() {
    _sudahDispose = true;
    super.dispose();
  }
}

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.blueGrey),
      home: Scaffold(
        appBar: AppBar(title: const Text('Menu')),
        body: Builder(
          builder: (context) => Center(
            child: FilledButton(
              onPressed: () => Navigator.push(
                context,
                MaterialPageRoute(
                  // Provider khusus halaman: dibuat saat halaman dibuka,
                  // di-dispose otomatis saat halaman ditutup.
                  builder: (_) => ChangeNotifierProvider(
                    create: (_) => PengumumanProvider(),
                    child: const PengumumanPage(),
                  ),
                ),
              ),
              child: const Text('Buka Pengumuman'),
            ),
          ),
        ),
      ),
    );
  }
}

class PengumumanPage extends StatefulWidget {
  const PengumumanPage({super.key});

  @override
  State<PengumumanPage> createState() => _PengumumanPageState();
}

class _PengumumanPageState extends State<PengumumanPage> {
  @override
  void initState() {
    super.initState();
    // Muat data SETELAH frame pertama, bukan langsung di initState
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (!mounted) return;
      context.read<PengumumanProvider>().muat();
    });
  }

  @override
  Widget build(BuildContext context) {
    final prov = context.watch<PengumumanProvider>();

    return Scaffold(
      appBar: AppBar(title: const Text('Pengumuman')),
      body: prov.isLoading
          ? const Center(child: CircularProgressIndicator())
          : ListView(
              children: [
                for (final p in prov.items)
                  ListTile(
                    title: Text(p),
                    trailing: prov.sudahDibaca(p)
                        ? const Icon(Icons.done_all, color: Colors.green)
                        : null,
                    onTap: () {
                      // Bagikan provider yang SAMA ke halaman detail
                      final pengumuman = context.read<PengumumanProvider>();
                      Navigator.push(
                        context,
                        MaterialPageRoute(
                          builder: (_) => ChangeNotifierProvider.value(
                            value: pengumuman,
                            child: DetailPengumumanPage(judul: p),
                          ),
                        ),
                      );
                    },
                  ),
              ],
            ),
    );
  }
}

class DetailPengumumanPage extends StatelessWidget {
  final String judul;

  const DetailPengumumanPage({super.key, required this.judul});

  @override
  Widget build(BuildContext context) {
    final dibaca = context.select((PengumumanProvider p) => p.sudahDibaca(judul));

    return Scaffold(
      appBar: AppBar(title: Text(judul)),
      body: Center(
        child: FilledButton.icon(
          icon: Icon(dibaca ? Icons.done_all : Icons.mark_email_read),
          label: Text(dibaca ? 'Sudah dibaca' : 'Tandai sudah dibaca'),
          onPressed: dibaca
              ? null
              : () => context.read<PengumumanProvider>().tandaiDibaca(judul),
        ),
      ),
    );
  }
}
Data dimuat setelah frame pertama
Halaman detail memakai provider yang sama (.value)

Penjelasan:

  • PengumumanProvider dibuat di dalam MaterialPageRoute, sehingga hidup selama halaman Pengumuman terbuka saja.
  • Data dimuat lewat addPostFrameCallback di initState, dengan cek mounted.
  • Halaman detail menerima provider yang sama lewat ChangeNotifierProvider.value. Status "sudah dibaca" langsung terlihat di daftar saat kembali.
  • _aman() mencegah error jika halaman ditutup sebelum muat() selesai.
