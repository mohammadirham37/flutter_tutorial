Aplikasi nyata biasanya punya lebih dari satu provider: sesi login, tema, keranjang, data mata kuliah, dan seterusnya. MultiProvider mendaftarkan semuanya sekaligus tanpa nesting yang menjorok ke kanan.

Tanpa vs dengan MultiProvider

// Tanpa MultiProvider (sulit dibaca)        // Dengan MultiProvider
ChangeNotifierProvider(                      MultiProvider(
  create: ... TemaProvider,                    providers: [
  child: ChangeNotifierProvider(                 ChangeNotifierProvider(create: ... Tema),
    create: ... SesiProvider,                    ChangeNotifierProvider(create: ... Sesi),
    child: ChangeNotifierProvider(               ChangeNotifierProvider(create: ... Favorit),
      create: ... FavoritProvider,             ],
      child: MyApp(),                          child: MyApp(),
    ),                                       )
  ),
)

Contoh lengkap: tema, sesi, dan favorit

📄 lib/main.dart

// lib/main.dart — salin SELURUH kode ini untuk menggantikan isi lib/main.dart
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

// Provider 1: pengaturan tampilan
class TemaProvider extends ChangeNotifier {
  bool _gelap = false;
  bool get gelap => _gelap;

  void toggle() {
    _gelap = !_gelap;
    notifyListeners();
  }
}

// Provider 2: data pengguna yang sedang login
class SesiProvider extends ChangeNotifier {
  String? _nama;
  String? get nama => _nama;
  bool get sudahLogin => _nama != null;

  void login(String nama) {
    _nama = nama;
    notifyListeners();
  }

  void logout() {
    _nama = null;
    notifyListeners();
  }
}

// Provider 3: daftar favorit
class FavoritProvider extends ChangeNotifier {
  final Set<String> _items = {};
  Set<String> get items => Set.unmodifiable(_items);
  bool isFavorit(String m) => _items.contains(m);

  void toggle(String m) {
    if (!_items.remove(m)) _items.add(m);
    notifyListeners();
  }
}

void main() {
  runApp(
    // MultiProvider = beberapa provider tanpa nesting bertingkat
    MultiProvider(
      providers: [
        ChangeNotifierProvider(create: (_) => TemaProvider()),
        ChangeNotifierProvider(create: (_) => SesiProvider()),
        ChangeNotifierProvider(create: (_) => FavoritProvider()),
      ],
      child: const MyApp(),
    ),
  );
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    // MaterialApp hanya rebuild saat nilai "gelap" berubah
    final gelap = context.select((TemaProvider t) => t.gelap);
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      darkTheme: ThemeData(colorSchemeSeed: Colors.indigo, brightness: Brightness.dark),
      themeMode: gelap ? ThemeMode.dark : ThemeMode.light,
      home: const BerandaPage(),
    );
  }
}

class BerandaPage extends StatelessWidget {
  const BerandaPage({super.key});

  static const matkul = ['Pemrograman Mobile', 'Basis Data', 'Kecerdasan Buatan'];

  @override
  Widget build(BuildContext context) {
    final sesi = context.watch<SesiProvider>();
    final favorit = context.watch<FavoritProvider>();

    return Scaffold(
      appBar: AppBar(
        title: Text(sesi.sudahLogin ? 'Halo, ${sesi.nama}' : 'Tamu'),
        actions: [
          IconButton(
            tooltip: 'Ganti tema',
            icon: const Icon(Icons.dark_mode),
            onPressed: () => context.read<TemaProvider>().toggle(),
          ),
          IconButton(
            tooltip: sesi.sudahLogin ? 'Logout' : 'Login',
            icon: Icon(sesi.sudahLogin ? Icons.logout : Icons.login),
            onPressed: () {
              final s = context.read<SesiProvider>();
              sesi.sudahLogin ? s.logout() : s.login('Andi');
            },
          ),
        ],
      ),
      body: ListView(
        children: [
          ListTile(title: Text('Favorit: ${favorit.items.length} mata kuliah')),
          const Divider(),
          for (final m in matkul)
            ListTile(
              title: Text(m),
              trailing: IconButton(
                tooltip: 'Favorit $m',
                icon: Icon(
                  favorit.isFavorit(m) ? Icons.favorite : Icons.favorite_border,
                  color: favorit.isFavorit(m) ? Colors.red : null,
                ),
                onPressed: () => context.read<FavoritProvider>().toggle(m),
              ),
            ),
        ],
      ),
    );
  }
}
Tiga provider: tema, sesi, favorit
Setelah login, menambah favorit, dan ganti tema

Penjelasan:

  • Tiga provider dengan tanggung jawab berbeda. Masing-masing kecil dan mudah dipahami (prinsip single responsibility).
  • MyApp memakai context.select((TemaProvider t) => t.gelap). MaterialApp hanya rebuild ketika tema berganti, bukan ketika favorit bertambah.
  • BerandaPage me-watch dua provider sekaligus. Halaman rebuild jika salah satunya berubah.
  • Urutan di providers: [...] penting hanya jika satu provider bergantung pada provider lain (lihat ProxyProvider di section 8).

⚠️ Kesalahan umum

  • Membuat satu "GodProvider" raksasa berisi semua state. Setiap perubahan kecil membuat semua pendengar rebuild, dan file-nya sulit dirawat.
  • Mendaftarkan dua provider dengan tipe yang sama. Widget hanya bisa menemukan yang paling dekat; yang lain tertutup.
