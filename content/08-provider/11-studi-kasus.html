Langkah 1: model

📄 lib/models/todo.dart

// lib/models/todo.dart
class Todo {
  final String id;
  final String judul;
  final String matkul;
  final bool selesai;

  const Todo({
    required this.id,
    required this.judul,
    required this.matkul,
    this.selesai = false,
  });

  // Membuat salinan dengan sebagian field diubah (objek tetap immutable)
  Todo copyWith({String? judul, String? matkul, bool? selesai}) {
    return Todo(
      id: id,
      judul: judul ?? this.judul,
      matkul: matkul ?? this.matkul,
      selesai: selesai ?? this.selesai,
    );
  }
}

enum FilterTodo { semua, aktif, selesai }

copyWith membuat salinan objek dengan sebagian field diganti. Objek Todo tidak pernah diubah langsung; yang diganti adalah elemen di list.

Langkah 2: provider

📄 lib/providers/todo_provider.dart

// lib/providers/todo_provider.dart
import 'package:flutter/foundation.dart';

import '../models/todo.dart';

class TodoProvider extends ChangeNotifier {
  final List<Todo> _todos = [
    const Todo(id: '1', judul: 'Laporan praktikum Flutter', matkul: 'Pemrograman Mobile'),
    const Todo(id: '2', judul: 'Normalisasi tabel KRS', matkul: 'Basis Data', selesai: true),
  ];
  FilterTodo _filter = FilterTodo.semua;

  // ---------- GETTER (data turunan dihitung, tidak disimpan) ----------
  FilterTodo get filter => _filter;
  int get jumlahTotal => _todos.length;
  int get jumlahSelesai => _todos.where((t) => t.selesai).length;
  int get jumlahAktif => jumlahTotal - jumlahSelesai;
  double get progres => jumlahTotal == 0 ? 0 : jumlahSelesai / jumlahTotal;

  List<Todo> get todosTampil {
    switch (_filter) {
      case FilterTodo.semua:
        return List.unmodifiable(_todos);
      case FilterTodo.aktif:
        return _todos.where((t) => !t.selesai).toList();
      case FilterTodo.selesai:
        return _todos.where((t) => t.selesai).toList();
    }
  }

  Todo? cariById(String id) {
    for (final t in _todos) {
      if (t.id == id) return t;
    }
    return null;
  }

  // ---------- AKSI (mengubah state + notifyListeners) ----------
  void tambah(String judul, String matkul) {
    final id = DateTime.now().microsecondsSinceEpoch.toString();
    _todos.insert(0, Todo(id: id, judul: judul, matkul: matkul));
    notifyListeners();
  }

  void toggle(String id) {
    final i = _todos.indexWhere((t) => t.id == id);
    if (i == -1) return;
    _todos[i] = _todos[i].copyWith(selesai: !_todos[i].selesai);
    notifyListeners();
  }

  void hapus(String id) {
    _todos.removeWhere((t) => t.id == id);
    notifyListeners();
  }

  void hapusYangSelesai() {
    _todos.removeWhere((t) => t.selesai);
    notifyListeners();
  }

  void ubahFilter(FilterTodo f) {
    if (f == _filter) return;
    _filter = f;
    notifyListeners();
  }
}
  • Data turunan (jumlahAktif, progres, todosTampil) dihitung dari _todos, tidak disimpan terpisah, sehingga tidak mungkin "tidak sinkron".
  • Setiap aksi diakhiri notifyListeners().

import 'package:provider/provider.dart';

Langkah 4: halaman daftar

📄 lib/screens/todo_list_screen.dart

// lib/screens/todo_list_screen.dart
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../models/todo.dart';
import '../providers/todo_provider.dart';
import 'statistik_screen.dart';
import 'tambah_todo_screen.dart';

class TodoListScreen extends StatelessWidget {
  const TodoListScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Tugas Kuliah'),
        actions: [
          // Hanya badge ini yang rebuild saat jumlah aktif berubah
          Selector<TodoProvider, int>(
            selector: (_, p) => p.jumlahAktif,
            builder: (context, aktif, child) => IconButton(
              tooltip: 'Statistik',
              icon: Badge(label: Text('$aktif'), child: const Icon(Icons.insights)),
              onPressed: () => Navigator.push(
                context,
                MaterialPageRoute(builder: (_) => const StatistikScreen()),
              ),
            ),
          ),
        ],
      ),
      body: const Column(
        children: [
          _FilterBar(),
          Expanded(child: _DaftarTodo()),
        ],
      ),
      floatingActionButton: FloatingActionButton.extended(
        icon: const Icon(Icons.add),
        label: const Text('Tugas Baru'),
        onPressed: () => Navigator.push(
          context,
          MaterialPageRoute(builder: (_) => const TambahTodoScreen()),
        ),
      ),
    );
  }
}

class _FilterBar extends StatelessWidget {
  const _FilterBar();

  @override
  Widget build(BuildContext context) {
    final filter = context.select((TodoProvider p) => p.filter);

    return Padding(
      padding: const EdgeInsets.all(12),
      child: SegmentedButton<FilterTodo>(
        segments: const [
          ButtonSegment(value: FilterTodo.semua, label: Text('Semua')),
          ButtonSegment(value: FilterTodo.aktif, label: Text('Aktif')),
          ButtonSegment(value: FilterTodo.selesai, label: Text('Selesai')),
        ],
        selected: {filter},
        onSelectionChanged: (pilihan) =>
            context.read<TodoProvider>().ubahFilter(pilihan.first),
      ),
    );
  }
}

class _DaftarTodo extends StatelessWidget {
  const _DaftarTodo();

  @override
  Widget build(BuildContext context) {
    final todos = context.watch<TodoProvider>().todosTampil;

    if (todos.isEmpty) {
      return const Center(child: Text('Tidak ada tugas di filter ini'));
    }

    return ListView.builder(
      itemCount: todos.length,
      itemBuilder: (context, i) {
        final t = todos[i];
        return Dismissible(
          key: ValueKey(t.id),
          direction: DismissDirection.endToStart,
          background: Container(
            color: Colors.red,
            alignment: Alignment.centerRight,
            padding: const EdgeInsets.only(right: 20),
            child: const Icon(Icons.delete, color: Colors.white),
          ),
          onDismissed: (_) {
            context.read<TodoProvider>().hapus(t.id);
            ScaffoldMessenger.of(context).showSnackBar(
              SnackBar(content: Text('"${t.judul}" dihapus')),
            );
          },
          child: CheckboxListTile(
            value: t.selesai,
            onChanged: (_) => context.read<TodoProvider>().toggle(t.id),
            title: Text(
              t.judul,
              style: TextStyle(
                decoration: t.selesai ? TextDecoration.lineThrough : null,
              ),
            ),
            subtitle: Text(t.matkul),
          ),
        );
      },
    );
  }
}

Langkah 5: halaman tambah

📄 lib/screens/tambah_todo_screen.dart

// lib/screens/tambah_todo_screen.dart
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../providers/todo_provider.dart';

class TambahTodoScreen extends StatefulWidget {
  const TambahTodoScreen({super.key});

  @override
  State<TambahTodoScreen> createState() => _TambahTodoScreenState();
}

class _TambahTodoScreenState extends State<TambahTodoScreen> {
  final _formKey = GlobalKey<FormState>();
  final _judulController = TextEditingController();
  String _matkul = 'Pemrograman Mobile';

  static const daftarMatkul = ['Pemrograman Mobile', 'Basis Data', 'Jaringan Komputer'];

  @override
  void dispose() {
    _judulController.dispose();
    super.dispose();
  }

  void _simpan() {
    if (!_formKey.currentState!.validate()) return;
    // Tidak perlu mengembalikan data lewat pop: provider yang menyimpan
    context.read<TodoProvider>().tambah(_judulController.text.trim(), _matkul);
    Navigator.pop(context);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Tugas Baru')),
      body: Form(
        key: _formKey,
        child: ListView(
          padding: const EdgeInsets.all(16),
          children: [
            TextFormField(
              controller: _judulController,
              decoration: const InputDecoration(
                labelText: 'Judul tugas',
                border: OutlineInputBorder(),
              ),
              validator: (v) =>
                  (v == null || v.trim().isEmpty) ? 'Judul wajib diisi' : null,
            ),
            const SizedBox(height: 16),
            DropdownButtonFormField<String>(
              initialValue: _matkul,
              decoration: const InputDecoration(
                labelText: 'Mata kuliah',
                border: OutlineInputBorder(),
              ),
              items: [
                for (final m in daftarMatkul) DropdownMenuItem(value: m, child: Text(m)),
              ],
              onChanged: (v) => setState(() => _matkul = v ?? _matkul),
            ),
            const SizedBox(height: 24),
            FilledButton(onPressed: _simpan, child: const Text('Simpan')),
          ],
        ),
      ),
    );
  }
}

Langkah 6: halaman statistik

📄 lib/screens/statistik_screen.dart

// lib/screens/statistik_screen.dart
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../providers/todo_provider.dart';

class StatistikScreen extends StatelessWidget {
  const StatistikScreen({super.key});

  @override
  Widget build(BuildContext context) {
    final p = context.watch<TodoProvider>();

    return Scaffold(
      appBar: AppBar(title: const Text('Statistik')),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Text('${(p.progres * 100).round()}% selesai',
                style: Theme.of(context).textTheme.headlineMedium),
            const SizedBox(height: 8),
            LinearProgressIndicator(value: p.progres, minHeight: 10),
            const SizedBox(height: 16),
            Card(
              child: Column(
                children: [
                  ListTile(title: const Text('Total tugas'), trailing: Text('${p.jumlahTotal}')),
                  ListTile(title: const Text('Aktif'), trailing: Text('${p.jumlahAktif}')),
                  ListTile(title: const Text('Selesai'), trailing: Text('${p.jumlahSelesai}')),
                ],
              ),
            ),
            const SizedBox(height: 16),
            OutlinedButton.icon(
              icon: const Icon(Icons.cleaning_services),
              label: const Text('Hapus semua yang selesai'),
              onPressed: p.jumlahSelesai == 0
                  ? null
                  : () => context.read<TodoProvider>().hapusYangSelesai(),
            ),
          ],
        ),
      ),
    );
  }
}
