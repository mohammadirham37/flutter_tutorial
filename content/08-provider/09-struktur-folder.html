Semakin besar aplikasi, semakin penting memisahkan kode berdasarkan tanggung jawabnya. Struktur berikut sederhana, cukup untuk proyek kuliah, dan dipakai lagi di proyek CRUD + Auth (Modul 13–14).

lib/
├── main.dart              # runApp + MultiProvider + MaterialApp
├── models/                # class data murni (User, Mahasiswa, Todo)
├── services/              # akses data: HTTP, database, Supabase
├── providers/             # ChangeNotifier: state + logika bisnis
├── screens/               # halaman penuh (Scaffold)
└── widgets/               # komponen kecil yang dipakai ulang
LapisanBoleh mengimporTidak boleh
modelsDart murniFlutter widget, provider, service
servicesmodels, package http/databasewidget, BuildContext, provider
providersmodels, services, foundation.dartwidget, BuildContext, Navigator
screens / widgetsmodels, providers, widget lainMemanggil service langsung

Arah ketergantungan: UI → Provider → Service → Model. Provider tidak boleh tahu tentang UI; itulah sebabnya provider tidak menampilkan SnackBar atau melakukan navigasi sendiri.

Contoh lengkap: Auth gate dengan Provider

Auth gate adalah widget yang memilih menampilkan LoginScreen atau HomeScreen berdasarkan status sesi di provider. Login dan logout tidak memerlukan Navigator.push sama sekali.

import '../models/user.dart';

class AuthService {
  Future<User> login(String email, String password) async {
    await Future.delayed(const Duration(seconds: 1));
    if (password != '123456') {
      throw Exception('Email atau password salah');
    }
    return User(email: email, nama: email.split('@').first);
  }

  Future<void> logout() async {
    await Future.delayed(const Duration(milliseconds: 300));
  }
}

📄 lib/providers/auth_provider.dart

// lib/providers/auth_provider.dart
import 'package:flutter/foundation.dart';

import '../models/user.dart';
import '../services/auth_service.dart';

class AuthProvider extends ChangeNotifier {
  AuthProvider(this._service);

  final AuthService _service;

  User? _user;
  bool _isLoading = false;
  String? _error;

  User? get user => _user;
  bool get isLoggedIn => _user != null;
  bool get isLoading => _isLoading;
  String? get error => _error;

  Future<bool> login(String email, String password) async {
    _isLoading = true;
    _error = null;
    notifyListeners();
    try {
      _user = await _service.login(email, password);
      return true;
    } catch (e) {
      _error = e.toString().replaceFirst('Exception: ', '');
      return false;
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }

  Future<void> logout() async {
    await _service.logout();
    _user = null;
    notifyListeners();
  }
}

// lib/widgets/auth_gate.dart
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../providers/auth_provider.dart';
import '../screens/home_screen.dart';
import '../screens/login_screen.dart';

// Auth gate: memilih halaman berdasarkan status sesi.
// Tidak ada Navigator.push sama sekali untuk login/logout!
class AuthGate extends StatelessWidget {
  const AuthGate({super.key});

  @override
  Widget build(BuildContext context) {
    final sudahLogin = context.select((AuthProvider a) => a.isLoggedIn);
    return sudahLogin ? const HomeScreen() : const LoginScreen();
  }
}

📄 lib/screens/login_screen.dart

// lib/screens/login_screen.dart
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../providers/auth_provider.dart';

class LoginScreen extends StatefulWidget {
  const LoginScreen({super.key});

  @override
  State<LoginScreen> createState() => _LoginScreenState();
}

class _LoginScreenState extends State<LoginScreen> {
  final _emailController = TextEditingController(text: 'andi@mhs.unsoed.ac.id');
  final _passwordController = TextEditingController();

  @override
  void dispose() {
    _emailController.dispose();
    _passwordController.dispose();
    super.dispose();
  }

  Future<void> _login() async {
    FocusScope.of(context).unfocus(); // tutup keyboard
    final berhasil = await context.read<AuthProvider>().login(
          _emailController.text.trim(),
          _passwordController.text,
        );
    // Jika berhasil, AuthGate otomatis menampilkan HomeScreen.
    if (!berhasil && mounted) {
      final pesan = context.read<AuthProvider>().error ?? 'Login gagal';
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(pesan)));
    }
  }

  @override
  Widget build(BuildContext context) {
    final isLoading = context.select((AuthProvider a) => a.isLoading);

    return Scaffold(
      appBar: AppBar(title: const Text('Login')),
      body: Padding(
        padding: const EdgeInsets.all(24),
        child: Column(
          children: [
            TextField(
              controller: _emailController,
              decoration: const InputDecoration(labelText: 'Email'),
            ),
            TextField(
              controller: _passwordController,
              obscureText: true,
              decoration: const InputDecoration(
                labelText: 'Password',
                helperText: 'Petunjuk: 123456',
              ),
            ),
            const SizedBox(height: 24),
            SizedBox(
              width: double.infinity,
              child: FilledButton(
                onPressed: isLoading ? null : _login,
                child: isLoading
                    ? const SizedBox(
                        width: 20,
                        height: 20,
                        child: CircularProgressIndicator(strokeWidth: 2),
                      )
                    : const Text('Masuk'),
              ),
            ),
          ],
        ),
      ),
    );
  }
}

// lib/screens/home_screen.dart
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../providers/auth_provider.dart';

class HomeScreen extends StatelessWidget {
  const HomeScreen({super.key});

  @override
  Widget build(BuildContext context) {
    final user = context.watch<AuthProvider>().user;

    return Scaffold(
      appBar: AppBar(
        title: const Text('Beranda'),
        actions: [
          IconButton(
            tooltip: 'Logout',
            icon: const Icon(Icons.logout),
            // Setelah logout, AuthGate otomatis kembali ke LoginScreen
            onPressed: () => context.read<AuthProvider>().logout(),
          ),
        ],
      ),
      body: Center(
        child: Text(
          'Selamat datang, ${user?.nama}!',
          style: Theme.of(context).textTheme.headlineSmall,
        ),
      ),
    );
  }
}

📄 lib/main.dart

// lib/main.dart
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import 'providers/auth_provider.dart';
import 'services/auth_service.dart';
import 'widgets/auth_gate.dart';

void main() {
  runApp(
    MultiProvider(
      providers: [
        ChangeNotifierProvider(create: (_) => AuthProvider(AuthService())),
        // Provider lain (MatkulProvider, TodoProvider, ...) ditambahkan di sini
      ],
      child: const MyApp(),
    ),
  );
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const AuthGate(),
    );
  }
}

Penjelasan:

  • AuthGate memakai select pada isLoggedIn. Ia hanya rebuild saat status login berubah, bukan saat isLoading berubah.
  • AuthProvider.login mengembalikan bool. Layar login yang memutuskan cara menampilkan error (SnackBar), sedangkan provider hanya menyimpan pesannya.
  • Tombol Masuk dinonaktifkan dan diganti spinner selama isLoading, mencegah login ganda.
  • Logout cukup memanggil logout(). Begitu _user menjadi null, AuthGate menampilkan LoginScreen lagi.

⚠️ Kesalahan umum

  • Meng-import material.dart di provider untuk memanggil ScaffoldMessenger atau Navigator. Kembalikan hasil (bool/pesan) dan biarkan UI yang bereaksi.
  • Mencampur AuthGate dengan pushReplacement manual setelah login. Pilih satu cara; jika memakai AuthGate, biarkan ia yang menentukan halaman.
  • Menaruh semua class di main.dart sampai ratusan baris. Pecah sesuai struktur di atas sejak awal.
