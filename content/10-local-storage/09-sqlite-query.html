Kekuatan utama SQLite dibanding SharedPreferences adalah kemampuan mencari, memfilter, mengurutkan, dan menghitung data. Di section ini kita mengenal parameter lengkap db.query(), rawQuery() untuk SQL bebas, dan cara aman memasukkan input pengguna ke query.

Parameter db.query()

import 'package:sqflite/sqflite.dart';

Future<List<Map<String, Object?>>> contohQuery(Database db) {
  return db.query(
    'mahasiswa',                       // FROM mahasiswa
    columns: ['nim', 'nama', 'ipk'],   // SELECT nim, nama, ipk  (null = semua kolom)
    where: 'prodi = ? AND ipk >= ?',   // WHERE prodi = ? AND ipk >= ?
    whereArgs: ['Informatika', 3.0],   // nilai untuk tanda ? secara berurutan
    orderBy: 'ipk DESC, nama ASC',     // ORDER BY
    limit: 10,                         // LIMIT 10
    offset: 0,                         // OFFSET 0 (untuk halaman berikutnya: 10, 20, ...)
  );
}

⚠️ SQL Injection: Mengapa Wajib Memakai ? dan whereArgs

Bayangkan fitur pencarian yang menyambung input pengguna langsung ke SQL:

import 'package:sqflite/sqflite.dart';

// ❌ BERBAHAYA: input pengguna disambung langsung ke string SQL
Future<List<Map<String, Object?>>> cariTidakAman(Database db, String input) {
  return db.rawQuery("SELECT * FROM mahasiswa WHERE nama = '$input'");
}

// ✅ AMAN: nilai dikirim terpisah, tidak pernah dianggap perintah SQL
Future<List<Map<String, Object?>>> cariAman(Database db, String input) {
  return db.rawQuery('SELECT * FROM mahasiswa WHERE nama = ?', [input]);
}

Jika pengguna mengetik x' OR '1'='1, versi tidak aman menjadi WHERE nama = 'x' OR '1'='1' yang selalu benar, sehingga semua data bocor. Input seperti '; DROP TABLE mahasiswa; -- bahkan bisa merusak data. Dengan ?, input tersebut diperlakukan sebagai teks biasa. Tanda kutip di nama seperti Ma'ruf pun tidak membuat error.

Pencarian dengan LIKE

import 'package:sqflite/sqflite.dart';

Future<List<Map<String, Object?>>> cariNama(Database db, String kata) {
  return db.query(
    'mahasiswa',
    where: 'nama LIKE ? OR nim LIKE ?',
    whereArgs: ['%$kata%', '%$kata%'], // % = karakter apa saja, ditaruh di whereArgs
  );
}

Tanda % ditempel pada nilai di whereArgs, bukan di string where.

rawQuery untuk Agregasi

Untuk COUNT, AVG, GROUP BY, dan JOIN, SQL mentah lebih jelas:

import 'package:sqflite/sqflite.dart';

Future<void> statistik(Database db) async {
  // Satu angka: gunakan Sqflite.firstIntValue
  final total = Sqflite.firstIntValue(
    await db.rawQuery('SELECT COUNT(*) FROM mahasiswa'),
  ) ?? 0;

  // Rata-rata IPK per prodi
  final rows = await db.rawQuery('''
    SELECT prodi, COUNT(*) AS jumlah, ROUND(AVG(ipk), 2) AS rata
    FROM mahasiswa
    GROUP BY prodi
    ORDER BY rata DESC
  ''');
  print('Total: $total');
  for (final r in rows) {
    print('${r['prodi']}: ${r['jumlah']} mhs, rata-rata ${r['rata']}');
  }
}

Contoh Lengkap: Pencarian & Filter Mahasiswa

📄 lib/data/mahasiswa_query_db.dart

import 'package:path/path.dart';
import 'package:sqflite/sqflite.dart';

class MahasiswaQueryDb {
  static Database? _db;

  static Future<Database> get database async => _db ??= await openDatabase(
        join(await getDatabasesPath(), 'query_demo.db'),
        version: 1,
        onCreate: (db, version) async {
          await db.execute('''
            CREATE TABLE mahasiswa (
              id INTEGER PRIMARY KEY AUTOINCREMENT,
              nim TEXT NOT NULL UNIQUE,
              nama TEXT NOT NULL,
              prodi TEXT NOT NULL,
              ipk REAL NOT NULL
            )
          ''');
          // Data contoh
          const contoh = [
            ['H1D022001', 'Budi Santoso', 'Informatika', 3.45],
            ['H1D022002', 'Siti Aminah', 'Informatika', 3.82],
            ['H1D022003', "Ma'ruf Hidayat", 'Informatika', 2.95],
            ['H1A022004', 'Dewi Lestari', 'Teknik Elektro', 3.60],
            ['H1A022005', 'Rizky Pratama', 'Teknik Elektro', 3.10],
            ['H1C022006', 'Andi Wijaya', 'Teknik Sipil', 3.25],
          ];
          for (final m in contoh) {
            await db.insert('mahasiswa', {'nim': m[0], 'nama': m[1], 'prodi': m[2], 'ipk': m[3]});
          }
        },
      );

  /// Gabungan pencarian + filter prodi + filter IPK minimum
  static Future<List<Map<String, Object?>>> cari({
    String kata = '',
    String? prodi,
    double ipkMin = 0,
  }) async {
    final db = await database;
    final kondisi = <String>['ipk >= ?'];
    final args = <Object?>[ipkMin];
    if (kata.isNotEmpty) {
      kondisi.add('(nama LIKE ? OR nim LIKE ?)');
      args.addAll(['%$kata%', '%$kata%']);
    }
    if (prodi != null) {
      kondisi.add('prodi = ?');
      args.add(prodi);
    }
    return db.query(
      'mahasiswa',
      where: kondisi.join(' AND '),
      whereArgs: args,
      orderBy: 'ipk DESC',
    );
  }

  static Future<List<String>> daftarProdi() async {
    final db = await database;
    final rows = await db.rawQuery('SELECT DISTINCT prodi FROM mahasiswa ORDER BY prodi');
    return rows.map((r) => r['prodi'] as String).toList();
  }

  static Future<double> rataIpk() async {
    final db = await database;
    final rows = await db.rawQuery('SELECT AVG(ipk) AS rata FROM mahasiswa');
    return (rows.first['rata'] as num? ?? 0).toDouble();
  }
}

Penjelasan: fungsi cari menyusun klausa WHERE secara dinamis. String where hanya berisi nama kolom dan ? yang kita tulis sendiri, sedangkan semua nilai dari pengguna masuk ke args. Itulah cara aman membuat filter dinamis.

📄 lib/main.dart

// Pastikan sudah: flutter pub add sqflite path, dan membuat lib/data/mahasiswa_query_db.dart
import 'package:flutter/material.dart';

import 'data/mahasiswa_query_db.dart';

void main() => runApp(const MaterialApp(home: CariMahasiswaScreen()));

class CariMahasiswaScreen extends StatefulWidget {
  const CariMahasiswaScreen({super.key});

  @override
  State<CariMahasiswaScreen> createState() => _CariMahasiswaScreenState();
}

class _CariMahasiswaScreenState extends State<CariMahasiswaScreen> {
  String _kata = '';
  String? _prodi;
  bool _cumlaude = false;
  List<String> _semuaProdi = [];
  List<Map<String, Object?>> _hasil = [];
  double _rata = 0;

  @override
  void initState() {
    super.initState();
    _init();
  }

  Future<void> _init() async {
    final prodi = await MahasiswaQueryDb.daftarProdi();
    final rata = await MahasiswaQueryDb.rataIpk();
    if (!mounted) return;
    setState(() {
      _semuaProdi = prodi;
      _rata = rata;
    });
    await _cari();
  }

  Future<void> _cari() async {
    final hasil = await MahasiswaQueryDb.cari(
      kata: _kata,
      prodi: _prodi,
      ipkMin: _cumlaude ? 3.5 : 0,
    );
    if (mounted) setState(() => _hasil = hasil);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Rata-rata IPK: ${_rata.toStringAsFixed(2)}')),
      body: Column(
        children: [
          Padding(
            padding: const EdgeInsets.all(12),
            child: TextField(
              decoration: const InputDecoration(
                prefixIcon: Icon(Icons.search),
                hintText: 'Cari nama atau NIM',
                border: OutlineInputBorder(),
              ),
              onChanged: (v) {
                _kata = v.trim();
                _cari();
              },
            ),
          ),
          SingleChildScrollView(
            scrollDirection: Axis.horizontal,
            padding: const EdgeInsets.symmetric(horizontal: 12),
            child: Row(
              children: [
                FilterChip(
                  label: const Text('IPK ≥ 3.5'),
                  selected: _cumlaude,
                  onSelected: (v) {
                    setState(() => _cumlaude = v);
                    _cari();
                  },
                ),
                for (final p in _semuaProdi)
                  Padding(
                    padding: const EdgeInsets.only(left: 8),
                    child: ChoiceChip(
                      label: Text(p),
                      selected: _prodi == p,
                      onSelected: (v) {
                        setState(() => _prodi = v ? p : null);
                        _cari();
                      },
                    ),
                  ),
              ],
            ),
          ),
          Padding(
            padding: const EdgeInsets.all(8),
            child: Text('${_hasil.length} mahasiswa ditemukan'),
          ),
          Expanded(
            child: ListView.builder(
              itemCount: _hasil.length,
              itemBuilder: (context, i) {
                final m = _hasil[i];
                return ListTile(
                  leading: CircleAvatar(child: Text((m['ipk'] as num).toStringAsFixed(1))),
                  title: Text(m['nama'] as String),
                  subtitle: Text('${m['nim']} • ${m['prodi']}'),
                );
              },
            ),
          ),
        ],
      ),
    );
  }
}

⚠️ Kesalahan Umum

  • Menyambung input ke SQL dengan '$input' (SQL injection).
  • Jumlah ? tidak sama dengan jumlah elemen whereArgs: error bind or column index out of range.
  • Menulis where: "nama LIKE '%?%'". Tanda ? di dalam kutip tidak dianggap parameter; letakkan % di nilai.
  • Mengambil semua data lalu memfilter di Dart dengan where() untuk ribuan baris. Biarkan SQLite yang memfilter, jauh lebih cepat.
