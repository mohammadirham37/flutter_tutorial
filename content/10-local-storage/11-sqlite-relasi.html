Database relasional bisa menghubungkan tabel. Contoh: satu mata kuliah punya banyak tugas (relasi one-to-many). Daripada menulis nama mata kuliah berulang di setiap tugas, tabel tugas cukup menyimpan matkul_id yang menunjuk ke tabel mata kuliah.

Rancangan Tabel

mata_kuliah                     tugas
┌────┬──────────────────┬─────┐  ┌────┬──────────────┬───────────┐
│ id │ nama             │ sks │  │ id │ judul        │ matkul_id │──┐
├────┼──────────────────┼─────┤  ├────┼──────────────┼───────────┤  │
│ 1  │ Pemrograman Mob. │ 3   │<─┤ 1  │ Laporan 1    │ 1         │  │ FOREIGN KEY
│ 2  │ Basis Data       │ 3   │<─┤ 2  │ Kuis ERD     │ 2         │  │ -> mata_kuliah.id
└────┴──────────────────┴─────┘  │ 3  │ Laporan 2    │ 1         │──┘
                                 └────┴──────────────┴───────────┘

Konsep Penting

  • FOREIGN KEY (matkul_id) REFERENCES mata_kuliah(id) memastikan tugas hanya bisa menunjuk mata kuliah yang ada.
  • ON DELETE CASCADE: jika mata kuliah dihapus, semua tugasnya ikut terhapus otomatis.
  • SQLite mematikan foreign key secara default. Aktifkan dengan PRAGMA foreign_keys = ON di onConfigure (dijalankan setiap database dibuka).
  • JOIN menggabungkan baris dari dua tabel dalam satu query.

Contoh Kecil: Skema dan JOIN

CREATE TABLE mata_kuliah (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nama TEXT NOT NULL,
  sks INTEGER NOT NULL
);

CREATE TABLE tugas (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  judul TEXT NOT NULL,
  matkul_id INTEGER NOT NULL,
  FOREIGN KEY (matkul_id) REFERENCES mata_kuliah(id) ON DELETE CASCADE
);

-- Tugas beserta nama mata kuliahnya
SELECT t.id, t.judul, m.nama AS matkul_nama
FROM tugas t
JOIN mata_kuliah m ON m.id = t.matkul_id
ORDER BY m.nama;

Model untuk Hasil JOIN

Hasil JOIN berisi kolom dari dua tabel, jadi kita buat model khusus untuk tampilan:

📄 lib/models/relasi_models.dart

class MataKuliah {
  final int? id;
  final String nama;
  final int sks;
  final int jumlahTugas; // hasil COUNT, bukan kolom tabel

  const MataKuliah({this.id, required this.nama, required this.sks, this.jumlahTugas = 0});

  Map<String, Object?> toMap() => {if (id != null) 'id': id, 'nama': nama, 'sks': sks};

  factory MataKuliah.fromMap(Map<String, Object?> m) => MataKuliah(
        id: m['id'] as int?,
        nama: m['nama'] as String,
        sks: m['sks'] as int,
        jumlahTugas: m['jumlah_tugas'] as int? ?? 0,
      );
}

class TugasDenganMatkul {
  final int id;
  final String judul;
  final String matkulNama;

  const TugasDenganMatkul({required this.id, required this.judul, required this.matkulNama});

  factory TugasDenganMatkul.fromMap(Map<String, Object?> m) => TugasDenganMatkul(
        id: m['id'] as int,
        judul: m['judul'] as String,
        matkulNama: m['matkul_nama'] as String, // alias dari "AS matkul_nama"
      );
}

Database Helper dengan Relasi

📄 lib/data/relasi_database.dart

import 'package:path/path.dart';
import 'package:sqflite/sqflite.dart';

import '../models/relasi_models.dart';

class RelasiDatabase {
  static Database? _db;

  static Future<Database> get database async => _db ??= await openDatabase(
        join(await getDatabasesPath(), 'relasi_demo.db'),
        version: 1,
        onConfigure: (db) => db.execute('PRAGMA foreign_keys = ON'),
        onCreate: (db, version) async {
          await db.execute('''
            CREATE TABLE mata_kuliah (
              id INTEGER PRIMARY KEY AUTOINCREMENT,
              nama TEXT NOT NULL,
              sks INTEGER NOT NULL
            )
          ''');
          await db.execute('''
            CREATE TABLE tugas (
              id INTEGER PRIMARY KEY AUTOINCREMENT,
              judul TEXT NOT NULL,
              matkul_id INTEGER NOT NULL,
              FOREIGN KEY (matkul_id) REFERENCES mata_kuliah(id) ON DELETE CASCADE
            )
          ''');
          // Data awal
          final mobile = await db.insert('mata_kuliah', {'nama': 'Pemrograman Mobile', 'sks': 3});
          final basdat = await db.insert('mata_kuliah', {'nama': 'Basis Data', 'sks': 3});
          await db.insert('tugas', {'judul': 'Laporan Praktikum 10', 'matkul_id': mobile});
          await db.insert('tugas', {'judul': 'Proyek CRUD', 'matkul_id': mobile});
          await db.insert('tugas', {'judul': 'Kuis ERD', 'matkul_id': basdat});
        },
      );

  /// Mata kuliah + jumlah tugasnya (LEFT JOIN agar matkul tanpa tugas tetap muncul)
  static Future<List<MataKuliah>> semuaMatkul() async {
    final db = await database;
    final rows = await db.rawQuery('''
      SELECT m.id, m.nama, m.sks, COUNT(t.id) AS jumlah_tugas
      FROM mata_kuliah m
      LEFT JOIN tugas t ON t.matkul_id = m.id
      GROUP BY m.id
      ORDER BY m.nama
    ''');
    return rows.map(MataKuliah.fromMap).toList();
  }

  /// Semua tugas beserta nama mata kuliahnya (INNER JOIN)
  static Future<List<TugasDenganMatkul>> semuaTugas() async {
    final db = await database;
    final rows = await db.rawQuery('''
      SELECT t.id, t.judul, m.nama AS matkul_nama
      FROM tugas t
      JOIN mata_kuliah m ON m.id = t.matkul_id
      ORDER BY m.nama, t.id
    ''');
    return rows.map(TugasDenganMatkul.fromMap).toList();
  }

  static Future<void> tambahTugas(String judul, int matkulId) async {
    final db = await database;
    await db.insert('tugas', {'judul': judul, 'matkul_id': matkulId});
  }

  /// Menghapus matkul: tugas-tugasnya ikut terhapus berkat ON DELETE CASCADE
  static Future<void> hapusMatkul(int id) async {
    final db = await database;
    await db.delete('mata_kuliah', where: 'id = ?', whereArgs: [id]);
  }
}

Contoh Lengkap: Mata Kuliah dan Tugasnya

📄 lib/main.dart

// Pastikan sudah: flutter pub add sqflite path, serta membuat
// lib/models/relasi_models.dart dan lib/data/relasi_database.dart
import 'package:flutter/material.dart';

import 'data/relasi_database.dart';
import 'models/relasi_models.dart';

void main() => runApp(const MaterialApp(home: RelasiScreen()));

class RelasiScreen extends StatefulWidget {
  const RelasiScreen({super.key});

  @override
  State<RelasiScreen> createState() => _RelasiScreenState();
}

class _RelasiScreenState extends State<RelasiScreen> {
  List<MataKuliah> _matkul = [];
  List<TugasDenganMatkul> _tugas = [];

  @override
  void initState() {
    super.initState();
    _muat();
  }

  Future<void> _muat() async {
    final matkul = await RelasiDatabase.semuaMatkul();
    final tugas = await RelasiDatabase.semuaTugas();
    if (!mounted) return;
    setState(() {
      _matkul = matkul;
      _tugas = tugas;
    });
  }

  @override
  Widget build(BuildContext context) {
    final judulStyle = Theme.of(context).textTheme.titleMedium;
    return Scaffold(
      appBar: AppBar(title: const Text('Relasi Mata Kuliah - Tugas')),
      body: ListView(
        padding: const EdgeInsets.all(12),
        children: [
          Text('Mata Kuliah', style: judulStyle),
          for (final m in _matkul)
            Card(
              child: ListTile(
                title: Text(m.nama),
                subtitle: Text('${m.sks} SKS • ${m.jumlahTugas} tugas'),
                trailing: Row(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    IconButton(
                      tooltip: 'Tambah tugas',
                      icon: const Icon(Icons.add_task),
                      onPressed: () async {
                        await RelasiDatabase.tambahTugas('Tugas baru ${m.nama}', m.id!);
                        await _muat();
                      },
                    ),
                    IconButton(
                      tooltip: 'Hapus matkul',
                      icon: const Icon(Icons.delete_outline),
                      onPressed: () async {
                        await RelasiDatabase.hapusMatkul(m.id!);
                        await _muat();
                      },
                    ),
                  ],
                ),
              ),
            ),
          const SizedBox(height: 16),
          Text('Semua Tugas (hasil JOIN)', style: judulStyle),
          for (final t in _tugas)
            ListTile(
              dense: true,
              leading: const Icon(Icons.assignment_outlined),
              title: Text(t.judul),
              subtitle: Text(t.matkulNama),
            ),
        ],
      ),
    );
  }
}

Penjelasan: tekan ikon hapus pada "Basis Data". Matkul hilang dan tugas "Kuis ERD" juga hilang dari daftar JOIN, padahal kita hanya menjalankan satu DELETE pada tabel mata_kuliah. Itulah efek ON DELETE CASCADE.

⚠️ Kesalahan Umum

  • Lupa PRAGMA foreign_keys = ON: CASCADE tidak berjalan dan tugas "yatim" (menunjuk matkul yang sudah tidak ada) tetap tersimpan.
  • Memakai JOIN biasa saat ingin menampilkan matkul yang belum punya tugas. Gunakan LEFT JOIN.
  • Dua tabel punya kolom bernama sama (id, nama). Gunakan alias (m.nama AS matkul_nama) agar kunci Map tidak bertabrakan.
  • Melakukan query per item di dalam loop (N+1 query). Satu query JOIN jauh lebih efisien.
