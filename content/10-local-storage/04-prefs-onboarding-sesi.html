Dua kebutuhan klasik aplikasi: layar onboarding yang hanya tampil sekali saat pertama dipasang, dan sesi login agar pengguna tidak perlu login setiap membuka aplikasi. Keduanya cukup memakai beberapa nilai bool dan String di SharedPreferences.

Alur Aplikasi

Buka aplikasi
   └─> SplashGate membaca penyimpanan
         ├─ sudahOnboarding == false ──> OnboardingScreen ──(Mulai)──> simpan true ──> Login
         ├─ token == null            ──> LoginScreen ──(Login)──> simpan token ──> Home
         └─ token ada                ──> HomeScreen ──(Logout)──> hapus token ──> Login

Langkah 1: SessionManager

Semua akses SharedPreferences dikumpulkan di satu class. Layar tidak perlu tahu nama kunci.

📄 lib/services/session_manager.dart

// lib/services/session_manager.dart
import 'package:shared_preferences/shared_preferences.dart';

class SessionManager {
  static const String keyOnboarding = 'ONBOARDING_DONE';
  static const String keyToken = 'USER_AUTH_TOKEN';
  static const String keyNama = 'USER_NAME';

  final SharedPreferencesAsync _prefs = SharedPreferencesAsync();

  // ---------- Onboarding ----------
  Future<bool> sudahOnboarding() async => await _prefs.getBool(keyOnboarding) ?? false;

  Future<void> selesaiOnboarding() => _prefs.setBool(keyOnboarding, true);

  // ---------- Sesi login ----------
  Future<void> simpanSesiLogin({required String token, required String nama}) async {
    await _prefs.setString(keyToken, token);
    await _prefs.setString(keyNama, nama);
  }

  Future<String?> ambilToken() => _prefs.getString(keyToken);

  Future<String?> ambilNama() => _prefs.getString(keyNama);

  Future<bool> cekStatusLogin() async => await ambilToken() != null;

  // Hapus HANYA data sesi; flag onboarding dan pengaturan lain tetap ada
  Future<void> logout() => _prefs.clear(allowList: {keyToken, keyNama});
}

Penjelasan: status login tidak disimpan sebagai bool terpisah, melainkan disimpulkan dari ada/tidaknya token. Dengan begitu tidak mungkin terjadi keadaan "login = true tetapi token kosong". Method logout() memakai clear(allowList: ...) agar hanya kunci sesi yang terhapus.

ℹ️ Tentang Keamanan Token

Contoh ini cocok untuk belajar. SharedPreferences tidak terenkripsi, jadi untuk aplikasi produksi simpan token di flutter_secure_storage. Polanya sama: ganti isi SessionManager saja, layar tidak berubah.

Langkah 2: Contoh Lengkap Onboarding + Login

📄 lib/main.dart

// lib/main.dart — salin SELURUH kode ini untuk menggantikan isi lib/main.dart
// Pastikan sudah: flutter pub add shared_preferences, serta membuat lib/services/session_manager.dart
import 'package:flutter/material.dart';

import 'services/session_manager.dart';

final session = SessionManager();

void main() {
  runApp(const MaterialApp(debugShowCheckedModeBanner: false, home: SplashGate()));
}

/// Membaca penyimpanan lalu memutuskan layar pertama.
class SplashGate extends StatefulWidget {
  const SplashGate({super.key});

  @override
  State<SplashGate> createState() => _SplashGateState();
}

class _SplashGateState extends State<SplashGate> {
  @override
  void initState() {
    super.initState();
    _tentukanTujuan();
  }

  Future<void> _tentukanTujuan() async {
    final onboarding = await session.sudahOnboarding();
    final login = await session.cekStatusLogin();
    if (!mounted) return;

    final Widget tujuan = !onboarding
        ? const OnboardingScreen()
        : login
            ? const HomeScreen()
            : const LoginScreen();
    // pushReplacement: SplashGate dibuang agar tombol Back tidak kembali ke sini
    Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => tujuan));
  }

  @override
  Widget build(BuildContext context) {
    return const Scaffold(body: Center(child: CircularProgressIndicator()));
  }
}

class OnboardingScreen extends StatelessWidget {
  const OnboardingScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Padding(
          padding: const EdgeInsets.all(32),
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              const Icon(Icons.school, size: 96, color: Colors.indigo),
              const SizedBox(height: 24),
              Text('Selamat Datang di SIA Mobile',
                  style: Theme.of(context).textTheme.headlineSmall, textAlign: TextAlign.center),
              const SizedBox(height: 12),
              const Text('Layar ini hanya muncul sekali saat aplikasi pertama dibuka.',
                  textAlign: TextAlign.center),
              const SizedBox(height: 32),
              FilledButton(
                onPressed: () async {
                  await session.selesaiOnboarding();
                  if (!context.mounted) return;
                  Navigator.pushReplacement(
                    context,
                    MaterialPageRoute(builder: (_) => const LoginScreen()),
                  );
                },
                child: const Text('Mulai'),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

class LoginScreen extends StatefulWidget {
  const LoginScreen({super.key});

  @override
  State<LoginScreen> createState() => _LoginScreenState();
}

class _LoginScreenState extends State<LoginScreen> {
  final _namaCtrl = TextEditingController();

  @override
  void dispose() {
    _namaCtrl.dispose();
    super.dispose();
  }

  Future<void> _login() async {
    final nama = _namaCtrl.text.trim();
    if (nama.isEmpty) return;
    // Pada aplikasi nyata token didapat dari API login (Modul 09)
    await session.simpanSesiLogin(token: 'token-${DateTime.now().millisecondsSinceEpoch}', nama: nama);
    if (!mounted) return;
    Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => const HomeScreen()));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Login')),
      body: Padding(
        padding: const EdgeInsets.all(24),
        child: Column(
          children: [
            TextField(
              controller: _namaCtrl,
              decoration: const InputDecoration(labelText: 'Nama', border: OutlineInputBorder()),
            ),
            const SizedBox(height: 16),
            SizedBox(
              width: double.infinity,
              child: FilledButton(onPressed: _login, child: const Text('Masuk')),
            ),
          ],
        ),
      ),
    );
  }
}

class HomeScreen extends StatelessWidget {
  const HomeScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Beranda'),
        actions: [
          IconButton(
            tooltip: 'Logout',
            icon: const Icon(Icons.logout),
            onPressed: () async {
              await session.logout();
              if (!context.mounted) return;
              Navigator.pushReplacement(
                context,
                MaterialPageRoute(builder: (_) => const LoginScreen()),
              );
            },
          ),
        ],
      ),
      body: Center(
        child: FutureBuilder<String?>(
          future: session.ambilNama(),
          builder: (context, snapshot) => Text(
            'Halo, ${snapshot.data ?? '...'}!\nTutup lalu buka aplikasi: kamu tetap login.',
            textAlign: TextAlign.center,
            style: Theme.of(context).textTheme.titleMedium,
          ),
        ),
      ),
    );
  }
}
Pembukaan pertama: onboarding
Setelah Mulai: layar login
Setelah login: beranda dengan nama tersimpan

Cara menguji: (1) buka pertama kali, muncul onboarding; (2) tekan Mulai lalu login; (3) tutup aplikasi sepenuhnya dan buka lagi, langsung masuk Beranda; (4) Logout lalu buka lagi, muncul Login (bukan onboarding).

⚠️ Kesalahan Umum

  • Memakai Navigator.push (bukan pushReplacement) dari splash/login, sehingga tombol Back kembali ke layar login.
  • Logout dengan clear() tanpa filter, sehingga onboarding muncul lagi.
  • Memakai context setelah await tanpa cek mounted / context.mounted.
  • Untuk mengulang dari awal saat testing, hapus data aplikasi (Settings → Apps → Clear storage) atau uninstall. Hot restart tidak menghapus data tersimpan.
