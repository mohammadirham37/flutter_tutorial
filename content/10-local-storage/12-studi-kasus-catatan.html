1. Model

📄 lib/models/catatan.dart

class Catatan {
  static const kategoriList = ['Kuliah', 'Tugas', 'Organisasi', 'Pribadi'];

  final int? id;
  final String judul;
  final String isi;
  final String kategori;
  final bool disematkan;
  final DateTime diubah;

  const Catatan({
    this.id,
    required this.judul,
    required this.isi,
    this.kategori = 'Kuliah',
    this.disematkan = false,
    required this.diubah,
  });

  Map<String, Object?> toMap() => {
        if (id != null) 'id': id,
        'judul': judul,
        'isi': isi,
        'kategori': kategori,
        'disematkan': disematkan ? 1 : 0,
        'diubah': diubah.toIso8601String(),
      };

  factory Catatan.fromMap(Map<String, Object?> m) => Catatan(
        id: m['id'] as int?,
        judul: m['judul'] as String,
        isi: m['isi'] as String,
        kategori: m['kategori'] as String,
        disematkan: m['disematkan'] == 1,
        diubah: DateTime.parse(m['diubah'] as String),
      );

  Catatan copyWith({String? judul, String? isi, String? kategori, bool? disematkan, DateTime? diubah}) {
    return Catatan(
      id: id,
      judul: judul ?? this.judul,
      isi: isi ?? this.isi,
      kategori: kategori ?? this.kategori,
      disematkan: disematkan ?? this.disematkan,
      diubah: diubah ?? this.diubah,
    );
  }
}

2. Lapisan Database

📄 lib/data/catatan_database.dart

import 'package:path/path.dart';
import 'package:sqflite/sqflite.dart';

import '../models/catatan.dart';

class CatatanDatabase {
  CatatanDatabase._();
  static final CatatanDatabase instance = CatatanDatabase._();

  static const _tabel = 'catatan';
  Database? _db;

  Future<Database> get _database async => _db ??= await openDatabase(
        join(await getDatabasesPath(), 'catatan_kuliah.db'),
        version: 1,
        onCreate: (db, version) => db.execute('''
          CREATE TABLE $_tabel (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            judul TEXT NOT NULL,
            isi TEXT NOT NULL,
            kategori TEXT NOT NULL,
            disematkan INTEGER NOT NULL DEFAULT 0,
            diubah TEXT NOT NULL
          )
        '''),
      );

  /// Ambil catatan: yang disematkan di atas, lalu yang terbaru.
  Future<List<Catatan>> getAll({String kata = '', String? kategori}) async {
    final db = await _database;
    final where = <String>[];
    final args = <Object?>[];
    if (kata.isNotEmpty) {
      where.add('(judul LIKE ? OR isi LIKE ?)');
      args.addAll(['%$kata%', '%$kata%']);
    }
    if (kategori != null) {
      where.add('kategori = ?');
      args.add(kategori);
    }
    final rows = await db.query(
      _tabel,
      where: where.isEmpty ? null : where.join(' AND '),
      whereArgs: args.isEmpty ? null : args,
      orderBy: 'disematkan DESC, diubah DESC',
    );
    return rows.map(Catatan.fromMap).toList();
  }

  Future<int> insert(Catatan c) async => (await _database).insert(_tabel, c.toMap());

  Future<void> update(Catatan c) async {
    final db = await _database;
    await db.update(_tabel, c.toMap(), where: 'id = ?', whereArgs: [c.id]);
  }

  Future<void> delete(int id) async {
    final db = await _database;
    await db.delete(_tabel, where: 'id = ?', whereArgs: [id]);
  }
}

3. Provider

📄 lib/providers/catatan_provider.dart

import 'package:flutter/foundation.dart';

import '../data/catatan_database.dart';
import '../models/catatan.dart';

class CatatanProvider extends ChangeNotifier {
  final CatatanDatabase _db;
  CatatanProvider(this._db);

  List<Catatan> _items = [];
  bool _loading = true;
  String _kata = '';
  String? _kategori;

  List<Catatan> get items => List.unmodifiable(_items);
  bool get loading => _loading;
  String? get kategori => _kategori;
  bool get sedangMemfilter => _kata.isNotEmpty || _kategori != null;

  Future<void> muat() async {
    _items = await _db.getAll(kata: _kata, kategori: _kategori);
    _loading = false;
    notifyListeners();
  }

  Future<void> cari(String kata) {
    _kata = kata.trim();
    return muat();
  }

  Future<void> pilihKategori(String? kategori) {
    _kategori = kategori;
    return muat();
  }

  Future<void> simpan(Catatan c) async {
    final baru = c.copyWith(diubah: DateTime.now());
    if (c.id == null) {
      await _db.insert(baru);
    } else {
      await _db.update(baru);
    }
    await muat();
  }

  Future<void> toggleSemat(Catatan c) async {
    await _db.update(c.copyWith(disematkan: !c.disematkan));
    await muat();
  }

  Future<void> hapus(Catatan c) async {
    _items.remove(c); // hilang dari layar seketika (dibutuhkan Dismissible)
    notifyListeners();
    await _db.delete(c.id!);
  }

  /// Urungkan hapus: masukkan lagi lengkap dengan id lamanya
  Future<void> pulihkan(Catatan c) async {
    await _db.insert(c);
    await muat();
  }
}

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../models/catatan.dart';
import '../providers/catatan_provider.dart';
import 'catatan_form_screen.dart';

class CatatanListScreen extends StatelessWidget {
  const CatatanListScreen({super.key});

  void _bukaForm(BuildContext context, [Catatan? c]) {
    Navigator.push(context, MaterialPageRoute(builder: (_) => CatatanFormScreen(catatan: c)));
  }

  @override
  Widget build(BuildContext context) {
    final prov = context.watch<CatatanProvider>();

    return Scaffold(
      appBar: AppBar(title: const Text('Catatan Kuliah')),
      body: Column(
        children: [
          Padding(
            padding: const EdgeInsets.fromLTRB(12, 12, 12, 4),
            child: SearchBar(
              hintText: 'Cari judul atau isi catatan',
              leading: const Icon(Icons.search),
              onChanged: context.read<CatatanProvider>().cari,
            ),
          ),
          SizedBox(
            height: 48,
            child: ListView(
              scrollDirection: Axis.horizontal,
              padding: const EdgeInsets.symmetric(horizontal: 12),
              children: [
                ChoiceChip(
                  label: const Text('Semua'),
                  selected: prov.kategori == null,
                  onSelected: (_) => prov.pilihKategori(null),
                ),
                for (final k in Catatan.kategoriList)
                  Padding(
                    padding: const EdgeInsets.only(left: 8),
                    child: ChoiceChip(
                      label: Text(k),
                      selected: prov.kategori == k,
                      onSelected: (v) => prov.pilihKategori(v ? k : null),
                    ),
                  ),
              ],
            ),
          ),
          Expanded(
            child: prov.loading
                ? const Center(child: CircularProgressIndicator())
                : prov.items.isEmpty
                    ? Center(
                        child: Text(prov.sedangMemfilter
                            ? 'Tidak ada catatan yang cocok.'
                            : 'Belum ada catatan. Tekan + untuk menulis.'),
                      )
                    : ListView.builder(
                        padding: const EdgeInsets.only(bottom: 80),
                        itemCount: prov.items.length,
                        itemBuilder: (context, i) => _CatatanTile(
                          catatan: prov.items[i],
                          onTap: () => _bukaForm(context, prov.items[i]),
                        ),
                      ),
          ),
        ],
      ),
      floatingActionButton: FloatingActionButton.extended(
        onPressed: () => _bukaForm(context),
        icon: const Icon(Icons.edit_note),
        label: const Text('Tulis'),
      ),
    );
  }
}

class _CatatanTile extends StatelessWidget {
  final Catatan catatan;
  final VoidCallback onTap;
  const _CatatanTile({required this.catatan, required this.onTap});

  @override
  Widget build(BuildContext context) {
    final prov = context.read<CatatanProvider>();
    final d = catatan.diubah;
    return Dismissible(
      key: ValueKey(catatan.id),
      direction: DismissDirection.endToStart,
      background: Container(
        color: Colors.red,
        alignment: Alignment.centerRight,
        padding: const EdgeInsets.only(right: 20),
        child: const Icon(Icons.delete, color: Colors.white),
      ),
      onDismissed: (_) {
        final messenger = ScaffoldMessenger.of(context);
        prov.hapus(catatan);
        messenger.showSnackBar(SnackBar(
          content: Text('"${catatan.judul}" dihapus'),
          action: SnackBarAction(label: 'URUNGKAN', onPressed: () => prov.pulihkan(catatan)),
        ));
      },
      child: Card(
        margin: const EdgeInsets.symmetric(horizontal: 12, vertical: 4),
        child: ListTile(
          onTap: onTap,
          title: Text(catatan.judul, maxLines: 1, overflow: TextOverflow.ellipsis),
          subtitle: Text(
            '${catatan.kategori} • ${d.day}/${d.month}/${d.year}\n${catatan.isi}',
            maxLines: 3,
            overflow: TextOverflow.ellipsis,
          ),
          isThreeLine: true,
          trailing: IconButton(
            tooltip: catatan.disematkan ? 'Lepas sematan' : 'Sematkan',
            icon: Icon(catatan.disematkan ? Icons.push_pin : Icons.push_pin_outlined),
            onPressed: () => prov.toggleSemat(catatan),
          ),
        ),
      ),
    );
  }
}

5. Layar Form (Tambah & Edit)

📄 lib/screens/catatan_form_screen.dart

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../models/catatan.dart';
import '../providers/catatan_provider.dart';

class CatatanFormScreen extends StatefulWidget {
  final Catatan? catatan; // null = catatan baru
  const CatatanFormScreen({super.key, this.catatan});

  @override
  State<CatatanFormScreen> createState() => _CatatanFormScreenState();
}

class _CatatanFormScreenState extends State<CatatanFormScreen> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _judul =
      TextEditingController(text: widget.catatan?.judul);
  late final TextEditingController _isi = TextEditingController(text: widget.catatan?.isi);
  late String _kategori = widget.catatan?.kategori ?? Catatan.kategoriList.first;

  @override
  void dispose() {
    _judul.dispose();
    _isi.dispose();
    super.dispose();
  }

  Future<void> _simpan() async {
    if (!_formKey.currentState!.validate()) return;
    final lama = widget.catatan;
    final data = lama == null
        ? Catatan(judul: _judul.text.trim(), isi: _isi.text.trim(), kategori: _kategori, diubah: DateTime.now())
        : lama.copyWith(judul: _judul.text.trim(), isi: _isi.text.trim(), kategori: _kategori);
    await context.read<CatatanProvider>().simpan(data);
    if (mounted) Navigator.pop(context);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(widget.catatan == null ? 'Catatan Baru' : 'Edit Catatan'),
        actions: [IconButton(tooltip: 'Simpan', onPressed: _simpan, icon: const Icon(Icons.check))],
      ),
      body: Form(
        key: _formKey,
        child: ListView(
          padding: const EdgeInsets.all(16),
          children: [
            TextFormField(
              controller: _judul,
              decoration: const InputDecoration(labelText: 'Judul', border: OutlineInputBorder()),
              validator: (v) => (v == null || v.trim().isEmpty) ? 'Judul wajib diisi' : null,
            ),
            const SizedBox(height: 12),
            DropdownButtonFormField<String>(
              initialValue: _kategori,
              decoration: const InputDecoration(labelText: 'Kategori', border: OutlineInputBorder()),
              items: [
                for (final k in Catatan.kategoriList) DropdownMenuItem(value: k, child: Text(k)),
              ],
              onChanged: (v) => setState(() => _kategori = v ?? _kategori),
            ),
            const SizedBox(height: 12),
            TextFormField(
              controller: _isi,
              minLines: 6,
              maxLines: 12,
              decoration: const InputDecoration(
                labelText: 'Isi catatan',
                alignLabelWithHint: true,
                border: OutlineInputBorder(),
              ),
              validator: (v) => (v == null || v.trim().isEmpty) ? 'Isi wajib diisi' : null,
            ),
            const SizedBox(height: 16),
            FilledButton.icon(onPressed: _simpan, icon: const Icon(Icons.save), label: const Text('Simpan')),
          ],
        ),
      ),
    );
  }
}

Penjelasan alur:

  • Pencarian & filter tidak dilakukan di Dart, melainkan diteruskan ke SQL (LIKE dan kategori = ?) lewat CatatanDatabase.getAll.
  • Urutan disematkan DESC, diubah DESC membuat catatan yang di-pin selalu di atas, lalu yang paling baru diubah.
  • Hapus menghapus dari list lebih dulu (kebutuhan Dismissible) lalu dari database. URUNGKAN memasukkan kembali objek beserta id lamanya.
  • Form memakai late final controller yang diinisialisasi dari widget.catatan, sehingga satu layar melayani tambah dan edit.

⚠️ Kesalahan Umum

  • Lupa ..muat() saat membuat provider: spinner tidak pernah hilang karena _loading tetap true.
  • Memanggil Navigator.pop sebelum await simpan(...) selesai, sehingga daftar belum ter-update saat kembali.
  • Memakai value: pada DropdownButtonFormField; versi Flutter terbaru memakai initialValue:.
