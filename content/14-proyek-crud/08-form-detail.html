Satu halaman form untuk dua keperluan, tambah dan edit, adalah pola yang sangat umum. Perbedaannya hanya ditentukan oleh ada atau tidaknya parameter note. Kedua View di section ini memakai NotesController yang sama dengan HomeView, diambil dengan Get.find.

📄 lib/views/note_form_view.dart

import 'package:flutter/material.dart';
import 'package:get/get.dart';

import '../controllers/notes_controller.dart';
import '../models/note.dart';

/// Satu halaman untuk dua keperluan:
/// - note == null  -> mode TAMBAH
/// - note != null  -> mode EDIT
class NoteFormView extends StatefulWidget {
  const NoteFormView({super.key, this.note});

  final Note? note;

  @override
  State<NoteFormView> createState() => _NoteFormViewState();
}

class _NoteFormViewState extends State<NoteFormView> {
  final _notes = Get.find<NotesController>();
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _titleController;
  late final TextEditingController _contentController;
  late String _category;
  bool _isSaving = false;

  bool get _isEdit => widget.note != null;

  @override
  void initState() {
    super.initState();
    // Mode edit: isi form dengan data lama. Mode tambah: form kosong.
    _titleController = TextEditingController(text: widget.note?.title ?? '');
    _contentController = TextEditingController(
      text: widget.note?.content ?? '',
    );
    _category = widget.note?.category ?? Note.categories.first;
  }

  @override
  void dispose() {
    _titleController.dispose();
    _contentController.dispose();
    super.dispose();
  }

  Future<void> _save() async {
    FocusScope.of(context).unfocus();
    if (!_formKey.currentState!.validate()) return;

    setState(() => _isSaving = true);
    final title = _titleController.text.trim();
    final content = _contentController.text.trim();

    final String? error;
    if (_isEdit) {
      error = await _notes.updateNote(
        widget.note!.copyWith(
          title: title,
          content: content,
          category: _category,
        ),
      );
    } else {
      error = await _notes.addNote(
        title: title,
        content: content,
        category: _category,
      );
    }

    if (!mounted) return;
    setState(() => _isSaving = false);

    final messenger = ScaffoldMessenger.of(context);
    if (error == null) {
      messenger.showSnackBar(
        SnackBar(
          content: Text(_isEdit ? 'Catatan diperbarui' : 'Catatan disimpan'),
        ),
      );
      Get.back(); // kembali ke halaman sebelumnya
    } else {
      messenger.showSnackBar(
        SnackBar(
          content: Text(error),
          backgroundColor: Theme.of(context).colorScheme.error,
        ),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text(_isEdit ? 'Edit Catatan' : 'Catatan Baru')),
      body: SafeArea(
        child: Form(
          key: _formKey,
          child: ListView(
            padding: const EdgeInsets.all(16),
            children: [
              TextFormField(
                controller: _titleController,
                textInputAction: TextInputAction.next,
                maxLength: 100,
                decoration: const InputDecoration(
                  labelText: 'Judul',
                  border: OutlineInputBorder(),
                ),
                validator: (value) {
                  if (value == null || value.trim().isEmpty) {
                    return 'Judul wajib diisi';
                  }
                  return null;
                },
              ),
              const SizedBox(height: 8),
              DropdownButtonFormField<String>(
                initialValue: _category,
                decoration: const InputDecoration(
                  labelText: 'Kategori',
                  border: OutlineInputBorder(),
                ),
                items: [
                  for (final category in Note.categories)
                    DropdownMenuItem(value: category, child: Text(category)),
                ],
                onChanged: (value) {
                  if (value != null) setState(() => _category = value);
                },
              ),
              const SizedBox(height: 16),
              TextFormField(
                controller: _contentController,
                minLines: 6,
                maxLines: 12,
                keyboardType: TextInputType.multiline,
                decoration: const InputDecoration(
                  labelText: 'Isi catatan',
                  alignLabelWithHint: true,
                  border: OutlineInputBorder(),
                ),
                validator: (value) {
                  if (value == null || value.trim().isEmpty) {
                    return 'Isi catatan wajib diisi';
                  }
                  return null;
                },
              ),
              const SizedBox(height: 24),
              FilledButton.icon(
                onPressed: _isSaving ? null : _save,
                style: FilledButton.styleFrom(
                  minimumSize: const Size.fromHeight(48),
                ),
                icon: _isSaving
                    ? const SizedBox(
                        width: 20,
                        height: 20,
                        child: CircularProgressIndicator(strokeWidth: 2.5),
                      )
                    : const Icon(Icons.save),
                label: Text(_isSaving ? 'Menyimpan...' : 'Simpan'),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

Membedah NoteFormView

  • bool get _isEdit => widget.note != null adalah satu sumber kebenaran untuk judul AppBar, pesan sukses, dan method controller yang dipanggil.
  • TextEditingController diisi di initState dengan widget.note?.title ?? ''. Mode edit langsung menampilkan data lama, sedangkan mode tambah kosong.
  • DropdownButtonFormField dengan initialValue memilih kategori awal. Itemnya diambil dari Note.categories.
  • maxLength: 100 menampilkan penghitung karakter dan sejalan dengan check di SQL.
  • final _notes = Get.find<NotesController>() mengambil controller global. Form memanggil _notes.addNote atau _notes.updateNote, lalu menerima null (sukses) atau pesan error.
  • _isSaving adalah state lokal dengan setState (bukan di controller) karena hanya relevan untuk halaman ini: menonaktifkan tombol dan menampilkan spinner. Tidak semua state harus masuk GetX. State yang hanya dipakai satu widget cukup disimpan di State widget itu.
  • final String? error; diisi di if/else: Dart mengizinkan variabel final diisi tepat sekali di setiap cabang.
  • Setelah await: if (!mounted) return;, baru setState, SnackBar, dan Get.back() untuk kembali ke halaman sebelumnya. SnackBar tetap terlihat setelah halaman ditutup karena ScaffoldMessenger berada di atas semua halaman.
  • Edit memakai copyWith, sehingga id catatan lama tetap dipakai untuk filter .eq('id', ...).

import 'package:flutter/material.dart';
import 'package:get/get.dart';

import '../controllers/notes_controller.dart';
import '../models/note.dart';
import '../utils/date_format.dart';
import '../widgets/confirm_dialog.dart';
import 'note_form_view.dart';

class NoteDetailView extends GetView<NotesController> {
  const NoteDetailView({super.key, required this.noteId});

  final String noteId;

  Future<void> _delete(BuildContext context, Note note) async {
    final yakin = await showConfirmDialog(
      context,
      title: 'Hapus catatan?',
      message: '"${note.title}" akan dihapus permanen.',
      confirmLabel: 'Hapus',
    );
    if (!yakin || !context.mounted) return;

    final messenger = ScaffoldMessenger.of(context);
    // Tutup halaman detail dulu, baru hapus (optimistic delete).
    Get.back();
    final error = await controller.deleteNote(note);
    messenger.showSnackBar(SnackBar(content: Text(error ?? 'Catatan dihapus')));
  }

  @override
  Widget build(BuildContext context) {
    // Obx membaca versi terbaru catatan dari controller, sehingga setelah
    // diedit halaman ini langsung menampilkan perubahan.
    return Obx(() {
      final note = controller.findById(noteId);
      if (note == null) {
        return Scaffold(
          appBar: AppBar(),
          body: const Center(child: Text('Catatan tidak ditemukan.')),
        );
      }
      return _buildDetail(context, note);
    });
  }

  Widget _buildDetail(BuildContext context, Note note) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Detail Catatan'),
        actions: [
          IconButton(
            tooltip: 'Edit',
            icon: const Icon(Icons.edit),
            onPressed: () => Get.to(() => NoteFormView(note: note)),
          ),
          IconButton(
            tooltip: 'Hapus',
            icon: const Icon(Icons.delete_outline),
            onPressed: () => _delete(context, note),
          ),
        ],
      ),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          Text(note.title, style: Theme.of(context).textTheme.headlineSmall),
          const SizedBox(height: 8),
          Wrap(
            spacing: 8,
            crossAxisAlignment: WrapCrossAlignment.center,
            children: [
              Chip(label: Text(note.category)),
              Text(
                'Diubah ${formatTanggal(note.updatedAt)}',
                style: Theme.of(context).textTheme.bodySmall,
              ),
            ],
          ),
          const Divider(height: 32),
          SelectableText(
            note.content,
            style: Theme.of(context).textTheme.bodyLarge,
          ),
        ],
      ),
    );
  }
}
  • extends GetView<NotesController>: GetView adalah StatelessWidget yang sudah punya getter controller (isinya Get.find<NotesController>()). Cocok untuk halaman tanpa state lokal seperti detail ini.
  • Menerima noteId, bukan objek Note. Di dalam Obx, halaman memanggil controller.findById(noteId), sehingga selalu menampilkan versi terbaru dari controller. Setelah diedit dan form ditutup, detail langsung berubah tanpa refresh.
  • Obx hanya membaca _notes (lewat findById), jadi halaman ini rebuild ketika daftar catatan berubah, tetapi tidak ketika query atau categoryFilter berubah. Inilah alasan kita memakai satu Obx kecil per kebutuhan.
  • Hapus dari detail: halaman ditutup dulu (Get.back()), baru optimistic delete berjalan, lalu SnackBar muncul di HomeView. Messenger diambil sebelum Get.back() karena setelah halaman ditutup context-nya tidak boleh dipakai lagi.
  • Edit: Get.to(() => NoteFormView(note: note)) membuka form yang sama dalam mode edit.
  • SelectableText membuat isi catatan bisa disalin, berguna untuk catatan kuliah.
