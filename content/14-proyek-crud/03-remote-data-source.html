import 'package:supabase_flutter/supabase_flutter.dart';

import '../models/note.dart';

/// Semua operasi CRUD ke tabel `notes` di Supabase.
/// RLS di database memastikan setiap user hanya melihat barisnya sendiri.
class NoteRemoteDataSource {
  NoteRemoteDataSource(this._client);

  final SupabaseClient _client;

  static const String _table = 'notes';

  Future<List<Note>> fetchNotes() async {
    final rows = await _client
        .from(_table)
        .select()
        .order('updated_at', ascending: false);
    return rows.map(Note.fromJson).toList();
  }

  Future<Note> createNote({
    required String title,
    required String content,
    required String category,
  }) async {
    final row = await _client
        .from(_table)
        .insert({'title': title, 'content': content, 'category': category})
        .select()
        .single();
    return Note.fromJson(row);
  }

  Future<Note> updateNote(Note note) async {
    final row = await _client
        .from(_table)
        .update(note.toMap())
        .eq('id', note.id)
        .select()
        .single();
    return Note.fromJson(row);
  }

  Future<void> deleteNote(String id) async {
    await _client.from(_table).delete().eq('id', id);
  }
}

ℹ️ Kenapa insert tidak mengirim user_id?

Kolom user_id memiliki default auth.uid(). Jika Anda tetap ingin mengirimnya secara eksplisit, ambil dari Supabase.instance.client.auth.currentUser!.id. Namun mengandalkan default database lebih sederhana dan tidak mungkin salah.

💡 Update jika RLS menolak

Jika user mencoba mengubah catatan yang bukan miliknya, policy using membuat 0 baris cocok. Karena kita memanggil .single(), hasil 0 baris berubah menjadi PostgrestException (code PGRST116), sehingga kegagalan tidak lolos diam-diam.
