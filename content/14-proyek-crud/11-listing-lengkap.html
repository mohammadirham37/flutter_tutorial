Future<void> main() async {
  // Wajib sebelum memanggil kode async/plugin sebelum runApp.
  WidgetsFlutterBinding.ensureInitialized();

  if (!Env.isConfigured) {
    runApp(const ConfigErrorApp());
    return;
  }

  // 1. Hive (cache lokal)
  await Hive.initFlutter();
  final localDataSource = await NoteLocalDataSource.open();

  // 2. Supabase (backend)
  await Supabase.initialize(
    url: Env.supabaseUrl,
    publishableKey: Env.supabasePublishableKey,
  );
  final client = Supabase.instance.client;

  // 3. Rakit lapisan data: Service/DataSource -> Repository
  final authService = AuthService(client);
  final repository = NoteRepository(
    remote: NoteRemoteDataSource(client),
    local: localDataSource,
  );

  runApp(
    MultiProvider(
      providers: [
        ChangeNotifierProvider(create: (_) => AuthProvider(authService)),
        ChangeNotifierProvider(create: (_) => NotesProvider(repository)),
      ],
      child: const CatatanKuliahApp(),
    ),
  );
}

/// Model satu catatan kuliah. Satu objek Note = satu baris di tabel `notes`.
class Note {
  const Note({
    required this.id,
    required this.userId,
    required this.title,
    required this.content,
    required this.category,
    required this.createdAt,
    required this.updatedAt,
  });

  /// Pilihan kategori yang tersedia di form.
  static const List<String> categories = [
    'Kuliah',
    'Tugas',
    'Ujian',
    'Lainnya',
  ];

  final String id;
  final String userId;
  final String title;
  final String content;
  final String category;
  final DateTime createdAt;
  final DateTime updatedAt;

  /// Membuat Note dari Map. Dipakai untuk data dari Supabase
  /// (`Map<String, dynamic>`) maupun dari Hive (`Map<dynamic, dynamic>`).
  factory Note.fromJson(Map<dynamic, dynamic> json) {
    return Note(
      id: json['id'] as String,
      userId: json['user_id'] as String,
      title: json['title'] as String? ?? '',
      content: json['content'] as String? ?? '',
      category: json['category'] as String? ?? 'Lainnya',
      createdAt: DateTime.parse(json['created_at'] as String),
      updatedAt: DateTime.parse(json['updated_at'] as String),
    );
  }

  /// Seluruh kolom, dipakai untuk menyimpan cache di Hive.
  Map<String, dynamic> toJson() {
    return {
      'id': id,
      'user_id': userId,
      'title': title,
      'content': content,
      'category': category,
      'created_at': createdAt.toIso8601String(),
      'updated_at': updatedAt.toIso8601String(),
    };
  }

  /// Hanya kolom yang boleh diisi/diubah aplikasi, dipakai untuk
  /// insert/update ke Supabase. id, user_id, created_at, dan updated_at
  /// diisi otomatis oleh database.
  Map<String, dynamic> toMap() {
    return {'title': title, 'content': content, 'category': category};
  }

  Note copyWith({
    String? title,
    String? content,
    String? category,
    DateTime? updatedAt,
  }) {
    return Note(
      id: id,
      userId: userId,
      title: title ?? this.title,
      content: content ?? this.content,
      category: category ?? this.category,
      createdAt: createdAt,
      updatedAt: updatedAt ?? this.updatedAt,
    );
  }
}

/// Semua operasi CRUD ke tabel `notes` di Supabase.
/// RLS di database memastikan setiap user hanya melihat barisnya sendiri.
class NoteRemoteDataSource {
  NoteRemoteDataSource(this._client);

  final SupabaseClient _client;

  static const String _table = 'notes';

  Future<List<Note>> fetchNotes() async {
    final rows = await _client
        .from(_table)
        .select()
        .order('updated_at', ascending: false);
    return rows.map(Note.fromJson).toList();
  }

  Future<Note> createNote({
    required String title,
    required String content,
    required String category,
  }) async {
    final row = await _client
        .from(_table)
        .insert({'title': title, 'content': content, 'category': category})
        .select()
        .single();
    return Note.fromJson(row);
  }

  Future<Note> updateNote(Note note) async {
    final row = await _client
        .from(_table)
        .update(note.toMap())
        .eq('id', note.id)
        .select()
        .single();
    return Note.fromJson(row);
  }

  Future<void> deleteNote(String id) async {
    await _client.from(_table).delete().eq('id', id);
  }
}

import '../models/note.dart';

/// Cache catatan di Hive. Setiap catatan disimpan sebagai Map
/// dengan key = id catatan, sehingga tidak perlu TypeAdapter/codegen.
class NoteLocalDataSource {
  NoteLocalDataSource(this._box);

  static const String boxName = 'notes_cache';

  final Box<Map> _box;

  /// Membuka box. Panggil setelah Hive.initFlutter() (atau Hive.init()).
  static Future<NoteLocalDataSource> open() async {
    final box = await Hive.openBox<Map>(boxName);
    return NoteLocalDataSource(box);
  }

  /// Semua catatan di cache, urut dari yang terakhir diubah.
  List<Note> getNotes() {
    final notes = _box.values.map(Note.fromJson).toList();
    notes.sort((a, b) => b.updatedAt.compareTo(a.updatedAt));
    return notes;
  }

  /// Mengganti seluruh isi cache dengan data terbaru dari server.
  Future<void> saveNotes(List<Note> notes) async {
    await _box.clear();
    await _box.putAll({for (final note in notes) note.id: note.toJson()});
  }

  Future<void> saveNote(Note note) => _box.put(note.id, note.toJson());

  Future<void> deleteNote(String id) => _box.delete(id);

  /// Dipanggil saat logout agar data user lain tidak tertinggal.
  Future<void> clear() => _box.clear();
}

import 'dart:async';

import 'package:supabase_flutter/supabase_flutter.dart';

import '../models/note.dart';
import 'note_local_data_source.dart';
import 'note_remote_data_source.dart';

/// Hasil pengambilan daftar catatan.
class NotesResult {
  const NotesResult(this.notes, {required this.fromCache});

  final List<Note> notes;

  /// true jika server tidak bisa dihubungi dan data diambil dari Hive.
  final bool fromCache;
}

/// Satu pintu untuk semua data catatan. Provider tidak perlu tahu
/// apakah data berasal dari Supabase atau dari cache Hive.
class NoteRepository {
  NoteRepository({
    required this.remote,
    required this.local,
    this.timeout = const Duration(seconds: 10),
  });

  final NoteRemoteDataSource remote;
  final NoteLocalDataSource local;
  final Duration timeout;

  /// Ambil dari server lalu simpan ke cache. Jika gagal karena jaringan,
  /// kembalikan isi cache (mode offline).
  Future<NotesResult> getNotes() async {
    try {
      final notes = await remote.fetchNotes().timeout(timeout);
      await local.saveNotes(notes);
      return NotesResult(notes, fromCache: false);
    } on PostgrestException {
      // Server menjawab tapi menolak (misalnya RLS / tabel tidak ada).
      // Ini bukan masalah jaringan, jadi teruskan errornya.
      rethrow;
    } catch (_) {
      // SocketException, ClientException, TimeoutException, dll.
      return NotesResult(local.getNotes(), fromCache: true);
    }
  }

  Future<Note> addNote({
    required String title,
    required String content,
    required String category,
  }) async {
    final note = await remote
        .createNote(title: title, content: content, category: category)
        .timeout(timeout);
    await local.saveNote(note);
    return note;
  }

  Future<Note> updateNote(Note note) async {
    final updated = await remote.updateNote(note).timeout(timeout);
    await local.saveNote(updated);
    return updated;
  }

  Future<void> deleteNote(String id) async {
    await remote.deleteNote(id).timeout(timeout);
    await local.deleteNote(id);
  }

  Future<void> clearCache() => local.clear();
}

import 'package:flutter/foundation.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../services/auth_service.dart';

/// Hasil proses registrasi, dipakai RegisterScreen untuk menentukan
/// pesan yang ditampilkan.
enum RegisterResult { failed, loggedIn, needsEmailConfirmation }

class AuthProvider extends ChangeNotifier {
  AuthProvider(this._authService);

  final AuthService _authService;

  bool _isLoading = false;
  String? _errorMessage;

  bool get isLoading => _isLoading;
  String? get errorMessage => _errorMessage;

  Stream<AuthState> get authStateChanges => _authService.authStateChanges;
  Session? get currentSession => _authService.currentSession;
  User? get currentUser => _authService.currentUser;

  /// Nama yang ditampilkan di AppBar. Diambil dari metadata saat register.
  String get displayName {
    final user = currentUser;
    if (user == null) return '';
    final name = user.userMetadata?['full_name'];
    if (name is String && name.isNotEmpty) return name;
    return user.email ?? '';
  }

  Future<bool> signIn({required String email, required String password}) async {
    _setLoading(true);
    try {
      await _authService.signIn(email: email.trim(), password: password);
      return true;
    } on AuthException catch (e) {
      _errorMessage = _mapAuthError(e);
      return false;
    } catch (e) {
      _errorMessage = 'Terjadi kesalahan tak terduga: $e';
      return false;
    } finally {
      _setLoading(false);
    }
  }

  Future<RegisterResult> register({
    required String fullName,
    required String email,
    required String password,
  }) async {
    _setLoading(true);
    try {
      final response = await _authService.signUp(
        fullName: fullName.trim(),
        email: email.trim(),
        password: password,
      );
      // Jika "Confirm email" aktif di dashboard, session bernilai null
      // sampai pengguna mengeklik link di email.
      if (response.session == null) {
        return RegisterResult.needsEmailConfirmation;
      }
      return RegisterResult.loggedIn;
    } on AuthException catch (e) {
      _errorMessage = _mapAuthError(e);
      return RegisterResult.failed;
    } catch (e) {
      _errorMessage = 'Terjadi kesalahan tak terduga: $e';
      return RegisterResult.failed;
    } finally {
      _setLoading(false);
    }
  }

  Future<void> signOut() async {
    try {
      await _authService.signOut();
    } on AuthException catch (e) {
      // signOut tetap menghapus sesi lokal walau request ke server gagal.
      debugPrint('signOut: ${e.message}');
    }
  }

  void clearError() {
    _errorMessage = null;
    notifyListeners();
  }

  void _setLoading(bool value) {
    _isLoading = value;
    if (value) _errorMessage = null;
    notifyListeners();
  }

  /// Menerjemahkan error Supabase Auth menjadi pesan berbahasa Indonesia.
  String _mapAuthError(AuthException e) {
    if (e is AuthRetryableFetchException) {
      return 'Tidak dapat terhubung ke server. Periksa koneksi internet.';
    }
    switch (e.code) {
      case 'invalid_credentials':
        return 'Email atau password salah.';
      case 'email_not_confirmed':
        return 'Email belum dikonfirmasi. Buka link verifikasi di inbox Anda.';
      case 'user_already_exists':
      case 'email_exists':
        return 'Email sudah terdaftar. Silakan login.';
      case 'weak_password':
        return 'Password terlalu lemah. Gunakan minimal 6 karakter.';
      case 'email_address_invalid':
        return 'Alamat email tidak valid.';
      case 'over_email_send_rate_limit':
      case 'over_request_rate_limit':
        return 'Terlalu banyak percobaan. Tunggu beberapa menit lalu coba lagi.';
      case 'signup_disabled':
        return 'Pendaftaran akun baru sedang dinonaktifkan.';
    }
    return e.message;
  }
}

import 'dart:async';

class NotesProvider extends ChangeNotifier {
  NotesProvider(this._repository);

  final NoteRepository _repository;

  List<Note> _notes = [];
  bool _isLoading = false;
  bool _isOffline = false;
  String? _errorMessage;
  String _query = '';
  String? _categoryFilter; // null = semua kategori

  bool get isLoading => _isLoading;
  bool get isOffline => _isOffline;
  String? get errorMessage => _errorMessage;
  String get query => _query;
  String? get categoryFilter => _categoryFilter;
  bool get hasNotes => _notes.isNotEmpty;

  /// Daftar yang ditampilkan di layar setelah difilter pencarian & kategori.
  List<Note> get visibleNotes {
    final q = _query.toLowerCase();
    return _notes.where((note) {
      final cocokKategori =
          _categoryFilter == null || note.category == _categoryFilter;
      final cocokCari =
          q.isEmpty ||
          note.title.toLowerCase().contains(q) ||
          note.content.toLowerCase().contains(q);
      return cocokKategori && cocokCari;
    }).toList();
  }

  Note? findById(String id) {
    for (final note in _notes) {
      if (note.id == id) return note;
    }
    return null;
  }

  /// Memuat catatan (dipanggil saat HomeScreen dibuka & saat pull-to-refresh).
  Future<void> loadNotes() async {
    _isLoading = true;
    _errorMessage = null;
    notifyListeners();

    try {
      final result = await _repository.getNotes();
      _notes = result.notes;
      _isOffline = result.fromCache;
    } catch (e) {
      _errorMessage = _friendlyError(e);
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }

  /// Mengembalikan null jika berhasil, atau pesan error jika gagal.
  Future<String?> addNote({
    required String title,
    required String content,
    required String category,
  }) async {
    try {
      final note = await _repository.addNote(
        title: title,
        content: content,
        category: category,
      );
      _notes.insert(0, note);
      _isOffline = false;
      notifyListeners();
      return null;
    } catch (e) {
      return _friendlyError(e);
    }
  }

  Future<String?> updateNote(Note note) async {
    try {
      final updated = await _repository.updateNote(note);
      _notes.removeWhere((n) => n.id == updated.id);
      _notes.insert(0, updated); // yang baru diubah naik ke atas
      _isOffline = false;
      notifyListeners();
      return null;
    } catch (e) {
      return _friendlyError(e);
    }
  }

  /// Optimistic delete: hapus dari layar dulu, baru kirim ke server.
  /// Jika server gagal, catatan dikembalikan ke posisi semula (rollback).
  Future<String?> deleteNote(Note note) async {
    final index = _notes.indexWhere((n) => n.id == note.id);
    if (index == -1) return null;

    _notes.removeAt(index);
    notifyListeners();

    try {
      await _repository.deleteNote(note.id);
      return null;
    } catch (e) {
      _notes.insert(index, note);
      notifyListeners();
      return _friendlyError(e);
    }
  }

  void setQuery(String value) {
    _query = value;
    notifyListeners();
  }

  void setCategoryFilter(String? category) {
    _categoryFilter = category;
    notifyListeners();
  }

  /// Dipanggil saat logout: kosongkan state dan cache Hive.
  Future<void> clear() async {
    _notes = [];
    _query = '';
    _categoryFilter = null;
    _errorMessage = null;
    _isOffline = false;
    await _repository.clearCache();
    notifyListeners();
  }

  String _friendlyError(Object e) {
    if (e is PostgrestException) {
      if (e.code == '42501') {
        return 'Akses ditolak oleh Row Level Security. Periksa policy tabel notes.';
      }
      if (e.code == '42P01' || e.code == 'PGRST205') {
        return 'Tabel notes belum dibuat. Jalankan SQL di Supabase.';
      }
      return 'Kesalahan database: ${e.message}';
    }
    if (e is TimeoutException) {
      return 'Server terlalu lama merespons. Coba lagi.';
    }
    return 'Tidak dapat terhubung ke server. Periksa koneksi internet.';
  }
}

  State<AuthGate> createState() => _AuthGateState();

class _AuthGateState extends State<AuthGate> {
  // Simpan stream di State agar tidak dibuat ulang setiap build.
  late final Stream<AuthState> _authStream;

  @override
  void initState() {
    super.initState();
    _authStream = context.read<AuthProvider>().authStateChanges;
  }

  @override
  Widget build(BuildContext context) {
    return StreamBuilder<AuthState>(
      stream: _authStream,
      builder: (context, snapshot) {
        // Sebelum event pertama datang, pakai sesi yang tersimpan di perangkat.
        final session = snapshot.hasData
            ? snapshot.data!.session
            : context.read<AuthProvider>().currentSession;

        if (session == null) {
          return const LoginScreen();
        }
        return const HomeScreen();
      },
    );
  }

          icon: const Icon(Icons.refresh),
          label: const Text('Coba lagi'),
        ),
      );
    }

    final notes = provider.visibleNotes;

    // 3. Kosong
    if (notes.isEmpty) {
      return _MessageView(
        icon: Icons.note_alt_outlined,
        message: provider.hasNotes
            ? 'Tidak ada catatan yang cocok.'
            : 'Belum ada catatan.\nTekan tombol + untuk menambah.',
      );

    return ListView.builder(

  late final TextEditingController _contentController;
  late String _category;
  bool _isSaving = false;

  bool get _isEdit => widget.note != null;

  @override
  void initState() {
    super.initState();
    // Mode edit: isi form dengan data lama. Mode tambah: form kosong.
    _titleController = TextEditingController(text: widget.note?.title ?? '');
    _contentController = TextEditingController(
      text: widget.note?.content ?? '',

    _category = widget.note?.category ?? Note.categories.first;

        appBar: AppBar(),

/// Dialog konfirmasi Ya/Batal yang dipakai ulang di beberapa layar.
/// Mengembalikan true jika pengguna menekan tombol konfirmasi.
Future<bool> showConfirmDialog(
  BuildContext context, {
  required String title,
  required String message,
  required String confirmLabel,
}) async {
  final hasil = await showDialog<bool>(
    context: context,
    builder: (dialogContext) => AlertDialog(
      title: Text(title),
      content: Text(message),
      actions: [
        TextButton(
          onPressed: () => Navigator.of(dialogContext).pop(false),
          child: const Text('Batal'),
        ),
        FilledButton(
          onPressed: () => Navigator.of(dialogContext).pop(true),
          child: Text(confirmLabel),
        ),
      ],
    ),
  );
  return hasil ?? false;
}

/// Kumpulan validator untuk TextFormField.
/// Setiap fungsi mengembalikan null jika valid, atau pesan error jika tidak.
class Validators {
  static String? fullName(String? value) {
    final text = value?.trim() ?? '';
    if (text.isEmpty) return 'Nama lengkap wajib diisi';
    if (text.length < 3) return 'Nama minimal 3 karakter';
    return null;
  }

  static String? email(String? value) {
    final text = value?.trim() ?? '';
    if (text.isEmpty) return 'Email wajib diisi';
    final pola = RegExp(r'^[^@\s]+@[^@\s]+\.[^@\s]+$');
    if (!pola.hasMatch(text)) return 'Format email tidak valid';
    return null;
  }

  static String? password(String? value) {
    final text = value ?? '';
    if (text.isEmpty) return 'Password wajib diisi';
    if (text.length < 6) return 'Password minimal 6 karakter';
    return null;
  }
}

/// Format tanggal sederhana tanpa package tambahan, contoh: 5 Okt 2026 14:07.
String formatTanggal(DateTime waktu) {
  const bulan = [
    'Jan',
    'Feb',
    'Mar',
    'Apr',
    'Mei',
    'Jun',
    'Jul',
    'Agu',
    'Sep',
    'Okt',
    'Nov',
    'Des',
  ];
  final lokal = waktu.toLocal();
  final jam = lokal.hour.toString().padLeft(2, '0');
  final menit = lokal.minute.toString().padLeft(2, '0');
  return '${lokal.day} ${bulan[lokal.month - 1]} ${lokal.year} $jam:$menit';
}

Note buatNote(String id, {String title = 'Judul'}) {

    userId: 'user-1',

/// Pengganti Supabase untuk test. `implements` membuat kita wajib
/// menulis ulang semua method, sehingga tidak ada request jaringan sungguhan.
class FakeRemoteDataSource implements NoteRemoteDataSource {
  List<Note> serverNotes = [];
  Object? errorToThrow; // jika diisi, setiap method melempar error ini

  void _maybeThrow() {
    final error = errorToThrow;
    if (error != null) throw error;
  }

  @override
  Future<List<Note>> fetchNotes() async {
    _maybeThrow();
    return serverNotes;
  }

  @override
  Future<Note> createNote({
    required String title,
    required String content,
    required String category,
  }) async {
    _maybeThrow();
    final note = buatNote('baru', title: title);
    serverNotes.add(note);
    return note;
  }

  @override
  Future<Note> updateNote(Note note) async {
    _maybeThrow();
    return note;
  }

  @override
  Future<void> deleteNote(String id) async {
    _maybeThrow();
    serverNotes.removeWhere((n) => n.id == id);
  }
}

void main() {
  late Directory tempDir;
  late NoteLocalDataSource local;
  late FakeRemoteDataSource remote;
  late NoteRepository repository;

  setUp(() async {
    tempDir = await Directory.systemTemp.createTemp('repo_test_');
    Hive.init(tempDir.path);
    local = await NoteLocalDataSource.open();
    remote = FakeRemoteDataSource();
    repository = NoteRepository(remote: remote, local: local);
  });

  tearDown(() async {
    await Hive.deleteFromDisk();
    await tempDir.delete(recursive: true);
  });

  test('online: data server dikembalikan dan disimpan ke cache', () async {
    remote.serverNotes = [buatNote('1'), buatNote('2')];

    final result = await repository.getNotes();

    expect(result.fromCache, isFalse);
    expect(result.notes, hasLength(2));
    expect(local.getNotes(), hasLength(2));
  });

  test('offline: jika jaringan gagal, data diambil dari cache', () async {
    await local.saveNotes([buatNote('cache-1', title: 'Dari Hive')]);
    remote.errorToThrow = const SocketException('Tidak ada internet');

    final result = await repository.getNotes();

    expect(result.fromCache, isTrue);
    expect(result.notes.single.title, 'Dari Hive');
  });

  test('error dari server (PostgrestException) tidak disembunyikan', () async {
    remote.errorToThrow = const PostgrestException(
      message: 'permission denied',
      code: '42501',
    );

    expect(repository.getNotes(), throwsA(isA<PostgrestException>()));
  });

  test('addNote dan deleteNote ikut memperbarui cache', () async {
    final note = await repository.addNote(
      title: 'Basis Data',
      content: 'Normalisasi',
      category: 'Kuliah',
    );
    expect(local.getNotes().single.id, note.id);

    await repository.deleteNote(note.id);
    expect(local.getNotes(), isEmpty);
  });

  test('deleteNote gagal: cache tidak berubah', () async {
    await local.saveNotes([buatNote('1')]);
    remote.errorToThrow = const SocketException('offline');

    await expectLater(
      repository.deleteNote('1'),
      throwsA(isA<SocketException>()),
    );
    expect(local.getNotes(), hasLength(1));
  });
}
