/// Format tanggal sederhana tanpa package tambahan, contoh: 5 Okt 2026 14:07.
String formatTanggal(DateTime waktu) {
  const bulan = [
    'Jan',
    'Feb',
    'Mar',
    'Apr',
    'Mei',
    'Jun',
    'Jul',
    'Agu',
    'Sep',
    'Okt',
    'Nov',
    'Des',
  ];
  final lokal = waktu.toLocal();
  final jam = lokal.hour.toString().padLeft(2, '0');
  final menit = lokal.minute.toString().padLeft(2, '0');
  return '${lokal.day} ${bulan[lokal.month - 1]} ${lokal.year} $jam:$menit';
}

toLocal() penting: Supabase menyimpan waktu dalam UTC, sedangkan pengguna di Purwokerto berada di WIB (UTC+7).

import 'package:flutter/material.dart';

/// Dialog konfirmasi Ya/Batal yang dipakai ulang di beberapa layar.
/// Mengembalikan true jika pengguna menekan tombol konfirmasi.
Future<bool> showConfirmDialog(
  BuildContext context, {
  required String title,
  required String message,
  required String confirmLabel,
}) async {
  final hasil = await showDialog<bool>(
    context: context,
    builder: (dialogContext) => AlertDialog(
      title: Text(title),
      content: Text(message),
      actions: [
        TextButton(
          onPressed: () => Navigator.of(dialogContext).pop(false),
          child: const Text('Batal'),
        ),
        FilledButton(
          onPressed: () => Navigator.of(dialogContext).pop(true),
          child: Text(confirmLabel),
        ),
      ],
    ),
  );
  return hasil ?? false;
}

showDialog<bool> mengembalikan nilai yang dikirim lewat Navigator.pop(true/false), atau null jika user menutup dialog dengan mengetuk area luar. Karena itu ada ?? false. Di dalam dialog, Get.back(result: true) juga bisa dipakai, tetapi Navigator.of(dialogContext).pop membuat fungsi ini tetap bisa dipakai tanpa GetX.

Hapus seluruh isi HomeView sementara dari Modul 13 dan ganti dengan kode berikut. Perhatikan bahwa kali ini HomeView berupa StatefulWidget, karena ia memiliki TextEditingController untuk kolom pencarian (state lokal milik halaman ini).

📄 lib/views/home_view.dart

import 'package:flutter/material.dart';
import 'package:get/get.dart';

import '../models/note.dart';
import '../controllers/auth_controller.dart';
import '../controllers/notes_controller.dart';
import '../widgets/confirm_dialog.dart';
import '../widgets/note_card.dart';
import '../widgets/offline_banner.dart';
import 'note_detail_view.dart';
import 'note_form_view.dart';

class HomeView extends StatefulWidget {
  const HomeView({super.key});

  @override
  State<HomeView> createState() => _HomeViewState();
}

class _HomeViewState extends State<HomeView> {
  // Controller global yang didaftarkan di main() dengan Get.put.
  final _auth = Get.find<AuthController>();
  final _notes = Get.find<NotesController>();
  final _searchController = TextEditingController();

  @override
  void initState() {
    super.initState();
    // HomeView dibuat setiap kali user berhasil login, jadi inilah saat
    // yang tepat untuk memuat catatan milik user tersebut.
    _notes.loadNotes();
  }

  @override
  void dispose() {
    _searchController.dispose();
    super.dispose();
  }

  Future<void> _logout() async {
    final yakin = await showConfirmDialog(
      context,
      title: 'Keluar?',
      message: 'Catatan yang tersimpan di perangkat akan dihapus.',
      confirmLabel: 'Keluar',
    );
    if (!yakin) return;

    await _notes.clear(); // kosongkan state + cache Hive
    await _auth.signOut(); // AuthGate otomatis kembali ke LoginView
  }

  Future<bool> _confirmDelete(Note note) {
    return showConfirmDialog(
      context,
      title: 'Hapus catatan?',
      message: '"${note.title}" akan dihapus permanen.',
      confirmLabel: 'Hapus',
    );
  }

  Future<void> _delete(Note note) async {
    final messenger = ScaffoldMessenger.of(context);
    final error = await _notes.deleteNote(note);
    messenger.showSnackBar(
      SnackBar(
        content: Text(
          error == null
              ? 'Catatan dihapus'
              : 'Gagal menghapus, catatan dikembalikan. $error',
        ),
      ),
    );
  }

  void _openForm() => Get.to(() => const NoteFormView());

  void _openDetail(Note note) => Get.to(() => NoteDetailView(noteId: note.id));

  @override
  Widget build(BuildContext context) {
    final name = _auth.displayName;

    return Scaffold(
      appBar: AppBar(
        title: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const Text('Catatan Kuliah'),
            Text('Halo, $name', style: Theme.of(context).textTheme.bodySmall),
          ],
        ),
        actions: [
          IconButton(
            tooltip: 'Keluar',
            icon: const Icon(Icons.logout),
            onPressed: _logout,
          ),
        ],
      ),
      floatingActionButton: FloatingActionButton.extended(
        onPressed: _openForm,
        icon: const Icon(Icons.add),
        label: const Text('Catatan'),
      ),
      body: Column(
        children: [
          // Setiap Obx hanya rebuild jika state yang dibacanya berubah.
          Obx(
            () => _notes.isOffline
                ? OfflineBanner(onRetry: _notes.loadNotes)
                : const SizedBox.shrink(),
          ),
          Padding(
            padding: const EdgeInsets.fromLTRB(16, 12, 16, 4),
            child: Obx(
              () => TextField(
                controller: _searchController,
                onChanged: _notes.setQuery,
                decoration: InputDecoration(
                  hintText: 'Cari judul atau isi catatan...',
                  prefixIcon: const Icon(Icons.search),
                  border: const OutlineInputBorder(),
                  isDense: true,
                  suffixIcon: _notes.query.isEmpty
                      ? null
                      : IconButton(
                          icon: const Icon(Icons.clear),
                          onPressed: () {
                            _searchController.clear();
                            _notes.setQuery('');
                          },
                        ),
                ),
              ),
            ),
          ),
          Obx(
            () => _CategoryFilter(
              selected: _notes.categoryFilter,
              onSelected: _notes.setCategoryFilter,
            ),
          ),
          Expanded(
            child: RefreshIndicator(
              onRefresh: _notes.loadNotes,
              child: Obx(_buildBody),
            ),
          ),
        ],
      ),
    );
  }

  Widget _buildBody() {
    // 1. Loading pertama kali (belum ada data sama sekali)
    if (_notes.isLoading && !_notes.hasNotes) {
      return const Center(child: CircularProgressIndicator());
    }

    // 2. Error dan tidak ada data cadangan
    if (_notes.errorMessage != null && !_notes.hasNotes) {
      return _MessageView(
        icon: Icons.error_outline,
        message: _notes.errorMessage!,
        action: FilledButton.icon(
          onPressed: _notes.loadNotes,
          icon: const Icon(Icons.refresh),
          label: const Text('Coba lagi'),
        ),
      );
    }

    final notes = _notes.visibleNotes;

    // 3. Kosong
    if (notes.isEmpty) {
      return _MessageView(
        icon: Icons.note_alt_outlined,
        message: _notes.hasNotes
            ? 'Tidak ada catatan yang cocok.'
            : 'Belum ada catatan.\nTekan tombol + untuk menambah.',
      );
    }

    // 4. Ada data
    return ListView.builder(
      physics: const AlwaysScrollableScrollPhysics(),
      padding: const EdgeInsets.only(top: 4, bottom: 88),
      itemCount: notes.length,
      itemBuilder: (context, index) {
        final note = notes[index];
        return Dismissible(
          key: ValueKey(note.id),
          direction: DismissDirection.endToStart,
          background: Container(
            alignment: Alignment.centerRight,
            padding: const EdgeInsets.only(right: 24),
            color: Theme.of(context).colorScheme.errorContainer,
            child: Icon(
              Icons.delete,
              color: Theme.of(context).colorScheme.onErrorContainer,
            ),
          ),
          confirmDismiss: (_) => _confirmDelete(note),
          onDismissed: (_) => _delete(note),
          child: NoteCard(note: note, onTap: () => _openDetail(note)),
        );
      },
    );
  }
}

/// Deretan chip untuk memfilter kategori.
class _CategoryFilter extends StatelessWidget {
  const _CategoryFilter({required this.selected, required this.onSelected});

  final String? selected;
  final ValueChanged<String?> onSelected;

  @override
  Widget build(BuildContext context) {
    return SizedBox(
      height: 48,
      child: ListView(
        scrollDirection: Axis.horizontal,
        padding: const EdgeInsets.symmetric(horizontal: 16),
        children: [
          Padding(
            padding: const EdgeInsets.only(right: 8),
            child: ChoiceChip(
              label: const Text('Semua'),
              selected: selected == null,
              onSelected: (_) => onSelected(null),
            ),
          ),
          for (final category in Note.categories)
            Padding(
              padding: const EdgeInsets.only(right: 8),
              child: ChoiceChip(
                label: Text(category),
                selected: selected == category,
                onSelected: (_) => onSelected(category),
              ),
            ),
        ],
      ),
    );
  }
}

/// Tampilan pesan di tengah layar yang tetap bisa ditarik (pull-to-refresh).
class _MessageView extends StatelessWidget {
  const _MessageView({required this.icon, required this.message, this.action});

  final IconData icon;
  final String message;
  final Widget? action;

  @override
  Widget build(BuildContext context) {
    return ListView(
      physics: const AlwaysScrollableScrollPhysics(),
      padding: const EdgeInsets.all(32),
      children: [
        const SizedBox(height: 48),
        Icon(icon, size: 64, color: Theme.of(context).colorScheme.outline),
        const SizedBox(height: 16),
        Text(message, textAlign: TextAlign.center),
        if (action != null) ...[
          const SizedBox(height: 16),
          Center(child: action),
        ],
      ],
    );
  }
}

Membedah HomeView

BagianPenjelasan
Get.find<AuthController>(), Get.find<NotesController>()Mengambil controller global yang sudah didaftarkan di main() dengan Get.put (section 9). Tidak butuh context, jadi bisa langsung dijadikan field.
_notes.loadNotes() di initStateMemuat catatan setiap kali HomeView dibuat, yaitu setiap kali user berhasil login. Mengubah Rx di initState aman, karena belum ada Obx di halaman ini yang sedang dibangun.
_logout()Konfirmasi, lalu _notes.clear() (kosongkan state + Hive), lalu _auth.signOut(). Tidak ada navigasi manual: AuthGate otomatis kembali ke LoginView.
Beberapa Obx kecilBanner, kolom pencarian, chip kategori, dan isi daftar masing-masing dibungkus Obx sendiri. Setiap Obx hanya rebuild jika Rx yang dibacanya berubah. Misalnya Obx banner hanya membaca isOffline, jadi tidak ikut rebuild saat user mengetik.
OfflineBanner(onRetry: _notes.loadNotes)Banner hanya muncul jika data berasal dari cache. Jika tidak, Obx mengembalikan SizedBox.shrink() (Obx wajib mengembalikan widget). Tombol "Coba lagi" memanggil ulang server.
TextField(onChanged: _notes.setQuery)Setiap ketikan memperbarui filter. Ikon ✕ muncul jika query tidak kosong dan mengosongkan pencarian.
_CategoryFilterDeretan ChoiceChip. Memilih "Semua" berarti null.
Obx(_buildBody)Obx menerima fungsi tanpa parameter yang mengembalikan widget, jadi method _buildBody bisa langsung dioper. Semua getter controller yang dibaca di dalamnya ikut didengarkan.
Get.to(() => const NoteFormView())Navigasi GetX tanpa context. Halaman detail menerima noteId (section 8).
RefreshIndicator(onRefresh: _notes.loadNotes)Tarik ke bawah untuk memuat ulang. Child-nya harus bisa di-scroll, karena itu tampilan kosong/error juga berupa ListView dengan AlwaysScrollableScrollPhysics.
_buildBody dengan 4 kondisiUrutan pengecekan: (1) loading pertama, (2) error tanpa data, (3) kosong, (4) daftar. Jika ada data cache, error tidak menutupi data.
Dismissiblekey: ValueKey(note.id) harus unik dan stabil. confirmDismiss menampilkan dialog; jika hasilnya false, kartu kembali ke tempatnya. onDismissed memanggil optimistic delete.
ScaffoldMessenger.of(context) sebelum awaitDi _delete, messenger diambil sebelum menunggu server, sehingga aman dipakai setelahnya tanpa melanggar aturan context setelah await.
padding: bottom: 88Memberi ruang agar kartu terakhir tidak tertutup FAB.

Error ini muncul jika item yang sudah digeser tidak segera dihapus dari list. Optimistic delete di NotesController menghapusnya secara sinkron sebelum await, sehingga error ini tidak terjadi.
