AuthController adalah huruf C dalam MVC: ia menjembatani View (halaman) dan AuthService. Tugasnya: menyimpan status loading dan status login secara reaktif, menangkap exception, lalu menerjemahkannya menjadi pesan Bahasa Indonesia yang ramah pengguna.

import 'dart:async';

import 'package:flutter/foundation.dart';
import 'package:get/get.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../services/auth_service.dart';

/// Hasil proses registrasi, dipakai RegisterView untuk menentukan
/// pesan yang ditampilkan.
enum RegisterResult { failed, loggedIn, needsEmailConfirmation }

class AuthController extends GetxController {
  AuthController(this._authService);

  final AuthService _authService;

  // State privat yang reaktif.
  final _isLoading = false.obs;
  final _errorMessage = RxnString();
  final _session = Rxn<Session>();

  StreamSubscription<AuthState>? _authSubscription;

  // Getter publik read-only. Membaca getter ini di dalam Obx
  // otomatis mendaftarkan Obx sebagai pendengar.
  bool get isLoading => _isLoading.value;
  String? get errorMessage => _errorMessage.value;

  Stream<AuthState> get authStateChanges => _authService.authStateChanges;
  Session? get currentSession => _session.value;
  User? get currentUser => _session.value?.user;

  /// Nama yang ditampilkan di AppBar. Diambil dari metadata saat register.
  String get displayName {
    final user = currentUser;
    if (user == null) return '';
    final name = user.userMetadata?['full_name'];
    if (name is String && name.isNotEmpty) return name;
    return user.email ?? '';
  }

  @override
  void onInit() {
    super.onInit();
    // Mulai dari sesi yang tersimpan di perangkat (jika ada) ...
    _session.value = _authService.currentSession;
    // ... lalu ikuti setiap perubahan status login.
    _authSubscription = _authService.authStateChanges.listen(
      (data) => _session.value = data.session,
      onError: (Object e) => debugPrint('authStateChanges: $e'),
    );
  }

  @override
  void onClose() {
    _authSubscription?.cancel();
    super.onClose();
  }

  Future<bool> signIn({required String email, required String password}) async {
    _setLoading(true);
    try {
      await _authService.signIn(email: email.trim(), password: password);
      return true;
    } on AuthException catch (e) {
      _errorMessage.value = _mapAuthError(e);
      return false;
    } catch (e) {
      _errorMessage.value = 'Terjadi kesalahan tak terduga: $e';
      return false;
    } finally {
      _setLoading(false);
    }
  }

  Future<RegisterResult> register({
    required String fullName,
    required String email,
    required String password,
  }) async {
    _setLoading(true);
    try {
      final response = await _authService.signUp(
        fullName: fullName.trim(),
        email: email.trim(),
        password: password,
      );
      // Jika "Confirm email" aktif di dashboard, session bernilai null
      // sampai pengguna mengeklik link di email.
      if (response.session == null) {
        return RegisterResult.needsEmailConfirmation;
      }
      return RegisterResult.loggedIn;
    } on AuthException catch (e) {
      _errorMessage.value = _mapAuthError(e);
      return RegisterResult.failed;
    } catch (e) {
      _errorMessage.value = 'Terjadi kesalahan tak terduga: $e';
      return RegisterResult.failed;
    } finally {
      _setLoading(false);
    }
  }

  Future<void> signOut() async {
    try {
      await _authService.signOut();
    } on AuthException catch (e) {
      // signOut tetap menghapus sesi lokal walau request ke server gagal.
      debugPrint('signOut: ${e.message}');
    }
  }

  void clearError() {
    _errorMessage.value = null;
  }

  void _setLoading(bool value) {
    _isLoading.value = value;
    if (value) _errorMessage.value = null;
  }

  /// Menerjemahkan error Supabase Auth menjadi pesan berbahasa Indonesia.
  String _mapAuthError(AuthException e) {
    if (e is AuthRetryableFetchException) {
      return 'Tidak dapat terhubung ke server. Periksa koneksi internet.';
    }
    switch (e.code) {
      case 'invalid_credentials':
        return 'Email atau password salah.';
      case 'email_not_confirmed':
        return 'Email belum dikonfirmasi. Buka link verifikasi di inbox Anda.';
      case 'user_already_exists':
      case 'email_exists':
        return 'Email sudah terdaftar. Silakan login.';
      case 'weak_password':
        return 'Password terlalu lemah. Gunakan minimal 6 karakter.';
      case 'email_address_invalid':
        return 'Alamat email tidak valid.';
      case 'over_email_send_rate_limit':
      case 'over_request_rate_limit':
        return 'Terlalu banyak percobaan. Tunggu beberapa menit lalu coba lagi.';
      case 'signup_disabled':
        return 'Pendaftaran akun baru sedang dinonaktifkan.';
    }
    return e.message;
  }
}

Membedah AuthController

BagianPenjelasan
enum RegisterResultRegister punya tiga kemungkinan hasil: gagal, langsung login, atau perlu konfirmasi email. Enum membuat View bisa memakai switch yang jelas, alih-alih bool yang ambigu.
_isLoading = false.obs, _errorMessage = RxnString()State privat yang reaktif (Modul 08). RxnString adalah Rx untuk String? yang boleh bernilai null. View hanya membaca lewat getter publik isLoading dan errorMessage, sehingga tidak bisa mengubah state dari luar.
_session = Rxn<Session>()Salinan reaktif status login. currentSession dan currentUser membaca nilai ini, jadi widget di dalam Obx otomatis dibangun ulang saat user login atau logout.
onInit()Dipanggil sekali saat controller didaftarkan dengan Get.put. Di sini kita mengisi sesi awal dari perangkat, lalu listen ke authStateChanges (stream onAuthStateChange). Setiap event membawa session terbaru.
onClose()Pasangan onInit: membatalkan StreamSubscription agar tidak bocor memori bila controller dihapus.
displayNameGetter turunan: mengambil full_name dari userMetadata, dengan email sebagai cadangan jika nama kosong. Karena membaca currentUser, getter ini juga reaktif di dalam Obx.
try / on AuthException / catch / finallyfinally selalu dijalankan, baik sukses maupun gagal, sehingga loading pasti dimatikan. Tanpa finally, spinner bisa berputar selamanya jika terjadi error.
response.session == nullTanda bahwa Confirm email aktif: akun dibuat tetapi belum login.
_setLoading(true)Sekaligus menghapus error lama, agar pesan error percobaan sebelumnya tidak muncul lagi. Tidak ada update() atau pemberitahuan manual: mengubah .value sudah cukup.
signOut()Error saat logout hanya dicatat dengan debugPrint. signOut tetap menghapus sesi lokal walaupun request ke server gagal (misalnya offline), jadi user tetap keluar.

ℹ️ Controller tidak mengenal widget

Perhatikan import di atas: tidak ada material.dart, tidak ada BuildContext, tidak ada SnackBar atau navigasi. Controller hanya mengembalikan hasil (bool atau RegisterResult) dan menyimpan errorMessage. View-lah yang memutuskan cara menampilkannya. GetX sebenarnya menyediakan Get.snackbar yang bisa dipanggil dari controller, tetapi di kursus ini kita sengaja tidak melakukannya agar controller tetap mudah di-unit-test tanpa widget.

Pemetaan error ke pesan ramah

AuthException memiliki properti code berisi kode error dari server Supabase Auth. Kode-kode yang kita tangani:

💡 Kenapa "Email atau password salah", bukan "Email tidak terdaftar"?

Supabase sengaja tidak membedakan kedua kasus. Pesan yang terlalu spesifik membantu penyerang menebak email mana yang terdaftar. Pesan gabungan seperti ini adalah praktik keamanan standar.
