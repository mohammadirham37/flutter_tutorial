import 'package:flutter/material.dart';
import 'package:get/get.dart';

import '../controllers/auth_controller.dart';
import '../utils/validators.dart';
import 'register_view.dart';

class LoginView extends StatefulWidget {
  const LoginView({super.key});

  @override
  State<LoginView> createState() => _LoginViewState();
}

class _LoginViewState extends State<LoginView> {
  final _auth = Get.find<AuthController>();
  final _formKey = GlobalKey<FormState>();
  final _emailController = TextEditingController();
  final _passwordController = TextEditingController();
  bool _obscurePassword = true;

  @override
  void dispose() {
    _emailController.dispose();
    _passwordController.dispose();
    super.dispose();
  }

  Future<void> _submit() async {
    // Tutup keyboard lalu validasi form.
    FocusScope.of(context).unfocus();
    if (!_formKey.currentState!.validate()) return;

    final success = await _auth.signIn(
      email: _emailController.text,
      password: _passwordController.text,
    );

    // Jika sukses, AuthGate otomatis pindah ke HomeView.
    if (!success && mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(_auth.errorMessage ?? 'Login gagal'),
          backgroundColor: Theme.of(context).colorScheme.error,
        ),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: Form(
              key: _formKey,
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.stretch,
                children: [
                  Icon(
                    Icons.menu_book_rounded,
                    size: 72,
                    color: Theme.of(context).colorScheme.primary,
                  ),
                  const SizedBox(height: 16),
                  Text(
                    'Catatan Kuliah',
                    textAlign: TextAlign.center,
                    style: Theme.of(context).textTheme.headlineMedium,
                  ),
                  const SizedBox(height: 8),
                  const Text(
                    'Masuk untuk melihat catatan Anda',
                    textAlign: TextAlign.center,
                  ),
                  const SizedBox(height: 32),
                  TextFormField(
                    controller: _emailController,
                    keyboardType: TextInputType.emailAddress,
                    textInputAction: TextInputAction.next,
                    autofillHints: const [AutofillHints.email],
                    decoration: const InputDecoration(
                      labelText: 'Email',
                      prefixIcon: Icon(Icons.email_outlined),
                      border: OutlineInputBorder(),
                    ),
                    validator: Validators.email,
                  ),
                  const SizedBox(height: 16),
                  TextFormField(
                    controller: _passwordController,
                    obscureText: _obscurePassword,
                    textInputAction: TextInputAction.done,
                    onFieldSubmitted: (_) => _submit(),
                    decoration: InputDecoration(
                      labelText: 'Password',
                      prefixIcon: const Icon(Icons.lock_outline),
                      border: const OutlineInputBorder(),
                      suffixIcon: IconButton(
                        tooltip: _obscurePassword
                            ? 'Tampilkan password'
                            : 'Sembunyikan password',
                        icon: Icon(
                          _obscurePassword
                              ? Icons.visibility_outlined
                              : Icons.visibility_off_outlined,
                        ),
                        onPressed: () {
                          setState(() => _obscurePassword = !_obscurePassword);
                        },
                      ),
                    ),
                    validator: Validators.password,
                  ),
                  const SizedBox(height: 24),
                  // Hanya bagian tombol yang bergantung pada isLoading,
                  // jadi hanya bagian ini yang dibungkus Obx.
                  Obx(() {
                    final isLoading = _auth.isLoading;
                    return Column(
                      crossAxisAlignment: CrossAxisAlignment.stretch,
                      children: [
                        FilledButton(
                          onPressed: isLoading ? null : _submit,
                          style: FilledButton.styleFrom(
                            minimumSize: const Size.fromHeight(48),
                          ),
                          child: isLoading
                              ? const SizedBox(
                                  width: 22,
                                  height: 22,
                                  child: CircularProgressIndicator(
                                    strokeWidth: 2.5,
                                  ),
                                )
                              : const Text('Masuk'),
                        ),
                        const SizedBox(height: 12),
                        TextButton(
                          onPressed: isLoading
                              ? null
                              : () => Get.to(() => const RegisterView()),
                          child: const Text('Belum punya akun? Daftar di sini'),
                        ),
                      ],
                    );
                  }),
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }
}

Membedah LoginView

BagianPenjelasan
StatefulWidgetButuh state lokal: dua controller, GlobalKey<FormState>, dan _obscurePassword. Ini ephemeral state milik satu halaman, jadi cukup setState. State yang dipakai bersama (loading, sesi, pesan error) tetap ada di AuthController.
dispose()Controller wajib di-dispose agar tidak bocor memori.
FocusScope.of(context).unfocus()Menutup keyboard sebelum proses login, supaya SnackBar tidak tertutup keyboard.
_formKey.currentState!.validate()Menjalankan semua validator. Jika ada yang gagal, pesan merah muncul di bawah field dan proses berhenti.
final _auth = Get.find<AuthController>()Mengambil controller yang sudah didaftarkan dengan Get.put di main(). Karena LoginView adalah StatefulWidget (tidak bisa memakai GetView), controller disimpan sebagai field di State dan dipakai di _submit maupun build.
if (!success && mounted)Setelah await, widget mungkin sudah tidak ada. Saat login sukses, LoginView justru dibuang oleh AuthGate, sehingga pengecekan mounted wajib sebelum memakai context.
Obx(() { final isLoading = _auth.isLoading; ... })Hanya kedua tombol yang bergantung pada isLoading, jadi hanya bagian itu yang dibungkus Obx. Saat loading berubah, form dan field tidak ikut dibangun ulang.
onPressed: isLoading ? null : _submitonPressed: null membuat tombol nonaktif, sehingga mencegah klik ganda yang mengirim dua request login.
suffixIcon: IconButton(...)Tombol mata untuk menampilkan atau menyembunyikan password dengan membalik nilai obscureText.
onFieldSubmitted: (_) => _submit()Menekan tombol "selesai" di keyboard langsung memicu login.
Get.to(() => const RegisterView())Register dibuka di atas Login (setara Navigator.push), dan tombol back mengembalikan user ke Login. Tidak perlu meneruskan controller ke halaman baru: RegisterView cukup memanggil Get.find.

⚠️ Kesalahan umum

  • Membungkus Obx di sekitar widget yang tidak membaca nilai Rx apa pun: muncul error "[Get] the improper use of a GetX has been detected". Pastikan _auth.isLoading dibaca di dalam fungsi builder Obx, bukan di luar lalu dikirim sebagai variabel.
  • Memanggil Get.off(() => const HomeView()) setelah login: Home akan muncul dua kali (satu dari navigasi, satu dari AuthGate), dan logout jadi kacau.
  • Lupa cek mounted: muncul warning use_build_context_synchronously dan berpotensi crash.
  • Memanggil Get.find<AuthController>() sebelum Get.put: muncul error "AuthController not found". Pastikan Get.put di main() dijalankan sebelum runApp.
