AuthProvider menjembatani UI dan AuthService. Tugasnya: menyimpan status loading, menangkap exception, dan menerjemahkannya menjadi pesan Bahasa Indonesia yang ramah pengguna.

import 'package:flutter/foundation.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../services/auth_service.dart';

/// Hasil proses registrasi, dipakai RegisterScreen untuk menentukan
/// pesan yang ditampilkan.
enum RegisterResult { failed, loggedIn, needsEmailConfirmation }

class AuthProvider extends ChangeNotifier {
  AuthProvider(this._authService);

  final AuthService _authService;

  bool _isLoading = false;
  String? _errorMessage;

  bool get isLoading => _isLoading;
  String? get errorMessage => _errorMessage;

  Stream<AuthState> get authStateChanges => _authService.authStateChanges;
  Session? get currentSession => _authService.currentSession;
  User? get currentUser => _authService.currentUser;

  /// Nama yang ditampilkan di AppBar. Diambil dari metadata saat register.
  String get displayName {
    final user = currentUser;
    if (user == null) return '';
    final name = user.userMetadata?['full_name'];
    if (name is String && name.isNotEmpty) return name;
    return user.email ?? '';
  }

  Future<bool> signIn({required String email, required String password}) async {
    _setLoading(true);
    try {
      await _authService.signIn(email: email.trim(), password: password);
      return true;
    } on AuthException catch (e) {
      _errorMessage = _mapAuthError(e);
      return false;
    } catch (e) {
      _errorMessage = 'Terjadi kesalahan tak terduga: $e';
      return false;
    } finally {
      _setLoading(false);
    }
  }

  Future<RegisterResult> register({
    required String fullName,
    required String email,
    required String password,
  }) async {
    _setLoading(true);
    try {
      final response = await _authService.signUp(
        fullName: fullName.trim(),
        email: email.trim(),
        password: password,
      );
      // Jika "Confirm email" aktif di dashboard, session bernilai null
      // sampai pengguna mengeklik link di email.
      if (response.session == null) {
        return RegisterResult.needsEmailConfirmation;
      }
      return RegisterResult.loggedIn;
    } on AuthException catch (e) {
      _errorMessage = _mapAuthError(e);
      return RegisterResult.failed;
    } catch (e) {
      _errorMessage = 'Terjadi kesalahan tak terduga: $e';
      return RegisterResult.failed;
    } finally {
      _setLoading(false);
    }
  }

  Future<void> signOut() async {
    try {
      await _authService.signOut();
    } on AuthException catch (e) {
      // signOut tetap menghapus sesi lokal walau request ke server gagal.
      debugPrint('signOut: ${e.message}');
    }
  }

  void clearError() {
    _errorMessage = null;
    notifyListeners();
  }

  void _setLoading(bool value) {
    _isLoading = value;
    if (value) _errorMessage = null;
    notifyListeners();
  }

  /// Menerjemahkan error Supabase Auth menjadi pesan berbahasa Indonesia.
  String _mapAuthError(AuthException e) {
    if (e is AuthRetryableFetchException) {
      return 'Tidak dapat terhubung ke server. Periksa koneksi internet.';
    }
    switch (e.code) {
      case 'invalid_credentials':
        return 'Email atau password salah.';
      case 'email_not_confirmed':
        return 'Email belum dikonfirmasi. Buka link verifikasi di inbox Anda.';
      case 'user_already_exists':
      case 'email_exists':
        return 'Email sudah terdaftar. Silakan login.';
      case 'weak_password':
        return 'Password terlalu lemah. Gunakan minimal 6 karakter.';
      case 'email_address_invalid':
        return 'Alamat email tidak valid.';
      case 'over_email_send_rate_limit':
      case 'over_request_rate_limit':
        return 'Terlalu banyak percobaan. Tunggu beberapa menit lalu coba lagi.';
      case 'signup_disabled':
        return 'Pendaftaran akun baru sedang dinonaktifkan.';
    }
    return e.message;
  }
}

BagianPenjelasan
enum RegisterResultRegister punya tiga kemungkinan hasil: gagal, langsung login, atau perlu konfirmasi email. Enum membuat UI bisa memakai switch yang jelas, alih-alih bool yang ambigu.
_isLoading + _errorMessageState privat dengan getter publik (pola Modul 08). UI membaca isLoading untuk menonaktifkan tombol dan menampilkan spinner.
displayNameMengambil full_name dari userMetadata, dengan email sebagai cadangan jika nama kosong.
try / on AuthException / catch / finallyfinally selalu dijalankan, baik sukses maupun gagal, sehingga loading pasti dimatikan. Tanpa finally, spinner bisa berputar selamanya jika terjadi error.
response.session == nullTanda bahwa Confirm email aktif: akun dibuat tetapi belum login.
_setLoading(true)Sekaligus menghapus error lama, agar pesan error percobaan sebelumnya tidak muncul lagi.
signOut()Error saat logout hanya dicatat. signOut tetap menghapus sesi lokal walaupun request ke server gagal (misalnya offline), jadi user tetap keluar.

Pemetaan error ke pesan ramah

AuthException memiliki properti code berisi kode error dari server Supabase Auth. Kode-kode yang kita tangani:

💡 Kenapa "Email atau password salah", bukan "Email tidak terdaftar"?

Supabase sengaja tidak membedakan kedua kasus. Pesan yang terlalu spesifik membantu penyerang menebak email mana yang terdaftar. Pesan gabungan seperti ini adalah praktik keamanan standar.
